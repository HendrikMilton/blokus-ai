@dataclass
class TrainingConfig:
    """Configuration for AlphaZero training."""
    # Self-play
    games_per_generation: int = 100
    mcts_simulations: int = 400
    mcts_c_puct: float = 1.5
    temperature_threshold: int = 20
    dirichlet_alpha: float = 0.3
    dirichlet_epsilon: float = 0.25
    num_parallel_games: int = 8

    # Batching (for GPU efficiency)
    mcts_batch_size: int = 16        # Leaves to batch per MCTS search
    inference_batch_size: int = 32   # Max positions per GPU batch across games
    use_batched_mcts: bool = True    # Use batched MCTS (vs sequential)
    use_parallel_games: bool = True  # Run multiple games in parallel

    # Training
    batch_size: int = 4096
    learning_rate: float = 0.002
    lr_decay: str = 'cosine'
    training_steps_per_gen: int = 2000  # Used if adaptive is disabled
    training_steps_min: int = 500       # Adaptive: min steps (small buffer)
    training_steps_max: int = 2000      # Adaptive: max steps (full buffer)
    adaptive_training_steps: bool = True  # Enable adaptive scaling
    replay_buffer_size: int = 1_000_000
    mixed_precision: bool = True

    # Gradient clipping (prevents explosion)
    max_grad_norm: float = 1.0       # Max gradient norm for clipping

    # Cold start fixes
    # 1. Heuristic bootstrapping: blend handcrafted policy with NN
    use_heuristic_bootstrap: bool = True
    heuristic_blend_initial: float = 0.5   # Initial blend (0=pure NN, 1=pure heuristic)
    heuristic_blend_decay_gens: int = 30   # Generations to decay to 0

    # 2. Increased MCTS simulations for early training
    mcts_simulations_initial: int = 1600   # Higher sims for first N generations
    mcts_simulations_decay_gens: int = 20  # Decay to base mcts_simulations over this many gens

    # 3. Temperature-adjusted targets for sharper policies
    policy_target_temperature: float = 0.5  # <1.0 makes MCTS targets more peaked

    # Network
    num_res_blocks: int = 15
    channels: int = 256

    # Evaluation
    eval_games: int = 100
    promotion_threshold: float = 0.55

    # === Anti-Collapse Fixes ===

    # 1. Entropy regularization (prevents policy collapse)
    entropy_regularization: bool = True
    entropy_weight: float = 0.01          # Base entropy bonus weight
    entropy_target: float = 1.0           # Target entropy level
    entropy_adaptive: bool = True         # Increase weight when entropy too low

    # 2. Dynamic promotion threshold
    dynamic_promotion: bool = True
    promotion_threshold_min: float = 0.45   # Minimum threshold
    promotion_decay_per_stuck_gen: float = 0.02  # Lower by 2% per stuck generation

    # 3. Recent-weighted replay sampling
    replay_recent_bias: float = 0.7  # Bias toward recent examples (0=uniform, 1=strong)

    # 4. Adaptive Dirichlet noise
    adaptive_noise: bool = True
    noise_entropy_threshold: float = 0.5  # Increase noise when entropy below this
    noise_multiplier_max: float = 2.0     # Max multiplier for noise when entropy low

    # Limits
    max_generations: int = 100

    # Heuristic pretraining (Phase 0 - before self-play)
    use_heuristic_pretraining: bool = True
    pretrain_positions: int = 10000
    pretrain_steps: int = 1000
    pretrain_batch_size: int = 256
    pretrain_lr: float = 0.001

class AlphaZeroTrainer:
    """
    Main training loop for AlphaZero.
    """

    def __init__(
        self,
        run_id: str,
        config: TrainingConfig,
        models_dir: Path,
        api_base_url: str,
        device: str = 'cuda',
    ):
        self.run_id = run_id
        self.config = config
        self.models_dir = models_dir
        self.api_base_url = api_base_url
        self.device = device

        # Create model directory
        self.run_dir = models_dir / 'alphazero' / run_id
        self.run_dir.mkdir(parents=True, exist_ok=True)

        # Initialize network
        self.network = create_network(
            num_res_blocks=config.num_res_blocks,
            channels=config.channels,
            device=device,
        )

        # Best network (for evaluation)
        self.best_network = create_network(
            num_res_blocks=config.num_res_blocks,
            channels=config.channels,
            device=device,
        )
        # Copy initial weights
        self.best_network.load_state_dict(self.network.state_dict())

        # Optimizer
        self.optimizer = optim.Adam(self.network.parameters(), lr=config.learning_rate)

        # Learning rate scheduler
        if config.lr_decay == 'cosine':
            total_steps = config.max_generations * config.training_steps_per_gen
            self.scheduler = optim.lr_scheduler.CosineAnnealingLR(
                self.optimizer, T_max=total_steps
            )
        else:
            self.scheduler = None

        # Mixed precision
        self.scaler = GradScaler('cuda') if config.mixed_precision else None

        # Replay buffer
        self.replay_buffer = ReplayBuffer(max_size=config.replay_buffer_size)

        # MCTS config
        self.mcts_config = MCTSConfig(
            num_simulations=config.mcts_simulations,
            c_puct=config.mcts_c_puct,
            temperature_threshold=config.temperature_threshold,
            dirichlet_alpha=config.dirichlet_alpha,
            dirichlet_epsilon=config.dirichlet_epsilon,
        )

        # Training state
        self.current_generation = 0
        self.total_games = 0
        self.total_steps = 0
        self.best_win_rate = 0.0
        self._stop_requested = False

        # Heuristic config for policy bootstrapping
        self.heuristic_config = HeuristicConfig() if config.use_heuristic_bootstrap else None

        # Timing for ETA calculation
        self.training_start_time: Optional[float] = None
        self.generation_start_time: Optional[float] = None
        self.phase_start_time: Optional[float] = None
        self.games_times: List[float] = []  # Track time per game for better ETA

        # Track actual phase durations from completed generations for accurate ETA
        self.completed_gen_durations: List[float] = []  # Total time per completed generation
        self.phase_durations: Dict[str, List[float]] = {
            'self_play': [],
            'training': [],
            'evaluation': [],
        }
        self.current_phase_durations: Dict[str, float] = {}  # Durations within current generation

        # Dynamic promotion threshold state
        self.generations_since_promotion = 0
        self.current_promotion_threshold = config.promotion_threshold

    def _report_progress(
        self,
        phase: str,
        phase_progress: float,
        phase_current: int = 0,
        phase_total: int = 0,
        policy_loss: float = 0.0,
        value_loss: float = 0.0,
        message: str = '',
        extra: Optional[Dict[str, Any]] = None,
    ):
        """Report detailed progress to API with ETA."""
        import httpx

        now = time.time()

        # Calculate phase ETA
        phase_eta_sec = 0.0
        phase_elapsed = 0.0
        if self.phase_start_time and phase_progress > 0:
            phase_elapsed = now - self.phase_start_time
            phase_eta_sec = (phase_elapsed / phase_progress) * (1 - phase_progress)

        # Calculate generation ETA based on actual measured durations
        gen_eta_sec = 0.0
        gen_progress = 0.0
        gen_elapsed = now - self.generation_start_time if self.generation_start_time else 0.0

        if self.generation_start_time:
            # Get phase weights from actual measurements or use defaults
            if self.completed_gen_durations:
                # Use actual measured durations from completed generations
                avg_self_play = sum(self.phase_durations['self_play']) / len(self.phase_durations['self_play']) if self.phase_durations['self_play'] else 0
                avg_training = sum(self.phase_durations['training']) / len(self.phase_durations['training']) if self.phase_durations['training'] else 0
                avg_eval = sum(self.phase_durations['evaluation']) / len(self.phase_durations['evaluation']) if self.phase_durations['evaluation'] else 0
                avg_total = avg_self_play + avg_training + avg_eval

                if avg_total > 0:
                    w_self_play = avg_self_play / avg_total
                    w_training = avg_training / avg_total
                    w_eval = avg_eval / avg_total
                else:
                    w_self_play, w_training, w_eval = 0.6, 0.3, 0.1
            else:
                # Default weights for first generation
                w_self_play, w_training, w_eval = 0.6, 0.3, 0.1

            # Calculate gen_progress based on completed phases + current phase progress
            completed_weight = 0.0
            for p, w in [('self_play', w_self_play), ('training', w_training), ('evaluation', w_eval)]:
                if p in self.current_phase_durations:
                    completed_weight += w
                elif p == phase:
                    completed_weight += phase_progress * w
                    break

            gen_progress = completed_weight

            if gen_progress > 0:
                gen_eta_sec = (gen_elapsed / gen_progress) * (1 - gen_progress)

        # Calculate overall ETA based on completed generation times
        overall_eta_sec = 0.0
        overall_progress = 0.0
        if self.training_start_time and self.current_generation > 0:
            completed_gens = self.current_generation - 1
            total_gens = self.config.max_generations

            if self.completed_gen_durations:
                # Use actual average generation time for remaining generations
                avg_gen_time = sum(self.completed_gen_durations) / len(self.completed_gen_durations)
                remaining_full_gens = total_gens - self.current_generation
                # Time remaining = remaining full generations * avg + current gen remaining
                overall_eta_sec = remaining_full_gens * avg_gen_time + gen_eta_sec
                overall_progress = (completed_gens + gen_progress) / total_gens
            else:
                # First generation: estimate based on gen_eta_sec
                # We don't have completed generations yet, so extrapolate from current gen
                overall_progress = gen_progress / total_gens
                remaining_full_gens = total_gens - 1  # Excluding current generation
                # Estimate: current gen remaining + (remaining gens * estimated gen time)
                if gen_progress > 0:
                    estimated_gen_time = gen_elapsed / gen_progress  # Estimated total time for this gen
                    overall_eta_sec = gen_eta_sec + remaining_full_gens * estimated_gen_time
                else:
                    overall_eta_sec = 0.0

        try:
            url = f"{self.api_base_url}/api/ml/alphazero/runs/{self.run_id}/progress"
            payload = {
                # Current state
                'generation': self.current_generation,
                'maxGenerations': self.config.max_generations,
                'gamesGenerated': self.total_games,
                'trainingSteps': self.total_steps,
                'policyLoss': policy_loss if policy_loss else None,
                'valueLoss': value_loss if value_loss else None,
                # Phase info
                'phase': phase,
                'phaseCurrent': phase_current,
                'phaseTotal': phase_total,
                'phaseProgress': round(phase_progress * 100, 1),
                'phaseEtaSec': round(phase_eta_sec),
                # Generation info
                'generationProgress': round(gen_progress * 100, 1),
                'generationEtaSec': round(gen_eta_sec),
                # Overall info
                'overallProgress': round(overall_progress * 100, 1),
                'overallEtaSec': round(overall_eta_sec),
                # Human-readable message
                'message': message,
                # Extra data (e.g., games per second, etc.)
                **(extra or {}),
            }
            httpx.post(url, json=payload, timeout=5.0)
        except Exception as e:
            log.info(f"Failed to report progress: {e}")

    def stop(self):
        """Request training to stop."""
        self._stop_requested = True

    def _record_phase_duration(self, phase: str):
        """Record the duration of a completed phase."""
        if self.phase_start_time:
            duration = time.time() - self.phase_start_time
            self.current_phase_durations[phase] = duration
            self.phase_durations[phase].append(duration)
            log.info(f"Phase {phase} completed in {duration:.1f}s")

    def _record_generation_complete(self):
        """Record total duration of completed generation."""
        if self.generation_start_time:
            duration = time.time() - self.generation_start_time
            self.completed_gen_durations.append(duration)
            log.info(f"Generation {self.current_generation} completed in {duration:.1f}s")

    def _get_mcts_simulations(self) -> int:
        """
        Get MCTS simulations for current generation.

        Starts high (mcts_simulations_initial) and decays to base (mcts_simulations)
        over the first mcts_simulations_decay_gens generations.
        """
        if self.current_generation >= self.config.mcts_simulations_decay_gens:
            return self.config.mcts_simulations

        # Linear decay from initial to base
        progress = self.current_generation / self.config.mcts_simulations_decay_gens
        sims = int(
            self.config.mcts_simulations_initial * (1 - progress) +
            self.config.mcts_simulations * progress
        )
        return sims

    def _get_heuristic_blend_factor(self) -> float:
        """
        Get heuristic blend factor for current generation.

        Returns 0.0 if heuristic bootstrapping is disabled.
        """
        if not self.config.use_heuristic_bootstrap:
            return 0.0

        return get_blend_factor_for_generation(
            generation=self.current_generation,
            initial_blend=self.config.heuristic_blend_initial,
            decay_generations=self.config.heuristic_blend_decay_gens,
            min_blend=0.0,
        )

    def _update_promotion_threshold(self, was_promoted: bool):
        """Update promotion threshold based on stuck generations."""
        if not self.config.dynamic_promotion:
            return

        if was_promoted:
            # Reset to base threshold after promotion
            self.generations_since_promotion = 0
            self.current_promotion_threshold = self.config.promotion_threshold
        else:
            # Lower threshold if stuck
            self.generations_since_promotion += 1
            if self.generations_since_promotion >= 2:  # After 2 stuck generations
                new_threshold = self.current_promotion_threshold - self.config.promotion_decay_per_stuck_gen
                self.current_promotion_threshold = max(new_threshold, self.config.promotion_threshold_min)
                log.info(f"Lowered promotion threshold to {self.current_promotion_threshold:.2%} "
                        f"(stuck for {self.generations_since_promotion} generations)")

    def _get_noise_multiplier(self) -> float:
        """
        Calculate noise multiplier based on current policy entropy.

        When entropy is low (policy is too confident/collapsed), increase noise
        to force more exploration.
        """
        if not self.config.adaptive_noise or not self._last_heuristics:
            return 1.0

        current_entropy = self._last_heuristics.get('policy_entropy', 2.0)
        if current_entropy < self.config.noise_entropy_threshold:
            noise_mult = min(
                self.config.noise_multiplier_max,
                self.config.noise_entropy_threshold / max(current_entropy, 0.1)
            )
            log.info(f"Adaptive noise: entropy={current_entropy:.3f} < threshold={self.config.noise_entropy_threshold}, "
                    f"noise_multiplier={noise_mult:.2f}")
            return noise_mult
        return 1.0

    def _run_heuristic_pretraining(self):
        """Phase 0: Pretrain policy head on heuristic-scored positions."""
        from .pretraining import pretrain_policy_head

        log.info("=" * 60)
        log.info("PHASE 0: Heuristic Pretraining")
        log.info("=" * 60)

        self._report_progress(
            phase='pretraining',
            phase_progress=0.0,
            message='Generating training positions...',
        )

        def progress_cb(step: int, total: int, loss: float):
            self._report_progress(
                phase='pretraining',
                phase_progress=step / total,
                phase_current=step,
                phase_total=total,
                message=f'Step {step}/{total} - loss: {loss:.4f}',
            )

        result = pretrain_policy_head(
            network=self.network,
            device=self.device,
            num_positions=self.config.pretrain_positions,
            num_steps=self.config.pretrain_steps,
            batch_size=self.config.pretrain_batch_size,
            learning_rate=self.config.pretrain_lr,
            progress_callback=progress_cb,
        )

        log.info(f"Pretraining complete! Avg loss: {result['avg_loss']:.4f}")

        self._report_progress(
            phase='pretraining',
            phase_progress=1.0,
            message='Pretraining complete',
        )

        return result

    def train(self) -> bool:
        """
        Run the full training loop.

        Returns:
            True if completed successfully, False if stopped/failed
        """
        log.info(f"Starting training run: {self.run_id}")
        log.info(f"Config: {self.config}")

        self.training_start_time = time.time()

        try:
            # Phase 0: Heuristic pretraining (before self-play)
            if self.config.use_heuristic_pretraining:
                self._run_heuristic_pretraining()

                if self._stop_requested:
                    log.info("Stop requested during pretraining, exiting")
                    return False

            for gen in range(self.config.max_generations):
                if self._stop_requested:
                    log.info("Stop requested, exiting")
                    return False

                self.current_generation = gen + 1
                self.generation_start_time = time.time()
                self.current_phase_durations = {}  # Reset for new generation
                log.info(f"=== Generation {self.current_generation}/{self.config.max_generations} ===")

                # Phase 1: Self-play (move best_network to CPU to save VRAM)
                log.info("Phase 1: Self-play")
                self.best_network.cpu()
                torch.cuda.empty_cache()
                self._run_self_play()
                self._record_phase_duration('self_play')

                if self._stop_requested:
                    return False

                # Clear GPU memory before training phase
                import gc
                import ctypes

                # Move network to CPU temporarily to force CUDA to release memory
                self.network.cpu()

                # Multiple rounds of cleanup to ensure all references are freed
                for _ in range(3):
                    gc.collect()
                torch.cuda.empty_cache()
                torch.cuda.synchronize()

                # Check memory before continuing
                allocated = torch.cuda.memory_allocated() / 1e9
                reserved = torch.cuda.memory_reserved() / 1e9
                log.info(f"GPU memory after initial cleanup: allocated={allocated:.2f}GB, reserved={reserved:.2f}GB")

                # If still using significant memory, try harder
                if reserved > 1.0:
                    log.info("Memory still high, forcing additional cleanup...")
                    torch.cuda.reset_peak_memory_stats()
                    torch.cuda.empty_cache()
                    gc.collect()

                    # Release memory back to OS (Linux glibc malloc_trim)
                    try:
                        libc = ctypes.CDLL("libc.so.6")
                        libc.malloc_trim(0)
                    except Exception:
                        pass

                    allocated = torch.cuda.memory_allocated() / 1e9
                    reserved = torch.cuda.memory_reserved() / 1e9
                    log.info(f"GPU memory after aggressive cleanup: allocated={allocated:.2f}GB, reserved={reserved:.2f}GB")

                # Move network back to GPU for training
                self.network.to(self.device)

                # Phase 2: Training
                log.info("Phase 2: Training")
                policy_loss, value_loss = self._run_training()
                self._record_phase_duration('training')

                if self._stop_requested:
                    return False

                # Phase 3: Evaluation (move best_network back to GPU)
                log.info("Phase 3: Evaluation")
                self.best_network.to(self.device)
                win_rate, is_best = self._run_evaluation()
                self._record_phase_duration('evaluation')
                self._record_generation_complete()

                # Track best win rate (regardless of promotion)
                if win_rate > self.best_win_rate:
                    self.best_win_rate = win_rate

                # Save checkpoint
                model_path = self._save_checkpoint(policy_loss, value_loss, win_rate, is_best)
                log.info(f"Checkpoint saved: {model_path}")

                if is_best:
                    # Update best network (promoted)
                    self.best_network.load_state_dict(self.network.state_dict())
                    log.info(f"New best model! Win rate: {win_rate:.2%}")

            # Training completed
            log.info("Training completed!")
            self._report_completion(policy_loss, value_loss, self.best_win_rate)
            return True

        except Exception as e:
            import traceback
            error_msg = f"{str(e)}\n{traceback.format_exc()}"
            log.info(f"Training failed: {error_msg}")
            self._report_failure(error_msg)
            return False

    def _run_self_play(self):
        """Generate self-play games using batched MCTS for GPU efficiency."""
        import sys
        self.phase_start_time = time.time()
        self.games_times = []

        # Choose between batched parallel or legacy sequential
        use_batched = self.config.use_batched_mcts
        use_parallel = self.config.use_parallel_games

        # Get dynamic values for cold start fixes
        current_mcts_sims = self._get_mcts_simulations()
        current_blend_factor = self._get_heuristic_blend_factor()

        log.info(f"Starting self-play: {self.config.games_per_generation} games, "
              f"{current_mcts_sims} MCTS sims (base={self.config.mcts_simulations}), "
              f"heuristic_blend={current_blend_factor:.3f}, "
              f"policy_temp={self.config.policy_target_temperature}, "
              f"batched={use_batched}, parallel={use_parallel}, "
              f"mcts_batch={self.config.mcts_batch_size}, "
              f"parallel_games={self.config.num_parallel_games}")

        last_game_time = time.time()
        last_intermediate_report = time.time()

        # Collect metrics from all games for aggregated logging
        game_metrics_list = []

        def progress_callback(completed, total, game_result=None, **kwargs):
            nonlocal last_game_time, last_intermediate_report
            now = time.time()

            # Handle intermediate progress updates (during game, before completion)
            if kwargs.get('intermediate'):
                moves_made = kwargs.get('moves_made', 0)
                moves_per_sec = kwargs.get('moves_per_sec', 0)
                active_games = kwargs.get('active_games', 0)

                # Estimate progress based on moves (assume ~70 moves per game)
                estimated_total_moves = total * 70
                move_progress = min(moves_made / estimated_total_moves, 0.95) if estimated_total_moves > 0 else 0

                message = f"Playing... {moves_made} moves ({moves_per_sec:.1f}/s), {active_games} active games"

                self._report_progress(
                    phase='self_play',
                    phase_progress=move_progress,
                    phase_current=completed,
                    phase_total=total,
                    message=message,
                    extra={
                        'movesPerSec': round(moves_per_sec, 2),
                        'totalMoves': moves_made,
                        'activeGames': active_games,
                        'batchedMcts': use_batched,
                        'parallelGames': use_parallel,
                    }
                )
                return

            # Collect metrics from completed game
            if game_result and game_result.metrics:
                game_metrics_list.append(game_result.metrics)

            # Track game completion time
            game_time = now - last_game_time
            self.games_times.append(game_time)
            last_game_time = now

            self.total_games = (self.current_generation - 1) * self.config.games_per_generation + completed

            # Calculate games per second (average of last 5 games)
            recent_times = self.games_times[-5:] if len(self.games_times) >= 5 else self.games_times
            avg_game_time = sum(recent_times) / len(recent_times) if recent_times else 0
            games_per_min = 60 / avg_game_time if avg_game_time > 0 else 0

            # Build message
            game_info = ""
            if game_result:
                game_info = f" ({game_result.num_moves} moves, winner={game_result.winner})"

            message = f"Game {completed}/{total}{game_info} - {games_per_min:.1f} games/min"

            self._report_progress(
                phase='self_play',
                phase_progress=completed / total,
                phase_current=completed,
                phase_total=total,
                message=message,
                extra={
                    'gamesPerMin': round(games_per_min, 2),
                    'avgGameTimeSec': round(avg_game_time, 1),
                    'lastGameMoves': game_result.num_moves if game_result else None,
                    'lastGameExamples': len(game_result.examples) if game_result else None,
                    'batchedMcts': use_batched,
                    'parallelGames': use_parallel,
                }
            )

        # Create generation-specific MCTS config with dynamic simulations
        gen_mcts_config = MCTSConfig(
            num_simulations=current_mcts_sims,
            c_puct=self.config.mcts_c_puct,
            temperature_threshold=self.config.temperature_threshold,
            dirichlet_alpha=self.config.dirichlet_alpha,
            dirichlet_epsilon=self.config.dirichlet_epsilon,
        )

        # Calculate adaptive noise multiplier based on entropy
        noise_multiplier = self._get_noise_multiplier()

        if use_batched:
            # Use new ParallelSelfPlay with batched MCTS
            manager = ParallelSelfPlay(
                network=self.network,
                mcts_config=gen_mcts_config,
                num_parallel_games=self.config.num_parallel_games,
                mcts_batch_size=self.config.mcts_batch_size,
                inference_batch_size=self.config.inference_batch_size,
                device=self.device,
                # Cold start fixes
                heuristic_config=self.heuristic_config if current_blend_factor > 0 else None,
                heuristic_blend_factor=current_blend_factor,
                policy_target_temperature=self.config.policy_target_temperature,
                # Adaptive noise
                noise_multiplier=noise_multiplier,
            )

            if use_parallel:
                # Full parallelization - multiple games with shared batching
                examples = manager.generate_games_fully_parallel(
                    num_games=self.config.games_per_generation,
                    progress_callback=progress_callback,
                )
            else:
                # Sequential games but batched MCTS within each game
                examples = manager.generate_games(
                    num_games=self.config.games_per_generation,
                    progress_callback=progress_callback,
                )
        else:
            # Legacy sequential MCTS (for comparison/debugging)
            manager = SelfPlayManager(
                network=self.network,
                mcts_config=gen_mcts_config,
                num_workers=self.config.num_parallel_games,
                device=self.device,
            )
            examples = manager.generate_games(
                num_games=self.config.games_per_generation,
                progress_callback=progress_callback,
            )

        self.replay_buffer.add(examples)
        elapsed = time.time() - self.phase_start_time
        games_per_min = (self.config.games_per_generation / elapsed) * 60 if elapsed > 0 else 0
        log.info(f"Generated {len(examples)} examples from {self.config.games_per_generation} games "
              f"in {elapsed:.1f}s ({games_per_min:.1f} games/min)")
        log.info(f"Replay buffer size: {len(self.replay_buffer)}")

        # Log aggregated heuristic metrics for diagnostics
        # Store on instance for use in training metrics
        self._last_heuristics = None
        if game_metrics_list:
            avg_corners = sum(m.avg_corner_count for m in game_metrics_list) / len(game_metrics_list)
            avg_mobility = sum(m.avg_mobility for m in game_metrics_list) / len(game_metrics_list)
            avg_center = sum(m.avg_center_coverage for m in game_metrics_list) / len(game_metrics_list)

            # Policy quality metrics (for diagnosing cold start problem)
            avg_entropy = sum(m.avg_policy_entropy for m in game_metrics_list) / len(game_metrics_list)
            avg_top1 = sum(m.avg_top1_prob for m in game_metrics_list) / len(game_metrics_list)
            avg_top5 = sum(m.avg_top5_prob for m in game_metrics_list) / len(game_metrics_list)

            # Store for training metrics
            self._last_heuristics = {
                'avg_corners': avg_corners,
                'avg_mobility': avg_mobility,
                'center_coverage': avg_center,
                'policy_entropy': avg_entropy,
                'policy_top1': avg_top1,
                'policy_top5': avg_top5,
            }

            # Average final corners per player across all games
            avg_final_corners = [
                sum(m.final_corner_counts[p] for m in game_metrics_list) / len(game_metrics_list)
                for p in range(4)
            ]

            # Log policy quality - this is key for diagnosing learning issues
            # High entropy (~4.0+) means MCTS is generating nearly uniform targets (bad)
            # Low entropy (<2.0) means MCTS is generating focused targets (good)
            log.info(f"[Heuristics] Gen {self.current_generation}: "
                  f"avg_corners={avg_corners:.1f}, avg_mobility={avg_mobility:.1f}, "
                  f"center_coverage={avg_center:.2%}, final_corners={[f'{c:.1f}' for c in avg_final_corners]}")
            log.info(f"[Policy Quality] Gen {self.current_generation}: "
                  f"entropy={avg_entropy:.3f}, top1={avg_top1:.3f}, top5={avg_top5:.3f} "
                  f"(low entropy + high top1/top5 = focused policies = good learning signal)")

            # Report metrics via progress API for frontend display
            self._report_progress(
                phase='self_play',
                phase_progress=1.0,
                phase_current=self.config.games_per_generation,
                phase_total=self.config.games_per_generation,
                message='Self-play complete',
                extra={
                    'heuristics': {
                        'avgCorners': round(avg_corners, 2),
                        'avgMobility': round(avg_mobility, 2),
                        'centerCoverage': round(avg_center, 4),
                        'finalCorners': [round(c, 2) for c in avg_final_corners],
                    },
                    'policyQuality': {
                        'entropy': round(avg_entropy, 4),
                        'top1Prob': round(avg_top1, 4),
                        'top5Prob': round(avg_top5, 4),
                    }
                }
            )

        # Clean up self-play objects to free memory before training
        del examples
        del manager

        # Force cleanup of any cached tensors
        import gc
        gc.collect()
        torch.cuda.empty_cache()

    def _get_training_steps(self) -> int:
        """Calculate training steps based on buffer size (adaptive) or fixed config."""
        if not self.config.adaptive_training_steps:
            return self.config.training_steps_per_gen

        # Scale steps linearly with buffer fullness
        # At 0% full -> training_steps_min
        # At 100% full -> training_steps_max
        buffer_ratio = min(len(self.replay_buffer) / self.config.replay_buffer_size, 1.0)
        steps = int(
            self.config.training_steps_min +
            buffer_ratio * (self.config.training_steps_max - self.config.training_steps_min)
        )
        return steps

    def _run_training(self) -> tuple[float, float]:
        """Train the network on replay buffer."""
        self.phase_start_time = time.time()
        self.network.train()

        total_policy_loss = 0.0
        total_value_loss = 0.0

        # Calculate adaptive training steps
        training_steps = self._get_training_steps()
        log.info(f"Training steps this generation: {training_steps} "
                 f"(buffer: {len(self.replay_buffer)}/{self.config.replay_buffer_size})")

        # Collect metrics for batch insert to database
        metrics_batch = []
        last_step_time = time.time()

        for step in range(training_steps):
            if self._stop_requested:
                break

            # Sample batch
            batch = self.replay_buffer.sample(self.config.batch_size, self.config.replay_recent_bias)
            if len(batch) < self.config.batch_size // 2 and step == 0:
                log.info(f"Warning: only {len(batch)} examples in buffer (batch_size={self.config.batch_size})")

            # Prepare batch tensors - stack numpy arrays first, then convert once
            board_np = np.stack([ex.board_tensor for ex in batch])
            pieces_np = np.stack([ex.pieces_vector for ex in batch])
            values_np = np.array([ex.value for ex in batch], dtype=np.float32)

            board_tensors = torch.from_numpy(board_np).to(self.device)
            pieces_vectors = torch.from_numpy(pieces_np).to(self.device)
            target_values = torch.from_numpy(values_np).unsqueeze(1).to(self.device)

            # Legal moves list for policy
            legal_moves_list = [ex.legal_moves for ex in batch]

            # Pre-compute target policies - stack numpy then convert
            max_policy_len = max(len(ex.policy) for ex in batch) if batch else 1
            target_np = np.zeros((len(batch), max_policy_len), dtype=np.float32)
            mask_np = np.zeros((len(batch), max_policy_len), dtype=np.float32)
            for i, ex in enumerate(batch):
                plen = len(ex.policy)
                if plen > 0:
                    target_np[i, :plen] = ex.policy
                    mask_np[i, :plen] = 1.0

            target_policies = torch.from_numpy(target_np).to(self.device)
            policy_mask = torch.from_numpy(mask_np).to(self.device)

            self.optimizer.zero_grad()

            if self.config.mixed_precision:
                with autocast('cuda'):
                    policy_logits, values = self.network(
                        board_tensors, pieces_vectors, legal_moves_list
                    )
                    value_loss = F.mse_loss(values, target_values)

                    # Vectorized policy loss - no Python loop!
                    # Mask out invalid positions with -inf before softmax
                    masked_logits = policy_logits[:, :max_policy_len].clone()
                    masked_logits[policy_mask == 0] = float('-inf')
                    log_probs = F.log_softmax(masked_logits, dim=1)
                    # Replace NaN with 0 (can happen if all logits are -inf)
                    log_probs = torch.nan_to_num(log_probs, nan=0.0)
                    # Cross-entropy: -sum(target * log_probs) per sample, then mean
                    policy_loss = -(target_policies * log_probs).sum(dim=1).mean()

                    # Entropy regularization: prevent policy collapse
                    if self.config.entropy_regularization:
                        # Compute policy entropy: -sum(p * log(p))
                        probs = F.softmax(masked_logits, dim=1)
                        log_probs_for_entropy = torch.log(probs + 1e-10)
                        entropy = -(probs * log_probs_for_entropy).sum(dim=1).mean()

                        # Adaptive weight: increase if entropy too low
                        if self.config.entropy_adaptive and self._last_heuristics:
                            current_entropy = self._last_heuristics.get('policy_entropy', 1.0)
                            if current_entropy < self.config.entropy_target:
                                # Scale up weight when entropy is low
                                adaptive_weight = self.config.entropy_weight * (self.config.entropy_target / max(current_entropy, 0.1))
                            else:
                                adaptive_weight = self.config.entropy_weight
                        else:
                            adaptive_weight = self.config.entropy_weight

                        # Add entropy bonus (negative because we want to maximize entropy)
                        loss = policy_loss + value_loss - adaptive_weight * entropy
                    else:
                        loss = policy_loss + value_loss

                self.scaler.scale(loss).backward()
                # Unscale gradients before clipping (required for mixed precision)
                self.scaler.unscale_(self.optimizer)
                # Gradient clipping to prevent explosion
                torch.nn.utils.clip_grad_norm_(
                    self.network.parameters(), self.config.max_grad_norm
                )
                self.scaler.step(self.optimizer)
                self.scaler.update()
            else:
                policy_logits, values = self.network(
                    board_tensors, pieces_vectors, legal_moves_list
                )
                value_loss = F.mse_loss(values, target_values)

                # Vectorized policy loss - no Python loop!
                masked_logits = policy_logits[:, :max_policy_len].clone()
                masked_logits[policy_mask == 0] = float('-inf')
                log_probs = F.log_softmax(masked_logits, dim=1)
                # Replace NaN with 0 (can happen if all logits are -inf)
                log_probs = torch.nan_to_num(log_probs, nan=0.0)
                policy_loss = -(target_policies * log_probs).sum(dim=1).mean()

                # Entropy regularization: prevent policy collapse
                if self.config.entropy_regularization:
                    probs = F.softmax(masked_logits, dim=1)
                    log_probs_for_entropy = torch.log(probs + 1e-10)
                    entropy = -(probs * log_probs_for_entropy).sum(dim=1).mean()

                    if self.config.entropy_adaptive and self._last_heuristics:
                        current_entropy = self._last_heuristics.get('policy_entropy', 1.0)
                        if current_entropy < self.config.entropy_target:
                            adaptive_weight = self.config.entropy_weight * (self.config.entropy_target / max(current_entropy, 0.1))
                        else:
                            adaptive_weight = self.config.entropy_weight
                    else:
                        adaptive_weight = self.config.entropy_weight

                    loss = policy_loss + value_loss - adaptive_weight * entropy
                else:
                    loss = policy_loss + value_loss

                loss.backward()
                # Gradient clipping to prevent explosion
                torch.nn.utils.clip_grad_norm_(
                    self.network.parameters(), self.config.max_grad_norm
                )
                self.optimizer.step()

            if self.scheduler and step > 0:
                self.scheduler.step()

            p_item = policy_loss.item()
            v_item = value_loss.item()

            # Check for NaN/Inf losses - indicates catastrophic training failure
            if np.isnan(p_item) or np.isnan(v_item) or np.isinf(p_item) or np.isinf(v_item):
                log.warning(f"NaN/Inf loss detected at step {step + 1}: "
                           f"policy={p_item}, value={v_item}")
                # Don't accumulate bad losses
            else:
                total_policy_loss += p_item
                total_value_loss += v_item

            self.total_steps += 1

            # Track step timing
            now = time.time()
            step_time_ms = (now - last_step_time) * 1000
            last_step_time = now

            # Collect metric for this step
            lr = self.optimizer.param_groups[0]['lr']
            metric = {
                'generation': self.current_generation,
                'step': step + 1,
                'policy_loss': p_item if not np.isnan(p_item) else None,
                'value_loss': v_item if not np.isnan(v_item) else None,
                'total_loss': (p_item + v_item) if not (np.isnan(p_item) or np.isnan(v_item)) else None,
                'learning_rate': lr,
                'step_time_ms': step_time_ms,
                'replay_buffer_size': len(self.replay_buffer),
            }
            # Include heuristics from self-play on first step of each generation
            if step == 0 and self._last_heuristics:
                metric['avg_corners'] = self._last_heuristics['avg_corners']
                metric['avg_mobility'] = self._last_heuristics['avg_mobility']
                metric['center_coverage'] = self._last_heuristics['center_coverage']
                # Policy quality metrics
                if 'policy_entropy' in self._last_heuristics:
                    metric['policy_entropy'] = self._last_heuristics['policy_entropy']
                    metric['policy_top1'] = self._last_heuristics['policy_top1']
                    metric['policy_top5'] = self._last_heuristics['policy_top5']
            metrics_batch.append(metric)

            # Report progress and send metrics every 50 steps
            if (step + 1) % 50 == 0:
                avg_p = total_policy_loss / (step + 1) if step > 0 else 0.0
                avg_v = total_value_loss / (step + 1) if step > 0 else 0.0
                elapsed = time.time() - self.phase_start_time
                steps_per_sec = (step + 1) / elapsed if elapsed > 0 else 0

                # Sanitize for JSON
                avg_p = 0.0 if np.isnan(avg_p) or np.isinf(avg_p) else avg_p
                avg_v = 0.0 if np.isnan(avg_v) or np.isinf(avg_v) else avg_v

                message = f"Step {step + 1}/{training_steps} - loss: {avg_p + avg_v:.4f}"

                self._report_progress(
                    phase='training',
                    phase_progress=(step + 1) / training_steps,
                    phase_current=step + 1,
                    phase_total=training_steps,
                    policy_loss=avg_p,
                    value_loss=avg_v,
                    message=message,
                    extra={
                        'stepsPerSec': round(steps_per_sec, 2),
                        'learningRate': lr,
                        'totalLoss': round(avg_p + avg_v, 4),
                        'adaptiveSteps': training_steps,
                    }
                )
                log.info(f"Step {step + 1}/{training_steps} - "
                      f"policy_loss: {avg_p:.4f}, value_loss: {avg_v:.4f}, {steps_per_sec:.1f} steps/s")

                # Send metrics batch to API
                self._send_metrics_batch(metrics_batch)
                metrics_batch = []

        # Send any remaining metrics
        if metrics_batch:
            self._send_metrics_batch(metrics_batch)

        avg_policy_loss = total_policy_loss / training_steps if training_steps > 0 else 0.0
        avg_value_loss = total_value_loss / training_steps if training_steps > 0 else 0.0

        return avg_policy_loss, avg_value_loss

    def _run_evaluation(self) -> tuple[float, bool]:
        """Evaluate new network against best network using batched MCTS."""
        self.phase_start_time = time.time()
        self.network.eval()
        self.best_network.eval()

        new_wins = 0
        games_played = 0

        for game_idx in range(self.config.eval_games):
            if self._stop_requested:
                break

            # Alternate who plays first
            new_player = game_idx % 2

            # Play a game
            game = BlokusGame()

            new_mcts = HybridMCTS(self.network, self.mcts_config, self.device)
            best_mcts = HybridMCTS(self.best_network, self.mcts_config, self.device)

            while not game.is_game_over:
                legal_moves = game.get_legal_moves()
                if not legal_moves:
                    game.pass_turn()
                    continue

                # Select which network to use
                current = game.current_player
                if current == new_player:
                    mcts = new_mcts
                else:
                    mcts = best_mcts

                # Select move (no noise during evaluation)
                move, _, _ = mcts.search(game, add_noise=False, batch_size=self.config.mcts_batch_size)
                game.make_move(move)

            # Check winner
            winner = game.get_winner()
            if winner == new_player:
                new_wins += 1

            games_played += 1
            current_win_rate = new_wins / games_played

            # Report progress every game
            message = f"Eval game {games_played}/{self.config.eval_games} - win rate: {current_win_rate:.1%}"

            self._report_progress(
                phase='evaluation',
                phase_progress=games_played / self.config.eval_games,
                phase_current=games_played,
                phase_total=self.config.eval_games,
                message=message,
                extra={
                    'evalWins': new_wins,
                    'evalGames': games_played,
                    'currentWinRate': round(current_win_rate, 3),
                    'promotionThreshold': self.current_promotion_threshold,
                    'rustMcts': True,
                }
            )

            if games_played % 5 == 0:
                log.info(f"Eval {games_played}/{self.config.eval_games}: {new_wins} wins ({current_win_rate:.1%})")

        win_rate = new_wins / games_played if games_played > 0 else 0.0
        # Use dynamic promotion threshold instead of fixed config value
        is_best = win_rate >= self.current_promotion_threshold

        # Update promotion threshold based on result
        self._update_promotion_threshold(is_best)

        threshold_info = ""
        if self.config.dynamic_promotion:
            threshold_info = f" (threshold: {self.current_promotion_threshold:.1%})"
        log.info(f"Evaluation complete: {new_wins}/{games_played} wins ({win_rate:.2%}) - "
              f"{'PROMOTED' if is_best else 'not promoted'}{threshold_info}")

        return win_rate, is_best

    def _save_checkpoint(
        self,
        policy_loss: float,
        value_loss: float,
        eval_win_rate: float,
        is_best: bool,
    ) -> str:
        """Save model checkpoint."""
        checkpoint_name = f"gen{self.current_generation:04d}_step{self.total_steps}"
        checkpoint_path = self.run_dir / f"{checkpoint_name}.pt"

        torch.save({
            'generation': self.current_generation,
            'step': self.total_steps,
            'model_state_dict': self.network.state_dict(),
            'optimizer_state_dict': self.optimizer.state_dict(),
            'policy_loss': policy_loss,
            'value_loss': value_loss,
            'eval_win_rate': eval_win_rate,
            'config': self.config.__dict__,
        }, checkpoint_path)

        # Also save as best if it is
        if is_best:
            best_path = self.run_dir / "best.pt"
            torch.save({
                'generation': self.current_generation,
                'step': self.total_steps,
                'model_state_dict': self.network.state_dict(),
                'policy_loss': policy_loss,
                'value_loss': value_loss,
                'eval_win_rate': eval_win_rate,
            }, best_path)

        # Report to API
        self._report_checkpoint(
            generation=self.current_generation,
            step=self.total_steps,
            policy_loss=policy_loss,
            value_loss=value_loss,
            eval_win_rate=eval_win_rate,
            eval_games=self.config.eval_games,
            is_best=is_best,
            model_path=str(checkpoint_path),
        )

def start_training(run_id: str, config_dict: Dict[str, Any], models_dir: Path, api_base_url: str):
    """Start a new training run in the background."""
    config = TrainingConfig.from_dict(config_dict)

    device = 'cuda' if torch.cuda.is_available() else 'cpu'
    log.info(f"Using device: {device}")

    trainer = AlphaZeroTrainer(
        run_id=run_id,
        config=config,
        models_dir=models_dir,
        api_base_url=api_base_url,
        device=device,
    )

    with _trainer_lock:
        _active_trainers[run_id] = trainer

    try:
        trainer.train()
    finally:
        with _trainer_lock:
            _active_trainers.pop(run_id, None)


def stop_training(run_id: str):
    """Stop a running training."""
    with _trainer_lock:
        trainer = _active_trainers.get(run_id)
        if trainer:
            trainer.stop()

class SelfPlayWorker:
    """Worker that runs self-play games."""

    def __init__(self, network: BlokusNet, mcts_config: MCTSConfig, device: str = 'cuda'):
        self.network = network
        self.mcts_config = mcts_config
        self.device = device

    def play_game(self, collect_metrics: bool = True) -> GameResult:
        """
        Play a single self-play game and collect training examples.

        Args:
            collect_metrics: Whether to collect heuristic metrics for diagnostics

        Returns:
            GameResult with examples and game outcome
        """
        from .mcts import MCTS
        import time
        import os

        import sys
        game = BlokusGame()
        mcts = MCTS(self.network, self.mcts_config, self.device)

        # Collect examples: (state, policy, player)
        examples_raw: List[Tuple[np.ndarray, np.ndarray, List[Move], int]] = []

        # Metrics tracking
        corner_counts: List[int] = []  # Track corner counts per turn
        mobility_counts: List[int] = []  # Track legal move counts per turn

        move_count = 0
        game_start = time.time()
        worker_id = os.getpid() % 1000  # Last 3 digits of PID for identification

        log.info(f"[worker-{worker_id}] Game loop starting, {len(game.get_legal_moves())} initial legal moves")

        while not game.is_game_over:
            # Get legal moves
            legal_moves = game.get_legal_moves()

            if not legal_moves:
                game.pass_turn()
                continue

            # Collect metrics
            if collect_metrics:
                # Track valid corners for current player
                corner_counts.append(len(game._valid_corners[game.current_player]))
                # Track mobility (legal move count)
                mobility_counts.append(len(legal_moves))

            # Get state encoding
            board_tensor, pieces_vector = game.get_state_for_neural_net()
            current_player = game.current_player

            # Use MCTS to select move
            add_noise = game.turn_number <= 30  # Add noise early in game
            log.info(f"[worker-{worker_id}] Move {move_count+1}: calling MCTS with {len(legal_moves)} legal moves...")
            move, policy, _ = mcts.select_move(game, add_noise=add_noise)
            log.info(f"[worker-{worker_id}] Move {move_count+1}: MCTS returned")

            # Store example
            examples_raw.append((
                board_tensor.copy(),
                pieces_vector.copy(),
                policy.copy(),
                legal_moves.copy(),
                current_player,
            ))

            # Make move
            game.make_move(move)
            move_count += 1

            # Log every 10 moves
            if move_count % 10 == 0:
                elapsed = time.time() - game_start
                moves_per_sec = move_count / elapsed if elapsed > 0 else 0
                import sys
                log.info(f"[worker-{worker_id}] Move {move_count}, {len(legal_moves)} legal, {moves_per_sec:.1f} moves/s")

        # Game finished - compute final values
        scores = game.get_scores()
        winner = game.get_winner()

        # Compute metrics
        metrics = None
        if collect_metrics:
            # Calculate center coverage (pieces in inner 10x10 region)
            center_coverage = 0
            for r in range(5, 15):
                for c in range(5, 15):
                    if game.board[r, c] != 0:
                        center_coverage += 1
            # Normalize by max possible (100 cells in inner region)
            center_coverage_ratio = center_coverage / 100.0

            # Final corner counts per player
            final_corners = [len(game._valid_corners[p]) for p in range(NUM_PLAYERS)]

            metrics = GameMetrics(
                avg_corner_count=sum(corner_counts) / len(corner_counts) if corner_counts else 0,
                avg_mobility=sum(mobility_counts) / len(mobility_counts) if mobility_counts else 0,
                avg_center_coverage=center_coverage_ratio,
                final_corner_counts=final_corners,
            )

        # Assign values to each example based on final outcome
        examples: List[TrainingExample] = []
        for board_tensor, pieces_vector, policy, legal_moves, player in examples_raw:
            if winner is not None:
                value = 1.0 if player == winner else -1.0
            else:
                # Tie: use relative score
                my_score = scores[player]
                max_other = max(scores[i] for i in range(NUM_PLAYERS) if i != player)
                if my_score > max_other:
                    value = 0.5
                elif my_score < max_other:
                    value = -0.5
                else:
                    value = 0.0

            examples.append(TrainingExample(
                board_tensor=board_tensor,
                pieces_vector=pieces_vector,
                policy=policy,
                legal_moves=legal_moves,
                value=value,
            ))

        return GameResult(
            examples=examples,
            winner=winner,
            scores=scores,
            num_moves=len(game.move_history),
            metrics=metrics,
        )


class SelfPlayManager:
    """
    Manages parallel self-play game generation.
    """

    def __init__(
        self,
        network: BlokusNet,
        mcts_config: MCTSConfig,
        num_workers: int = 8,
        device: str = 'cuda',
    ):
        self.network = network
        self.mcts_config = mcts_config
        self.num_workers = num_workers
        self.device = device

        # Statistics
        self.games_played = 0
        self.total_examples = 0
        self._lock = threading.Lock()

    def generate_games(
        self,
        num_games: int,
        progress_callback: Optional[callable] = None,
    ) -> List[TrainingExample]:
        """
        Generate training examples from self-play games.

        Note: Games are played sequentially because PyTorch CUDA doesn't
        handle multi-threaded access to the same model well.

        Args:
            num_games: Number of games to play
            progress_callback: Optional callback(games_complete, total_games)

        Returns:
            List of all training examples from all games
        """
        import time
        import sys

        all_examples: List[TrainingExample] = []
        start_time = time.time()

        worker = SelfPlayWorker(self.network, self.mcts_config, self.device)

        for game_idx in range(num_games):
            log.info(f"[alphazero] Starting game {game_idx + 1}/{num_games}...")

            result = worker.play_game()

            all_examples.extend(result.examples)
            self.games_played += 1
            self.total_examples += len(result.examples)
            completed = game_idx + 1

            elapsed = time.time() - start_time
            games_per_sec = completed / elapsed if elapsed > 0 else 0
            eta_sec = (num_games - completed) / games_per_sec if games_per_sec > 0 else 0

            log.info(f"[alphazero] Game {completed}/{num_games} complete "
                  f"({result.num_moves} moves, {len(result.examples)} examples, "
                  f"winner={result.winner}, {games_per_sec:.2f} games/s, "
                  f"ETA: {eta_sec/60:.1f}min)")

            if progress_callback:
                try:
                    progress_callback(completed, num_games, result)
                except TypeError:
                    progress_callback(completed, num_games)

        return all_examples

    def reset_stats(self):
        """Reset statistics."""
        with self._lock:
            self.games_played = 0
            self.total_examples = 0


class ReplayBuffer:
    """
    Replay buffer for storing training examples.
    Maintains a fixed-size buffer with oldest examples removed first.
    """

    def __init__(self, max_size: int = 1_000_000):
        self.max_size = max_size
        self.buffer: List[TrainingExample] = []
        self._lock = threading.Lock()

    def add(self, examples: List[TrainingExample]):
        """Add examples to the buffer."""
        with self._lock:
            self.buffer.extend(examples)
            # Remove oldest examples if over capacity
            if len(self.buffer) > self.max_size:
                self.buffer = self.buffer[-self.max_size:]

    def sample(self, batch_size: int, recent_bias: float = 0.0) -> List[TrainingExample]:
        """
        Sample a batch of examples with optional recent bias.

        Args:
            batch_size: Number of examples to sample
            recent_bias: Weight for recency (0.0 = uniform, 1.0 = strongly favor recent)
                        Default 0.0 means uniform sampling for backwards compatibility.
                        Values around 0.7 mean recent examples are ~10x more likely.

        Returns:
            List of sampled training examples
        """
        with self._lock:
            if len(self.buffer) < batch_size:
                # Return all if not enough
                return self.buffer.copy()

            if recent_bias > 0:
                # Exponential weights favoring recent examples
                # recent_bias controls the steepness
                positions = np.arange(len(self.buffer))
                weights = np.exp(recent_bias * (positions / len(self.buffer) - 1) * 3)
                weights = weights / weights.sum()
                indices = np.random.choice(len(self.buffer), batch_size, p=weights, replace=False)
            else:
                indices = np.random.choice(len(self.buffer), batch_size, replace=False)

            return [self.buffer[i] for i in indices]

    def __len__(self):
        return len(self.buffer)

    def clear(self):
        """Clear the buffer."""
        with self._lock:
            self.buffer = []

class ParallelSelfPlay:
    """
    Parallel self-play game generation with batched neural network inference.

    Runs multiple games simultaneously and batches neural network evaluations
    across all active games to maximize GPU utilization. This provides
    significant speedup over sequential game generation.

    Architecture:
    1. Multiple games run in parallel
    2. Each game's MCTS collects leaf nodes for evaluation
    3. Leaves from all games are batched together for GPU inference
    4. Results are distributed back to respective games
    """

    def __init__(
        self,
        network: BlokusNet,
        mcts_config: MCTSConfig,
        num_parallel_games: int = 8,
        mcts_batch_size: int = 16,
        inference_batch_size: int = 32,
        device: str = 'cuda',
        # Cold start fixes
        heuristic_config: Optional['HeuristicConfig'] = None,
        heuristic_blend_factor: float = 0.0,
        policy_target_temperature: float = 1.0,
        # Adaptive noise
        noise_multiplier: float = 1.0,
    ):
        self.network = network
        self.mcts_config = mcts_config
        self.num_parallel_games = num_parallel_games
        self.mcts_batch_size = mcts_batch_size
        self.inference_batch_size = inference_batch_size
        self.device = device

        # Cold start fixes
        self.heuristic_config = heuristic_config
        self.heuristic_blend_factor = heuristic_blend_factor
        self.policy_target_temperature = policy_target_temperature

        # Adaptive noise: multiply dirichlet_epsilon by this factor
        # when entropy is low to force more exploration
        self.noise_multiplier = noise_multiplier

        # Statistics
        self.games_completed = 0
        self.total_examples = 0
        self._lock = threading.Lock()

    def _play_single_game_batched(self, game_id: int, collect_metrics: bool = True) -> GameResult:
        """
        Play a single game using BatchedMCTS for efficient inference.

        This is used when we want to play games one at a time but still
        benefit from batched leaf evaluation within MCTS.
        """
        game = BlokusGame()
        mcts = BatchedMCTS(self.network, self.mcts_config, self.device)

        examples_raw: List[Tuple[np.ndarray, np.ndarray, np.ndarray, List[Move], int]] = []

        # Metrics tracking
        corner_counts: List[int] = [] if collect_metrics else None
        mobility_counts: List[int] = [] if collect_metrics else None

        move_count = 0

        while not game.is_game_over:
            legal_moves = game.get_legal_moves()

            if not legal_moves:
                game.pass_turn()
                continue

            # Collect metrics
            if collect_metrics:
                corner_counts.append(len(game._valid_corners[game.current_player]))
                mobility_counts.append(len(legal_moves))

            board_tensor, pieces_vector = game.get_state_for_neural_net()
            current_player = game.current_player

            # Use batched MCTS
            add_noise = game.turn_number <= 30
            move, policy, _ = mcts.select_move(
                game,
                add_noise=add_noise,
                batch_size=self.mcts_batch_size,
            )

            examples_raw.append((
                board_tensor.copy(),
                pieces_vector.copy(),
                policy.copy(),
                legal_moves.copy(),
                current_player,
            ))

            game.make_move(move)
            move_count += 1

        # Process game result
        return self._finalize_game(game, examples_raw, corner_counts, mobility_counts)

    def _finalize_game(
        self,
        game: BlokusGame,
        examples_raw: List[Tuple[np.ndarray, np.ndarray, np.ndarray, List[Move], int]],
        corner_counts: Optional[List[int]] = None,
        mobility_counts: Optional[List[int]] = None,
    ) -> GameResult:
        """Convert raw examples to TrainingExamples with final values."""
        scores = game.get_scores()
        winner = game.get_winner()

        # Compute metrics if tracking data provided
        metrics = None
        if corner_counts is not None and mobility_counts is not None:
            # Calculate center coverage (pieces in inner 10x10 region)
            center_coverage = 0
            for r in range(5, 15):
                for c in range(5, 15):
                    if game.board[r, c] != 0:
                        center_coverage += 1
            center_coverage_ratio = center_coverage / 100.0

            # Final corner counts per player
            final_corners = [len(game._valid_corners[p]) for p in range(NUM_PLAYERS)]

            # Calculate policy quality metrics from MCTS policies
            # This helps diagnose if MCTS is generating focused vs uniform targets
            entropies = []
            top1_probs = []
            top5_probs = []
            for _, _, policy, _, _ in examples_raw:
                if len(policy) > 0:
                    # Entropy: -sum(p * log(p)), higher = more uniform
                    # Add small epsilon to avoid log(0)
                    p = policy + 1e-10
                    p = p / p.sum()  # Renormalize
                    entropy = -np.sum(p * np.log(p))
                    entropies.append(entropy)

                    # Top-1 probability (how confident is the policy)
                    top1_probs.append(float(np.max(policy)))

                    # Top-5 probability mass
                    if len(policy) >= 5:
                        top5_probs.append(float(np.sum(np.sort(policy)[-5:])))
                    else:
                        top5_probs.append(float(np.sum(policy)))

            metrics = GameMetrics(
                avg_corner_count=sum(corner_counts) / len(corner_counts) if corner_counts else 0,
                avg_mobility=sum(mobility_counts) / len(mobility_counts) if mobility_counts else 0,
                avg_center_coverage=center_coverage_ratio,
                final_corner_counts=final_corners,
                avg_policy_entropy=sum(entropies) / len(entropies) if entropies else 0.0,
                avg_top1_prob=sum(top1_probs) / len(top1_probs) if top1_probs else 0.0,
                avg_top5_prob=sum(top5_probs) / len(top5_probs) if top5_probs else 0.0,
            )

        examples: List[TrainingExample] = []
        for board_tensor, pieces_vector, policy, legal_moves, player in examples_raw:
            if winner is not None:
                value = 1.0 if player == winner else -1.0
            else:
                my_score = scores[player]
                max_other = max(scores[i] for i in range(NUM_PLAYERS) if i != player)
                if my_score > max_other:
                    value = 0.5
                elif my_score < max_other:
                    value = -0.5
                else:
                    value = 0.0

            examples.append(TrainingExample(
                board_tensor=board_tensor,
                pieces_vector=pieces_vector,
                policy=policy,
                legal_moves=legal_moves,
                value=value,
            ))

        return GameResult(
            examples=examples,
            winner=winner,
            scores=scores,
            num_moves=len(game.move_history),
            metrics=metrics,
        )

    def generate_games(
        self,
        num_games: int,
        progress_callback: Optional[callable] = None,
    ) -> List[TrainingExample]:
        """
        Generate training examples from self-play games using batched MCTS.

        Games are played sequentially but each game's MCTS uses batched
        leaf evaluation for efficient GPU utilization.

        Args:
            num_games: Number of games to play
            progress_callback: Optional callback(games_complete, total_games, result)

        Returns:
            List of all training examples from all games
        """
        import sys

        all_examples: List[TrainingExample] = []
        start_time = time.time()

        for game_idx in range(num_games):
            log.info(f"[parallel-selfplay] Starting game {game_idx + 1}/{num_games}...")

            result = self._play_single_game_batched(game_idx)

            all_examples.extend(result.examples)

            with self._lock:
                self.games_completed += 1
                self.total_examples += len(result.examples)

            elapsed = time.time() - start_time
            completed = game_idx + 1
            games_per_min = (completed / elapsed) * 60 if elapsed > 0 else 0
            eta_sec = ((num_games - completed) / completed) * elapsed if completed > 0 else 0

            log.info(f"[parallel-selfplay] Game {completed}/{num_games} complete "
                  f"({result.num_moves} moves, {len(result.examples)} examples, "
                  f"winner={result.winner}, {games_per_min:.1f} games/min, "
                  f"ETA: {eta_sec/60:.1f}min)")

            if progress_callback:
                try:
                    progress_callback(completed, num_games, result)
                except TypeError:
                    progress_callback(completed, num_games)

        return all_examples

    def generate_games_fully_parallel(
        self,
        num_games: int,
        progress_callback: Optional[callable] = None,
    ) -> List[TrainingExample]:
        """
        Generate games with full parallelization - multiple games running
        simultaneously with shared batch inference across all games.

        This provides maximum GPU utilization by batching neural network
        calls across multiple concurrent games.

        Args:
            num_games: Number of games to play
            progress_callback: Optional callback(games_complete, total_games, result)

        Returns:
            List of all training examples from all games
        """
        import sys

        all_examples: List[TrainingExample] = []
        games_completed = 0
        start_time = time.time()
        last_progress_time = start_time
        total_moves_made = 0

        while games_completed < num_games:
            # Start a batch of parallel games
            batch_size = min(self.num_parallel_games, num_games - games_completed)
            active_games: List[ActiveGame] = [
                ActiveGame(game=BlokusGame()) for _ in range(batch_size)
            ]

            log.info(f"[parallel-selfplay] Starting batch of {batch_size} parallel games "
                  f"(games {games_completed + 1}-{games_completed + batch_size}/{num_games})")

            batch_moves = 0

            # Play all games in this batch to completion
            while any(not ag.game.is_game_over for ag in active_games):
                # Collect all active games that need a move
                games_needing_moves: List[Tuple[int, ActiveGame, List[Move]]] = []
                for i, ag in enumerate(active_games):
                    if ag.game.is_game_over:
                        continue

                    legal_moves = ag.game.get_legal_moves()
                    if not legal_moves:
                        ag.game.pass_turn()
                        continue

                    games_needing_moves.append((i, ag, legal_moves))

                if not games_needing_moves:
                    continue

                # Run batched MCTS for all active games
                self._batched_mcts_step(games_needing_moves)
                batch_moves += len(games_needing_moves)
                total_moves_made += len(games_needing_moves)

                # Report intermediate progress every 10 seconds
                now = time.time()
                if progress_callback and (now - last_progress_time) >= 10.0:
                    last_progress_time = now
                    elapsed = now - start_time
                    moves_per_sec = total_moves_made / elapsed if elapsed > 0 else 0
                    # Estimate progress based on moves (assume ~70 moves per game on average)
                    estimated_total_moves = num_games * 70
                    move_progress = min(total_moves_made / estimated_total_moves, 0.99)

                    # Create a partial result for progress reporting
                    try:
                        progress_callback(
                            games_completed,
                            num_games,
                            None,  # No game result yet
                            intermediate=True,
                            moves_made=total_moves_made,
                            moves_per_sec=moves_per_sec,
                            active_games=len(games_needing_moves),
                        )
                    except TypeError:
                        # Callback doesn't support intermediate updates
                        pass

            # Process completed games
            for ag in active_games:
                result = self._finalize_game(
                    ag.game, ag.examples_raw, ag.corner_counts, ag.mobility_counts
                )
                all_examples.extend(result.examples)
                games_completed += 1

                elapsed = time.time() - start_time
                games_per_min = (games_completed / elapsed) * 60 if elapsed > 0 else 0
                eta_sec = ((num_games - games_completed) / games_completed) * elapsed if games_completed > 0 else 0

                log.info(f"[parallel-selfplay] Game {games_completed}/{num_games} complete "
                      f"({result.num_moves} moves, {len(result.examples)} examples, "
                      f"winner={result.winner}, {games_per_min:.1f} games/min, "
                      f"ETA: {eta_sec/60:.1f}min)")

                if progress_callback:
                    try:
                        progress_callback(games_completed, num_games, result)
                    except TypeError:
                        progress_callback(games_completed, num_games)

        return all_examples

    def _batched_mcts_step(
        self,
        games_needing_moves: List[Tuple[int, ActiveGame, List[Move]]],
    ):
        """
        Run one MCTS decision for multiple games using Rust-accelerated MCTS.

        Uses HybridMCTS which is ~14x faster than Python for tree operations.

        Applies cold start fixes:
        1. Heuristic blending: combines NN policy prior with handcrafted heuristics
        2. Temperature adjustment: applies temperature to MCTS visit counts for sharper targets
        """
        mcts_start = time.time()

        for game_idx, ag, legal_moves in games_needing_moves:
            # Collect metrics before move
            ag.corner_counts.append(len(ag.game._valid_corners[ag.game.current_player]))
            ag.mobility_counts.append(len(legal_moves))

            # Create Rust MCTS and run search
            # Note: heuristic blending happens inside HybridMCTS when we pass it
            mcts = HybridMCTS(
                self.network,
                self.mcts_config,
                self.device,
                # Pass heuristic config for policy prior blending
                heuristic_config=self.heuristic_config,
                heuristic_blend_factor=self.heuristic_blend_factor,
                # Pass adaptive noise multiplier
                noise_multiplier=self.noise_multiplier,
            )
            add_noise = ag.game.turn_number <= 30

            move, policy, visits = mcts.search(
                ag.game,
                add_noise=add_noise,
                batch_size=self.mcts_batch_size,
            )

            # Apply temperature to policy targets (sharper policies for training)
            # Temperature < 1.0 makes the distribution more peaked
            if self.policy_target_temperature != 1.0 and len(policy) > 0:
                # Apply temperature: p_i^(1/T) / sum(p_j^(1/T))
                temp_policy = policy ** (1.0 / self.policy_target_temperature)
                temp_policy = temp_policy / temp_policy.sum()
                policy = temp_policy

            # Store example
            board_tensor, pieces_vector = ag.game.get_state_for_neural_net()
            ag.examples_raw.append((
                board_tensor.copy(),
                pieces_vector.copy(),
                policy.copy(),
                legal_moves.copy(),
                ag.game.current_player,
            ))

            # Make move
            ag.game.make_move(move)
            ag.move_count += 1

        log.info(f"Rust MCTS complete: {len(games_needing_moves)} games, "
              f"{(time.time()-mcts_start)*1000:.1f}ms total")

    def reset_stats(self):
        """Reset statistics."""
        with self._lock:
            self.games_completed = 0
            self.total_examples = 0

"""
Neural network for AlphaZero-style Blokus AI.

Architecture:
- Residual tower for board feature extraction
- Policy head: scores legal moves via attention mechanism
- Value head: predicts game outcome [-1, 1]

Optimized for RTX 5090 (32GB VRAM) with large batch sizes.
Uses fully vectorized move encoding for maximum GPU utilization.
"""

import torch
import torch.nn as nn
import torch.nn.functional as F
from typing import Tuple, List, Optional
import numpy as np

from .pieces import NUM_PIECES, PIECE_CATALOG


class ResBlock(nn.Module):
    """Residual block with two conv layers and skip connection."""

    def __init__(self, channels: int):
        super().__init__()
        self.conv1 = nn.Conv2d(channels, channels, 3, padding=1, bias=False)
        self.bn1 = nn.BatchNorm2d(channels)
        self.conv2 = nn.Conv2d(channels, channels, 3, padding=1, bias=False)
        self.bn2 = nn.BatchNorm2d(channels)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        residual = x
        out = F.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out += residual
        return F.relu(out)


class MoveEncoderVectorized(nn.Module):
    """
    Fully vectorized move encoder for GPU-efficient batched inference.

    Pre-computes piece/orientation encodings at init time, then uses
    tensor indexing for O(1) encoding lookup instead of Python loops.

    Move encoding (41-dim):
    - Piece ID: 21-dim one-hot
    - Orientation: 8-dim one-hot
    - Position: 2-dim normalized (row/20, col/20)
    - Piece shape: 10-dim (5 cells × 2 coords, normalized)
    """

    def __init__(self, embed_dim: int = 64):
        super().__init__()
        self.input_dim = 41
        self.embed_dim = embed_dim

        self.embed = nn.Sequential(
            nn.Linear(self.input_dim, embed_dim),
            nn.ReLU(),
            nn.Linear(embed_dim, embed_dim),
        )

        # Pre-compute static encodings for piece_id and orientation one-hots
        # and shape encodings for all (piece, orientation) combinations
        self._precompute_static_encodings()

    def _precompute_static_encodings(self):
        """Pre-compute all static parts of move encodings."""
        # Piece one-hot lookup: (21, 21)
        self.register_buffer('piece_onehot', torch.eye(NUM_PIECES, dtype=torch.float32))

        # Orientation one-hot lookup: (8, 8)
        self.register_buffer('orient_onehot', torch.eye(8, dtype=torch.float32))

        # Shape encodings for all (piece_idx, orient_idx) combinations: (21, 8, 10)
        shape_encodings = torch.zeros(NUM_PIECES, 8, 10, dtype=torch.float32)
        for piece_idx, piece in enumerate(PIECE_CATALOG):
            for orient_idx, shape in enumerate(piece.orientations):
                for i, (r, c) in enumerate(shape[:5]):
                    shape_encodings[piece_idx, orient_idx, i * 2] = r / 5.0
                    shape_encodings[piece_idx, orient_idx, i * 2 + 1] = c / 5.0
        self.register_buffer('shape_encodings', shape_encodings)

    def encode_moves_vectorized(
        self,
        piece_indices: torch.Tensor,    # (N,) int64
        orient_indices: torch.Tensor,   # (N,) int64
        rows: torch.Tensor,             # (N,) float32
        cols: torch.Tensor,             # (N,) float32
    ) -> torch.Tensor:
        """
        Encode moves using pure tensor operations - no Python loops.

        Args:
            piece_indices: Piece IDs for each move
            orient_indices: Orientation IDs for each move
            rows: Row positions (0-19) for each move
            cols: Column positions (0-19) for each move

        Returns:
            Tensor of shape (N, 41) - encoded moves
        """
        N = piece_indices.shape[0]
        if N == 0:
            return torch.zeros((0, self.input_dim), dtype=torch.float32, device=piece_indices.device)

        # Gather one-hot encodings using advanced indexing
        piece_enc = self.piece_onehot[piece_indices]  # (N, 21)
        orient_enc = self.orient_onehot[orient_indices]  # (N, 8)

        # Normalized positions
        pos_enc = torch.stack([rows / 20.0, cols / 20.0], dim=1)  # (N, 2)

        # Gather shape encodings
        shape_enc = self.shape_encodings[piece_indices, orient_indices]  # (N, 10)

        # Concatenate all parts
        return torch.cat([piece_enc, orient_enc, pos_enc, shape_enc], dim=1)  # (N, 41)

    def forward(self, move_encodings: torch.Tensor) -> torch.Tensor:
        """
        Args:
            move_encodings: Tensor of shape (N, 41)

        Returns:
            Tensor of shape (N, embed_dim)
        """
        return self.embed(move_encodings)


class MoveEncoder(nn.Module):
    """
    Legacy move encoder (for backward compatibility).
    Use MoveEncoderVectorized for batched inference.
    """

    def __init__(self, embed_dim: int = 64):
        super().__init__()
        self.input_dim = 41
        self.embed = nn.Sequential(
            nn.Linear(self.input_dim, embed_dim),
            nn.ReLU(),
            nn.Linear(embed_dim, embed_dim),
        )

    def encode_moves(self, moves: List[Tuple[int, int, int, int]]) -> torch.Tensor:
        """Encode moves (legacy Python loop version)."""
        if not moves:
            return torch.zeros((0, self.input_dim), dtype=torch.float32)

        encodings = []
        for piece_idx, orient_idx, row, col in moves:
            piece = PIECE_CATALOG[piece_idx]

            piece_onehot = torch.zeros(NUM_PIECES, dtype=torch.float32)
            piece_onehot[piece_idx] = 1.0

            orient_onehot = torch.zeros(8, dtype=torch.float32)
            orient_onehot[orient_idx] = 1.0

            pos = torch.tensor([row / 20.0, col / 20.0], dtype=torch.float32)

            shape = piece.orientations[orient_idx]
            shape_encoding = torch.zeros(10, dtype=torch.float32)
            for i, (r, c) in enumerate(shape[:5]):
                shape_encoding[i * 2] = r / 5.0
                shape_encoding[i * 2 + 1] = c / 5.0

            encoding = torch.cat([piece_onehot, orient_onehot, pos, shape_encoding])
            encodings.append(encoding)

        return torch.stack(encodings)

    def forward(self, move_encodings: torch.Tensor) -> torch.Tensor:
        return self.embed(move_encodings)


class BlokusNet(nn.Module):
    """
    AlphaZero-style neural network for Blokus.

    Input:
    - board_tensor: (batch, 10, 20, 20) - board state encoding
        Channel 0: Current player's pieces
        Channel 1-3: Other players' pieces (in turn order)
        Channel 4: Valid corner attachment points
        Channel 5: Empty cells
        Channel 6: Turn number (normalized)
        Channel 7: Current player indicator
        Channel 8: Center distance heatmap (closer to center = higher value)
        Channel 9: Game phase (0=early, 0.5=mid, 1=late)
    - pieces_vector: (batch, 84) - remaining pieces for all players

    Output:
    - policy_logits: (batch, num_legal_moves) - scores for each legal move
    - value: (batch, 1) - predicted game outcome [-1, 1]

    Uses fully vectorized move encoding and batched attention for
    maximum GPU utilization during batched inference.
    """

    def __init__(
        self,
        num_res_blocks: int = 15,
        channels: int = 256,
        policy_embed_dim: int = 128,
        value_hidden: int = 256,
        input_channels: int = 10,
    ):
        super().__init__()

        self.channels = channels
        self.policy_embed_dim = policy_embed_dim
        self.input_channels = input_channels

        # Stem: initial convolution
        self.stem = nn.Sequential(
            nn.Conv2d(input_channels, channels, 3, padding=1, bias=False),
            nn.BatchNorm2d(channels),
            nn.ReLU(),
        )

        # Residual tower
        self.res_tower = nn.Sequential(*[
            ResBlock(channels) for _ in range(num_res_blocks)
        ])

        # Global pooling
        self.global_pool = nn.AdaptiveAvgPool2d(1)

        # Combine board features with pieces vector
        self.fc_combine = nn.Sequential(
            nn.Linear(channels + 84, value_hidden * 2),
            nn.ReLU(),
        )

        # Policy head - vectorized encoder for GPU efficiency
        self.move_encoder = MoveEncoderVectorized(embed_dim=policy_embed_dim)

        # Legacy encoder for backward compatibility
        self.move_encoder_legacy = MoveEncoder(embed_dim=policy_embed_dim)

        # Project combined features to policy embedding space for attention
        self.policy_query = nn.Linear(value_hidden * 2, policy_embed_dim)

        # Value head
        self.value_head = nn.Sequential(
            nn.Linear(value_hidden * 2, value_hidden),
            nn.ReLU(),
            nn.Linear(value_hidden, 1),
            nn.Tanh(),
        )

    def _extract_move_tensors(
        self,
        legal_moves_list: List[List],
        device: torch.device,
    ) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor, List[int]]:
        """
        Extract move components into flat tensors for vectorized encoding.

        Returns:
            piece_indices: (total_moves,) int64
            orient_indices: (total_moves,) int64
            rows: (total_moves,) float32
            cols: (total_moves,) float32
            move_counts: List of move counts per batch item
        """
        move_counts = [len(moves) for moves in legal_moves_list]
        total_moves = sum(move_counts)

        if total_moves == 0:
            return (
                torch.tensor([], dtype=torch.int64, device=device),
                torch.tensor([], dtype=torch.int64, device=device),
                torch.tensor([], dtype=torch.float32, device=device),
                torch.tensor([], dtype=torch.float32, device=device),
                move_counts,
            )

        # Pre-allocate numpy arrays
        all_pieces = np.empty(total_moves, dtype=np.int64)
        all_orients = np.empty(total_moves, dtype=np.int64)
        all_rows = np.empty(total_moves, dtype=np.float32)
        all_cols = np.empty(total_moves, dtype=np.float32)

        idx = 0
        for moves in legal_moves_list:
            for m in moves:
                if hasattr(m, 'piece_idx'):
                    all_pieces[idx] = m.piece_idx
                    all_orients[idx] = m.orientation_idx
                    all_rows[idx] = m.row
                    all_cols[idx] = m.col
                else:
                    # Tuple format (piece_idx, orient_idx, row, col)
                    all_pieces[idx] = m[0]
                    all_orients[idx] = m[1]
                    all_rows[idx] = m[2]
                    all_cols[idx] = m[3]
                idx += 1

        return (
            torch.from_numpy(all_pieces).to(device),
            torch.from_numpy(all_orients).to(device),
            torch.from_numpy(all_rows).to(device),
            torch.from_numpy(all_cols).to(device),
            move_counts,
        )

    def forward(
        self,
        board_tensor: torch.Tensor,
        pieces_vector: torch.Tensor,
        legal_moves_list: Optional[List[List[Tuple[int, int, int, int]]]] = None,
    ) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        Forward pass with fully vectorized policy computation.

        Args:
            board_tensor: (batch, 8, 20, 20)
            pieces_vector: (batch, 84)
            legal_moves_list: List of legal moves per batch item

        Returns:
            policy_logits: (batch, max_num_moves) - padded with -inf
            value: (batch, 1)
        """
        batch_size = board_tensor.size(0)
        device = board_tensor.device

        # Board feature extraction
        x = self.stem(board_tensor)
        x = self.res_tower(x)
        x = self.global_pool(x).view(batch_size, -1)

        # Combine with pieces vector
        combined = torch.cat([x, pieces_vector], dim=1)
        features = self.fc_combine(combined)

        # Value prediction
        value = self.value_head(features)

        # Policy prediction
        if legal_moves_list is None:
            policy_logits = torch.zeros(batch_size, 1, device=device)
        else:
            policy_logits = self._compute_policy_vectorized(features, legal_moves_list, device)

        return policy_logits, value

    def _compute_policy_vectorized(
        self,
        features: torch.Tensor,
        legal_moves_list: List[List],
        device: torch.device,
    ) -> torch.Tensor:
        """
        Compute policy logits using fully vectorized operations.

        Instead of looping over batch items, this:
        1. Flattens all moves into single tensors
        2. Encodes all moves in one GPU operation
        3. Embeds all moves in one GPU operation
        4. Computes attention scores using batched operations
        """
        batch_size = features.size(0)

        # Get query vectors for all positions
        query = self.policy_query(features)  # (batch, embed_dim)

        # Extract all moves into flat tensors
        piece_idx, orient_idx, rows, cols, move_counts = self._extract_move_tensors(
            legal_moves_list, device
        )

        total_moves = piece_idx.shape[0]
        max_moves = max(move_counts) if move_counts else 1
        max_moves = max(max_moves, 1)

        if total_moves == 0:
            # Initialize output with -inf (illegal moves)
            return torch.full((batch_size, max_moves), float('-inf'), device=device)

        # Encode ALL moves in one vectorized operation (no Python loops!)
        move_encodings = self.move_encoder.encode_moves_vectorized(
            piece_idx, orient_idx, rows, cols
        )  # (total_moves, 41)

        # Embed ALL moves in one GPU operation
        move_embeddings = self.move_encoder(move_encodings)  # (total_moves, embed_dim)

        # Initialize output with -inf (illegal moves) - use same dtype as embeddings for mixed precision
        policy_logits = torch.full(
            (batch_size, max_moves), float('-inf'),
            device=device, dtype=move_embeddings.dtype
        )

        # Compute attention scores using batched matrix multiplication
        # Pad move embeddings into (batch_size, max_moves, embed_dim) tensor
        # Use same dtype as move_embeddings to handle mixed precision (float16/float32)
        embed_dim = move_embeddings.size(1)
        padded_embeddings = torch.zeros(
            batch_size, max_moves, embed_dim,
            device=device, dtype=move_embeddings.dtype
        )

        # Create indices for scatter - this avoids the Python loop
        # Build batch indices and position indices for all moves
        batch_indices = []
        pos_indices = []
        offset = 0
        for i, count in enumerate(move_counts):
            if count > 0:
                batch_indices.extend([i] * count)
                pos_indices.extend(range(count))
            offset += count

        if batch_indices:
            batch_idx = torch.tensor(batch_indices, dtype=torch.long, device=device)
            pos_idx = torch.tensor(pos_indices, dtype=torch.long, device=device)
            padded_embeddings[batch_idx, pos_idx] = move_embeddings

        # Batched attention: (batch, 1, embed) @ (batch, embed, max_moves) -> (batch, 1, max_moves)
        scores = torch.bmm(query.unsqueeze(1), padded_embeddings.transpose(1, 2)).squeeze(1)

        # Create validity mask and apply scores
        move_counts_t = torch.tensor(move_counts, device=device)
        position_indices = torch.arange(max_moves, device=device).unsqueeze(0)  # (1, max_moves)
        valid_mask = position_indices < move_counts_t.unsqueeze(1)  # (batch, max_moves)

        # Copy scores where valid, keep -inf elsewhere
        policy_logits[valid_mask] = scores[valid_mask]

        return policy_logits

    def forward_fully_batched(
        self,
        board_tensor: torch.Tensor,
        pieces_vector: torch.Tensor,
        legal_moves_list: List[List],
    ) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        Fully batched forward pass using padded move tensors and bmm.

        Uses numpy for fast move tensor creation, then GPU for all neural ops.
        """
        batch_size = board_tensor.size(0)
        device = board_tensor.device

        # Board feature extraction (GPU)
        x = self.stem(board_tensor)
        x = self.res_tower(x)
        x = self.global_pool(x).view(batch_size, -1)

        combined = torch.cat([x, pieces_vector], dim=1)
        features = self.fc_combine(combined)
        value = self.value_head(features)

        # Query vectors
        query = self.policy_query(features)  # (batch, embed_dim)

        # Fast move tensor creation using numpy
        move_counts = [len(moves) for moves in legal_moves_list]
        max_moves = max(move_counts) if move_counts else 1
        max_moves = max(max_moves, 1)

        # Pre-allocate numpy arrays (much faster than torch tensor item assignment)
        all_piece_idx = np.zeros((batch_size, max_moves), dtype=np.int64)
        all_orient_idx = np.zeros((batch_size, max_moves), dtype=np.int64)
        all_rows = np.zeros((batch_size, max_moves), dtype=np.float32)
        all_cols = np.zeros((batch_size, max_moves), dtype=np.float32)
        mask = np.zeros((batch_size, max_moves), dtype=bool)

        # Fill using numpy (faster than torch tensor assignment)
        for i, moves in enumerate(legal_moves_list):
            n = len(moves)
            if n == 0:
                continue
            mask[i, :n] = True

            # Check if moves are objects or tuples
            if moves and hasattr(moves[0], 'piece_idx'):
                # Object format - extract attributes
                for j, m in enumerate(moves):
                    all_piece_idx[i, j] = m.piece_idx
                    all_orient_idx[i, j] = m.orientation_idx
                    all_rows[i, j] = m.row
                    all_cols[i, j] = m.col
            else:
                # Tuple format - use numpy array conversion for speed
                move_arr = np.array(moves, dtype=np.float32)
                all_piece_idx[i, :n] = move_arr[:, 0].astype(np.int64)
                all_orient_idx[i, :n] = move_arr[:, 1].astype(np.int64)
                all_rows[i, :n] = move_arr[:, 2]
                all_cols[i, :n] = move_arr[:, 3]

        # Convert to GPU tensors in one shot
        all_piece_idx = torch.from_numpy(all_piece_idx).to(device)
        all_orient_idx = torch.from_numpy(all_orient_idx).to(device)
        all_rows = torch.from_numpy(all_rows).to(device)
        all_cols = torch.from_numpy(all_cols).to(device)
        mask = torch.from_numpy(mask).to(device)

        # Flatten for vectorized encoding
        flat_piece = all_piece_idx.view(-1)
        flat_orient = all_orient_idx.view(-1)
        flat_rows = all_rows.view(-1)
        flat_cols = all_cols.view(-1)

        # Encode all moves at once (GPU)
        move_encodings = self.move_encoder.encode_moves_vectorized(
            flat_piece, flat_orient, flat_rows, flat_cols
        )

        # Embed all moves at once (GPU)
        move_embeddings = self.move_encoder(move_encodings)

        # Reshape for batched attention
        move_embeddings = move_embeddings.view(batch_size, max_moves, self.policy_embed_dim)

        # Batched attention using bmm (GPU)
        query_expanded = query.unsqueeze(1)
        keys_transposed = move_embeddings.transpose(1, 2)
        policy_logits = torch.bmm(query_expanded, keys_transposed).squeeze(1)

        # Mask out invalid moves
        policy_logits = policy_logits.masked_fill(~mask, float('-inf'))

        return policy_logits, value

    def predict(
        self,
        board_tensor: np.ndarray,
        pieces_vector: np.ndarray,
        legal_moves: List,
        device: str = 'cuda',
    ) -> Tuple[np.ndarray, float]:
        """
        Get policy and value predictions for a single position.

        Args:
            board_tensor: (8, 20, 20) numpy array
            pieces_vector: (84,) numpy array
            legal_moves: List of Move objects or tuples
            device: torch device

        Returns:
            policy: (num_moves,) numpy array of probabilities
            value: float in [-1, 1]
        """
        self.eval()
        with torch.no_grad():
            bt = torch.from_numpy(board_tensor).float().unsqueeze(0).to(device)
            pv = torch.from_numpy(pieces_vector).float().unsqueeze(0).to(device)

            policy_logits, value = self.forward(bt, pv, [legal_moves])

            # Convert logits to probabilities
            policy = F.softmax(policy_logits[0, :len(legal_moves)], dim=0)

            return policy.cpu().numpy(), value.item()

    def predict_batch(
        self,
        board_tensors: List[np.ndarray],
        pieces_vectors: List[np.ndarray],
        legal_moves_list: List[List],
        device: str = 'cuda',
    ) -> Tuple[List[np.ndarray], List[float]]:
        """
        Evaluate multiple positions in one GPU batch for efficient inference.

        Uses fully vectorized move encoding and batched attention (torch.bmm)
        to maximize GPU utilization. This is the key method for batched MCTS.

        Args:
            board_tensors: List of (8, 20, 20) numpy arrays
            pieces_vectors: List of (84,) numpy arrays
            legal_moves_list: List of legal moves for each position
            device: torch device

        Returns:
            policies: List of probability arrays, one per position
            values: List of floats in [-1, 1]
        """
        if not board_tensors:
            return [], []

        batch_size = len(board_tensors)
        self.eval()

        with torch.no_grad():
            # Stack into batch tensors on GPU
            bt = torch.stack([
                torch.from_numpy(b).float() for b in board_tensors
            ]).to(device)
            pv = torch.stack([
                torch.from_numpy(p).float() for p in pieces_vectors
            ]).to(device)

            # Use fully batched forward pass with bmm
            policy_logits, values = self.forward_fully_batched(bt, pv, legal_moves_list)

            # Extract individual policies
            policies = []
            for i in range(batch_size):
                num_moves = len(legal_moves_list[i])
                if num_moves > 0:
                    policy = F.softmax(policy_logits[i, :num_moves], dim=0)
                    policies.append(policy.cpu().numpy())
                else:
                    policies.append(np.array([], dtype=np.float32))

            # Extract values as list
            values_list = values.squeeze(-1).cpu().tolist()

        return policies, values_list

    def save(self, path: str):
        """Save model weights."""
        torch.save(self.state_dict(), path)

    def load(self, path: str, device: str = 'cuda'):
        """Load model weights."""
        self.load_state_dict(torch.load(path, map_location=device))


def create_network(
    num_res_blocks: int = 15,
    channels: int = 256,
    device: str = 'cuda',
    input_channels: int = 10,
) -> BlokusNet:
    """Create a new network with specified architecture."""
    net = BlokusNet(
        num_res_blocks=num_res_blocks,
        channels=channels,
        input_channels=input_channels,
    )
    return net.to(device)


def count_parameters(model: nn.Module) -> int:
    """Count trainable parameters."""
    return sum(p.numel() for p in model.parameters() if p.requires_grad)


class BlokusNetONNX(nn.Module):
    """
    ONNX-exportable wrapper for BlokusNet.

    Exports the feature extraction + value head for fast CPU inference.
    The policy head requires dynamic legal move encoding, which is handled
    separately in inference code using the exported features.

    Input:
        board_tensor: (batch, 10, 20, 20) - board state encoding
        pieces_vector: (batch, 84) - remaining pieces for all players

    Output:
        features: (batch, hidden_dim) - combined features for policy computation
        value: (batch, 1) - game outcome prediction [-1, 1]

    For policy inference with ONNX:
        1. Run ONNX model to get features and value
        2. Use the policy_query_weight and policy_query_bias to compute query
        3. Encode legal moves and compute attention in NumPy/ONNX Runtime
    """

    def __init__(self, blokus_net: BlokusNet):
        super().__init__()
        self.stem = blokus_net.stem
        self.res_tower = blokus_net.res_tower
        self.global_pool = blokus_net.global_pool
        self.fc_combine = blokus_net.fc_combine
        self.value_head = blokus_net.value_head
        # Store policy query layer for optional export
        self.policy_query = blokus_net.policy_query

    def forward(
        self,
        board_tensor: torch.Tensor,
        pieces_vector: torch.Tensor,
    ) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
        """
        Forward pass for ONNX export.

        Args:
            board_tensor: (batch, 10, 20, 20)
            pieces_vector: (batch, 84)

        Returns:
            features: (batch, hidden_dim) - combined features
            value: (batch, 1) - game outcome prediction
            policy_query: (batch, embed_dim) - query vector for policy attention
        """
        batch_size = board_tensor.size(0)

        # Board feature extraction
        x = self.stem(board_tensor)
        x = self.res_tower(x)
        x = self.global_pool(x).view(batch_size, -1)

        # Combine with pieces vector
        combined = torch.cat([x, pieces_vector], dim=1)
        features = self.fc_combine(combined)

        # Value prediction
        value = self.value_head(features)

        # Policy query for attention computation
        query = self.policy_query(features)

        return features, value, query


def export_to_onnx(
    model: BlokusNet,
    output_path: str,
    opset_version: int = 17,
    input_channels: int = None,
) -> dict:
    """
    Export BlokusNet to ONNX format.

    Args:
        model: Trained BlokusNet model
        output_path: Path to save the ONNX file
        opset_version: ONNX opset version
        input_channels: Number of input channels (auto-detected if None)

    Returns:
        dict with export metadata (input_shapes, output_names, file_size)
    """
    import os

    model.eval()

    # Create ONNX wrapper
    onnx_model = BlokusNetONNX(model)
    onnx_model.eval()

    # Detect input channels from model if not specified
    if input_channels is None:
        # Get from stem layer
        input_channels = model.stem[0].in_channels

    # Create dummy inputs
    batch_size = 1
    board_tensor = torch.randn(batch_size, input_channels, 20, 20)
    pieces_vector = torch.randn(batch_size, 84)

    # Move to same device as model
    device = next(model.parameters()).device
    board_tensor = board_tensor.to(device)
    pieces_vector = pieces_vector.to(device)
    onnx_model = onnx_model.to(device)

    # Export to ONNX
    torch.onnx.export(
        onnx_model,
        (board_tensor, pieces_vector),
        output_path,
        export_params=True,
        opset_version=opset_version,
        do_constant_folding=True,
        input_names=['board_tensor', 'pieces_vector'],
        output_names=['features', 'value', 'policy_query'],
        dynamic_axes={
            'board_tensor': {0: 'batch_size'},
            'pieces_vector': {0: 'batch_size'},
            'features': {0: 'batch_size'},
            'value': {0: 'batch_size'},
            'policy_query': {0: 'batch_size'},
        },
    )

    # Get file size
    file_size_bytes = os.path.getsize(output_path)
    file_size_mb = file_size_bytes / (1024 * 1024)

    # Get output shapes by running inference
    with torch.no_grad():
        features, value, query = onnx_model(board_tensor, pieces_vector)

    return {
        'input_shapes': {
            'board_tensor': [batch_size, input_channels, 20, 20],
            'pieces_vector': [batch_size, 84],
        },
        'output_shapes': {
            'features': list(features.shape),
            'value': list(value.shape),
            'policy_query': list(query.shape),
        },
        'output_names': ['features', 'value', 'policy_query'],
        'file_size_bytes': file_size_bytes,
        'file_size_mb': round(file_size_mb, 2),
        'opset_version': opset_version,
    }

"""
Monte Carlo Tree Search implementation for AlphaZero-style Blokus AI.

Uses PUCT formula for node selection and neural network for policy/value evaluation.
"""

import math
import numpy as np
from typing import Dict, List, Optional, Tuple
from dataclasses import dataclass, field
from collections import defaultdict

from .game import BlokusGame, Move


@dataclass
class MCTSConfig:
    """Configuration for MCTS."""
    num_simulations: int = 400
    c_puct: float = 1.5
    dirichlet_alpha: float = 0.3
    dirichlet_epsilon: float = 0.25
    temperature: float = 1.0
    temperature_threshold: int = 20  # Use temperature=0.1 after this many moves


class MCTSNode:
    """
    A node in the MCTS tree with lazy child expansion.

    Key optimization: Children are NOT fully created during expand().
    Instead, we store policy priors and only create the actual child
    game state when that child is selected for traversal.

    This avoids cloning game states for moves that are never explored.
    """

    def __init__(self, game: BlokusGame, parent: Optional['MCTSNode'] = None,
                 move: Optional[Move] = None, prior: float = 0.0):
        self.game = game
        self.parent = parent
        self.move = move  # Move that led to this state
        self.prior = prior  # P(s, a) from neural network

        # Statistics
        self.visit_count = 0
        self.value_sum = 0.0
        self.children: Dict[Move, 'MCTSNode'] = {}

        # Legal moves at this state (cached)
        self._legal_moves: Optional[List[Move]] = None
        self._is_expanded = False

        # Lazy expansion: store priors without creating child game states
        self._child_priors: Optional[Dict[Move, float]] = None

    @property
    def legal_moves(self) -> List[Move]:
        """Get legal moves (cached)."""
        if self._legal_moves is None:
            self._legal_moves = self.game.get_legal_moves()
        return self._legal_moves

    @property
    def is_terminal(self) -> bool:
        """Check if this is a terminal state."""
        return self.game.is_game_over

    @property
    def q_value(self) -> float:
        """Get average value (Q) for this node."""
        if self.visit_count == 0:
            return 0.0
        return self.value_sum / self.visit_count

    def ucb_score(self, c_puct: float) -> float:
        """
        Calculate UCB score for node selection.

        UCB(s,a) = Q(s,a) + c_puct * P(s,a) * sqrt(N(parent)) / (1 + N(s,a))
        """
        if self.parent is None:
            return 0.0

        exploration = c_puct * self.prior * math.sqrt(self.parent.visit_count) / (1 + self.visit_count)
        return self.q_value + exploration

    def _get_unexpanded_ucb(self, move: Move, c_puct: float) -> float:
        """Calculate UCB score for a move that hasn't been expanded yet."""
        prior = self._child_priors.get(move, 0.0)
        # Unexpanded nodes have visit_count=0, value_sum=0, so Q=0
        exploration = c_puct * prior * math.sqrt(self.visit_count) / 1
        return exploration

    def select_child(self, c_puct: float) -> 'MCTSNode':
        """
        Select best child according to UCB score.

        With lazy expansion, we consider both:
        1. Already-expanded children (have game states)
        2. Unexpanded moves (only have priors)

        If an unexpanded move has the best UCB, we expand it on-demand.
        """
        best_score = float('-inf')
        best_child = None
        best_unexpanded_move = None

        # Check expanded children
        for child in self.children.values():
            score = child.ucb_score(c_puct)
            if score > best_score:
                best_score = score
                best_child = child
                best_unexpanded_move = None

        # Check unexpanded moves (if any)
        if self._child_priors:
            for move, prior in self._child_priors.items():
                if move not in self.children:
                    score = self._get_unexpanded_ucb(move, c_puct)
                    if score > best_score:
                        best_score = score
                        best_child = None
                        best_unexpanded_move = move

        # If best is unexpanded, expand it now (lazy expansion)
        if best_unexpanded_move is not None:
            best_child = self._expand_child(best_unexpanded_move)

        assert best_child is not None, "No children to select from"
        return best_child

    def _expand_child(self, move: Move) -> 'MCTSNode':
        """Lazily expand a single child when it's actually selected."""
        prior = self._child_priors.get(move, 0.0)

        # Clone game and apply move
        child_game = self.game.clone()
        child_game.make_move(move)

        # Auto-pass if needed
        while not child_game.is_game_over and not child_game.get_legal_moves():
            child_game.pass_turn()

        child = MCTSNode(
            game=child_game,
            parent=self,
            move=move,
            prior=prior,
        )
        self.children[move] = child
        return child

    def expand(self, policy: np.ndarray, legal_moves: List[Move]):
        """
        Expand node by storing policy priors (lazy expansion).

        Unlike eager expansion, this does NOT create child game states.
        Children are created on-demand in select_child() when selected.

        This is a massive speedup: O(1) instead of O(num_legal_moves) game clones.
        """
        assert len(policy) == len(legal_moves)

        # Store priors for lazy expansion
        self._child_priors = {move: policy[i] for i, move in enumerate(legal_moves)}
        self._is_expanded = True

    def expand_eager(self, policy: np.ndarray, legal_moves: List[Move]):
        """
        Eager expansion (legacy) - creates all children immediately.
        Use expand() for lazy expansion which is much faster.
        """
        assert len(policy) == len(legal_moves)

        for i, move in enumerate(legal_moves):
            if move not in self.children:
                child_game = self.game.clone()
                child_game.make_move(move)

                while not child_game.is_game_over and not child_game.get_legal_moves():
                    child_game.pass_turn()

                child = MCTSNode(
                    game=child_game,
                    parent=self,
                    move=move,
                    prior=policy[i],
                )
                self.children[move] = child

        self._child_priors = {move: policy[i] for i, move in enumerate(legal_moves)}
        self._is_expanded = True

    def backpropagate(self, value: float, perspective_player: int):
        """
        Backpropagate value up the tree.

        Value is from the perspective of perspective_player.
        Nodes flip the sign based on whose turn it is.
        """
        node = self
        while node is not None:
            node.visit_count += 1
            # Flip value if it's the opponent's perspective
            if node.parent is not None:
                parent_player = node.parent.game.current_player
                if parent_player == perspective_player:
                    node.value_sum += value
                else:
                    node.value_sum -= value
            node = node.parent


class MCTS:
    """
    Monte Carlo Tree Search with neural network guidance.
    """

    def __init__(self, network, config: MCTSConfig, device: str = 'cuda'):
        self.network = network
        self.config = config
        self.device = device

    def _get_policy_value(self, game: BlokusGame, legal_moves: List[Move]) -> Tuple[np.ndarray, float]:
        """Get policy and value from neural network."""
        board_tensor, pieces_vector = game.get_state_for_neural_net()
        policy, value = self.network.predict(board_tensor, pieces_vector, legal_moves, self.device)
        return policy, value

    def search(self, root_game: BlokusGame, add_noise: bool = True) -> MCTSNode:
        """
        Run MCTS from the given game state.

        Args:
            root_game: Current game state
            add_noise: Whether to add Dirichlet noise at root (for exploration during training)

        Returns:
            Root node after search
        """
        root = MCTSNode(root_game.clone())

        # Check terminal
        if root.is_terminal:
            return root

        # Get initial policy and expand root
        legal_moves = root.legal_moves
        if not legal_moves:
            return root

        policy, value = self._get_policy_value(root.game, legal_moves)

        # Add Dirichlet noise at root for exploration
        if add_noise and len(policy) > 0:
            noise = np.random.dirichlet([self.config.dirichlet_alpha] * len(policy))
            policy = (1 - self.config.dirichlet_epsilon) * policy + self.config.dirichlet_epsilon * noise

        root.expand(policy, legal_moves)

        # Run simulations
        for _ in range(self.config.num_simulations):
            node = root
            perspective_player = root.game.current_player

            # Selection: traverse tree to leaf
            while node._is_expanded and not node.is_terminal:
                node = node.select_child(self.config.c_puct)

            # Evaluation and expansion
            if node.is_terminal:
                # Terminal node: use actual game result
                scores = node.game.get_scores()
                winner = node.game.get_winner()
                if winner is not None:
                    value = 1.0 if winner == perspective_player else -1.0
                else:
                    value = 0.0  # Tie
            else:
                # Leaf node: expand and evaluate
                legal_moves = node.legal_moves
                if legal_moves:
                    policy, value = self._get_policy_value(node.game, legal_moves)
                    node.expand(policy, legal_moves)
                else:
                    # No legal moves - pass turn
                    value = 0.0

            # Backpropagation
            node.backpropagate(value, perspective_player)

        return root

    def get_action_probs(self, root: MCTSNode, temperature: float = 1.0) -> Tuple[List[Move], np.ndarray]:
        """
        Get action probabilities from MCTS visit counts.

        Args:
            root: Root node after search
            temperature: Temperature for action selection
                        1.0 = proportional to visit counts
                        0 (or very small) = argmax

        Returns:
            moves: List of legal moves
            probs: Action probabilities
        """
        moves = []
        visits = []

        for move, child in root.children.items():
            moves.append(move)
            visits.append(child.visit_count)

        if not moves:
            return [], np.array([])

        visits = np.array(visits, dtype=np.float32)

        if temperature < 0.01:
            # Greedy selection
            probs = np.zeros_like(visits)
            probs[np.argmax(visits)] = 1.0
        else:
            # Temperature-weighted
            visits = visits ** (1.0 / temperature)
            probs = visits / visits.sum()

        return moves, probs

    def select_move(self, game: BlokusGame, add_noise: bool = True,
                    temperature: Optional[float] = None) -> Tuple[Move, np.ndarray, MCTSNode]:
        """
        Select a move using MCTS.

        Args:
            game: Current game state
            add_noise: Whether to add exploration noise
            temperature: Override temperature (None = use config based on turn)

        Returns:
            selected_move: The chosen move
            policy: Action probabilities (for training)
            root: Root node (for analysis)
        """
        root = self.search(game, add_noise=add_noise)

        # Determine temperature
        if temperature is None:
            if game.turn_number <= self.config.temperature_threshold:
                temperature = self.config.temperature
            else:
                temperature = 0.1

        moves, probs = self.get_action_probs(root, temperature)

        if not moves:
            raise ValueError("No legal moves available")

        # Sample move according to probabilities
        idx = np.random.choice(len(moves), p=probs)
        selected_move = moves[idx]

        # Create full policy vector for training (over all legal moves, not just children)
        full_policy = np.zeros(len(moves))
        for i, m in enumerate(moves):
            full_policy[i] = probs[i]

        return selected_move, full_policy, root


class BatchedMCTS:
    """
    Monte Carlo Tree Search with batched neural network evaluation.

    Instead of evaluating one leaf at a time, this implementation collects
    multiple leaf nodes during the selection phase and evaluates them in
    a single GPU batch. This dramatically improves GPU utilization.

    Uses "virtual loss" to encourage exploration of different paths when
    selecting multiple leaves in parallel.
    """

    def __init__(self, network, config: MCTSConfig, device: str = 'cuda'):
        self.network = network
        self.config = config
        self.device = device
        self.virtual_loss = 3.0  # Pessimistic value during selection

    def _select_leaf(self, root: MCTSNode, c_puct: float) -> Tuple[MCTSNode, List[MCTSNode]]:
        """
        Select a leaf node by traversing down the tree using UCB scores.

        Returns:
            leaf: The selected leaf node
            path: List of nodes from root to leaf (inclusive)
        """
        node = root
        path = [node]

        while node._is_expanded and not node.is_terminal:
            node = node.select_child(c_puct)
            path.append(node)

        return node, path

    def _apply_virtual_loss(self, path: List[MCTSNode]):
        """
        Apply virtual loss to path to discourage other simulations from
        selecting the same path. This is key for parallel leaf collection.
        """
        for node in path[1:]:  # Skip root
            node.visit_count += 1
            node.value_sum -= self.virtual_loss

    def _remove_virtual_loss(self, path: List[MCTSNode]):
        """Remove virtual loss after real evaluation."""
        for node in path[1:]:  # Skip root
            node.visit_count -= 1
            node.value_sum += self.virtual_loss

    def _get_batch_policy_value(
        self,
        games: List[BlokusGame],
        legal_moves_list: List[List[Move]],
    ) -> Tuple[List[np.ndarray], List[float]]:
        """Get policies and values for a batch of positions."""
        board_tensors = []
        pieces_vectors = []

        for game in games:
            bt, pv = game.get_state_for_neural_net()
            board_tensors.append(bt)
            pieces_vectors.append(pv)

        return self.network.predict_batch(
            board_tensors, pieces_vectors, legal_moves_list, self.device
        )

    def search(self, root_game: BlokusGame, add_noise: bool = True,
               batch_size: int = 16) -> MCTSNode:
        """
        Run batched MCTS from the given game state.

        Args:
            root_game: Current game state
            add_noise: Whether to add Dirichlet noise at root
            batch_size: Number of leaves to evaluate per batch

        Returns:
            Root node after search
        """
        root = MCTSNode(root_game.clone())

        if root.is_terminal:
            return root

        legal_moves = root.legal_moves
        if not legal_moves:
            return root

        # Get initial policy and expand root
        board_tensor, pieces_vector = root.game.get_state_for_neural_net()
        policy, value = self.network.predict(
            board_tensor, pieces_vector, legal_moves, self.device
        )

        # Add Dirichlet noise at root for exploration
        if add_noise and len(policy) > 0:
            noise = np.random.dirichlet([self.config.dirichlet_alpha] * len(policy))
            policy = (1 - self.config.dirichlet_epsilon) * policy + self.config.dirichlet_epsilon * noise

        root.expand(policy, legal_moves)
        perspective_player = root.game.current_player

        # Run simulations in batches
        simulations_done = 0
        while simulations_done < self.config.num_simulations:
            current_batch_size = min(batch_size, self.config.num_simulations - simulations_done)

            # Phase 1: Selection - collect leaves with virtual loss
            leaves = []
            paths = []
            leaf_legal_moves = []

            for _ in range(current_batch_size):
                leaf, path = self._select_leaf(root, self.config.c_puct)
                leaves.append(leaf)
                paths.append(path)

                if not leaf.is_terminal:
                    self._apply_virtual_loss(path)
                    leaf_legal_moves.append(leaf.legal_moves)
                else:
                    leaf_legal_moves.append([])

            # Phase 2: Batch evaluation
            # Separate terminal and non-terminal leaves
            to_evaluate_indices = []
            to_evaluate_games = []
            to_evaluate_moves = []

            for i, (leaf, moves) in enumerate(zip(leaves, leaf_legal_moves)):
                if not leaf.is_terminal and moves:
                    to_evaluate_indices.append(i)
                    to_evaluate_games.append(leaf.game)
                    to_evaluate_moves.append(moves)

            # Batch evaluate non-terminal leaves
            if to_evaluate_games:
                policies, values = self._get_batch_policy_value(
                    to_evaluate_games, to_evaluate_moves
                )
            else:
                policies, values = [], []

            # Phase 3: Expansion and Backpropagation
            eval_idx = 0
            for i, (leaf, path, moves) in enumerate(zip(leaves, paths, leaf_legal_moves)):
                if leaf.is_terminal:
                    # Terminal: use actual game result
                    winner = leaf.game.get_winner()
                    if winner is not None:
                        value = 1.0 if winner == perspective_player else -1.0
                    else:
                        value = 0.0
                elif moves:
                    # Non-terminal with moves: expand with NN policy
                    policy = policies[eval_idx]
                    value = values[eval_idx]
                    leaf.expand(policy, moves)
                    self._remove_virtual_loss(path)
                    eval_idx += 1
                else:
                    # No legal moves
                    value = 0.0
                    self._remove_virtual_loss(path)

                # Backpropagation
                leaf.backpropagate(value, perspective_player)

            simulations_done += current_batch_size

        return root

    def get_action_probs(self, root: MCTSNode, temperature: float = 1.0) -> Tuple[List[Move], np.ndarray]:
        """Get action probabilities from MCTS visit counts."""
        moves = []
        visits = []

        for move, child in root.children.items():
            moves.append(move)
            visits.append(child.visit_count)

        if not moves:
            return [], np.array([])

        visits = np.array(visits, dtype=np.float32)

        if temperature < 0.01:
            probs = np.zeros_like(visits)
            probs[np.argmax(visits)] = 1.0
        else:
            visits = visits ** (1.0 / temperature)
            probs = visits / visits.sum()

        return moves, probs

    def select_move(self, game: BlokusGame, add_noise: bool = True,
                    temperature: Optional[float] = None,
                    batch_size: int = 16) -> Tuple[Move, np.ndarray, MCTSNode]:
        """
        Select a move using batched MCTS.

        Args:
            game: Current game state
            add_noise: Whether to add exploration noise
            temperature: Override temperature (None = use config based on turn)
            batch_size: Number of leaves to evaluate per batch

        Returns:
            selected_move: The chosen move
            policy: Action probabilities (for training)
            root: Root node (for analysis)
        """
        root = self.search(game, add_noise=add_noise, batch_size=batch_size)

        if temperature is None:
            if game.turn_number <= self.config.temperature_threshold:
                temperature = self.config.temperature
            else:
                temperature = 0.1

        moves, probs = self.get_action_probs(root, temperature)

        if not moves:
            raise ValueError("No legal moves available")

        idx = np.random.choice(len(moves), p=probs)
        selected_move = moves[idx]

        full_policy = np.zeros(len(moves))
        for i, m in enumerate(moves):
            full_policy[i] = probs[i]

        return selected_move, full_policy, root


def create_mcts(network, config: Optional[MCTSConfig] = None, device: str = 'cuda') -> MCTS:

"""
Rust-accelerated MCTS for AlphaZero training.

Uses the rust_mcts module for fast tree operations while keeping
neural network inference in Python/PyTorch.

The key optimization is that tree operations (selection, UCB calculation,
backpropagation) are done in Rust, which is ~14x faster than Python.

    RUST_AVAILABLE = False
    log.warning("rust_mcts not available, falling back to Python MCTS")

# Try to import heuristics for cold start bootstrapping
try:
    from .heuristics import compute_heuristic_policy, blend_policies
    HEURISTICS_AVAILABLE = True
except ImportError:
    HEURISTICS_AVAILABLE = False
    log.info("Heuristics module not available, disabling policy bootstrapping")


def encode_move(move: Move) -> int:
    """Encode a Move object as a u32 for Rust."""
    if not RUST_AVAILABLE:
        return (move.piece_idx << 24) | (move.orientation_idx << 16) | (move.row << 8) | move.col
    return rust_mcts.GameState.encode_move(
        move.piece_idx, move.orientation_idx, move.row, move.col
    )


def decode_move(move_id: int, player: int) -> Move:
    """Decode a u32 back to a Move object."""
    if not RUST_AVAILABLE:
        piece_idx = (move_id >> 24) & 0xFF
        orientation = (move_id >> 16) & 0xFF
        row = (move_id >> 8) & 0xFF
        col = move_id & 0xFF
        return Move(player, piece_idx, orientation, row, col)
    piece_idx, orientation, row, col = rust_mcts.GameState.decode_move(move_id)
    return Move(player, piece_idx, orientation, row, col)


@dataclass
class GameTreeNode:
    """
    Wrapper that associates a Rust tree node index with a game state.

    The Rust tree only stores tree structure and statistics.
    Game states are stored here in Python for leaf evaluation.
    """
    node_idx: int  # Index in Rust tree
    game: BlokusGame  # Game state at this node
    move_id_to_move: Dict[int, Move] = field(default_factory=dict)  # Cached move mappings


class HybridMCTS:
    """
    Hybrid MCTS that uses Rust for tree operations and Python for game logic.

    Architecture:
    - Rust: Tree structure, visit counts, UCB scores, backpropagation
    - Python: Game state tracking, legal move generation, neural network inference

    This provides ~14x speedup on tree operations while maintaining
    compatibility with the existing Python game implementation.

    Cold Start Fixes:
    - Supports heuristic blending to bootstrap policy learning
    - Blends handcrafted move priors with neural network output at root
    """

    def __init__(
        self,
        network: BlokusNet,
        config: MCTSConfig,
        device: str = 'cuda',
        # Cold start fix: heuristic policy bootstrapping
        heuristic_config: Optional[Any] = None,
        heuristic_blend_factor: float = 0.0,
        # Adaptive noise: multiplier for dirichlet_epsilon
        noise_multiplier: float = 1.0,
    ):
        if not RUST_AVAILABLE:
            raise RuntimeError(
                "rust_mcts module not available. "
                "Build with: cd ml-service/rust_mcts && maturin develop --release"
            )

        self.network = network
        self.config = config
        self.device = device

        # Cold start fix parameters
        self.heuristic_config = heuristic_config
        self.heuristic_blend_factor = heuristic_blend_factor

        # Adaptive noise: multiply dirichlet_epsilon by this factor
        # when entropy is low to force more exploration
        self.noise_multiplier = noise_multiplier

    def search(
        self,
        game: BlokusGame,
        add_noise: bool = True,
        batch_size: int = 16,
    ) -> Tuple[Move, np.ndarray, int]:
        """
        Run MCTS search from the given game state.

        Args:
            game: Current game state
            add_noise: Whether to add Dirichlet noise at root
            batch_size: Number of leaves to evaluate per iteration

        Returns:
            (selected_move, policy, root_visits)
        """
        legal_moves = game.get_legal_moves()
        if not legal_moves:
            raise ValueError("No legal moves available")

        # Create Rust tree
        tree = rust_mcts.MCTSTree(game.current_player)
        perspective_player = game.current_player

        # Build move mappings
        move_id_to_move = {}
        move_to_id = {}
        for move in legal_moves:
            move_id = encode_move(move)
            move_id_to_move[move_id] = move
            move_to_id[id(move)] = move_id

        # Get initial policy from neural network
        board_tensor, pieces_vector = game.get_state_for_neural_net()
        moves_tuples = [(m.piece_idx, m.orientation_idx, m.row, m.col) for m in legal_moves]

        policy, value = self.network.predict(
            board_tensor, pieces_vector, moves_tuples, self.device
        )

        # COLD START FIX: Blend neural network policy with handcrafted heuristics
        # This helps bootstrap learning when the NN outputs uniform distributions
        if (HEURISTICS_AVAILABLE and
            self.heuristic_config is not None and
            self.heuristic_blend_factor > 0.0 and
            len(policy) > 0):

            heuristic_policy = compute_heuristic_policy(
                game, legal_moves, self.heuristic_config
            )
            policy = blend_policies(policy, heuristic_policy, self.heuristic_blend_factor)

        # Add Dirichlet noise if requested
        if add_noise and len(policy) > 0:
            noise = np.random.dirichlet([self.config.dirichlet_alpha] * len(policy))
            # Apply adaptive noise multiplier (capped at 0.5 to avoid complete randomness)
            adjusted_epsilon = min(0.5, self.config.dirichlet_epsilon * self.noise_multiplier)
            policy = (
                (1 - adjusted_epsilon) * policy +
                adjusted_epsilon * noise
            )

        # Set root policy
        root_priors = [(encode_move(m), float(policy[i])) for i, m in enumerate(legal_moves)]
        tree.set_root_policy(root_priors)

        # Track game states at each node
        # Key: tuple of path indices, Value: (game_state, legal_moves, move_id_to_move)
        node_states: Dict[tuple, Tuple[BlokusGame, List[Move], Dict[int, Move]]] = {
            (0,): (game.clone(), legal_moves, move_id_to_move)
        }

        # Run simulations
        simulations_done = 0
        virtual_loss = 3.0

        while simulations_done < self.config.num_simulations:
            batch_this_iter = min(batch_size, self.config.num_simulations - simulations_done)
            leaves_to_eval = []  # (path, game, legal_moves, move_id_to_move)

            for _ in range(batch_this_iter):
                # Select leaf
                result = tree.select_leaf(self.config.c_puct)
                if result is None:
                    continue

                path, priors_at_leaf = result
                path_tuple = tuple(path)

                # Find the game state at this leaf
                # Walk down from the longest known prefix
                leaf_game = None
                leaf_moves = None
                leaf_move_map = None

                # Check if we have this exact path cached
                if path_tuple in node_states:
                    leaf_game, leaf_moves, leaf_move_map = node_states[path_tuple]
                else:
                    # Find longest known prefix and extend
                    for prefix_len in range(len(path) - 1, 0, -1):
                        prefix = tuple(path[:prefix_len])
                        if prefix in node_states:
                            parent_game, parent_moves, parent_move_map = node_states[prefix]

                            # We need to know which move leads to next node
                            # This requires tracking which move was selected
                            # For now, use simplified approach: re-evaluate from parent
                            leaf_game = parent_game.clone()
                            leaf_moves = leaf_game.get_legal_moves()
                            leaf_move_map = {encode_move(m): m for m in leaf_moves}
                            break

                    if leaf_game is None:
                        # Fallback to root
                        root_game, root_moves, root_move_map = node_states[(0,)]
                        leaf_game = root_game.clone()
                        leaf_moves = root_moves
                        leaf_move_map = root_move_map

                # If leaf needs expansion (has priors but no children)
                if priors_at_leaf and leaf_moves:
                    leaves_to_eval.append((path, leaf_game, leaf_moves, leaf_move_map))
                else:
                    # Terminal or already expanded - backprop with current value
                    if leaf_game.is_game_over:
                        winner = leaf_game.get_winner()
                        if winner is not None:
                            value = 1.0 if winner == perspective_player else -1.0
                        else:
                            value = 0.0
                    else:
                        value = 0.0

                    tree.expand_and_backprop(list(path), [], float(value))
                    simulations_done += 1

            # Batch evaluate leaves
            if leaves_to_eval:
                eval_games = [g for _, g, _, _ in leaves_to_eval]
                eval_moves = [m for _, _, m, _ in leaves_to_eval]

                board_tensors = [g.get_state_for_neural_net()[0] for g in eval_games]
                pieces_vectors = [g.get_state_for_neural_net()[1] for g in eval_games]
                moves_tuples = [
                    [(m.piece_idx, m.orientation_idx, m.row, m.col) for m in moves]
                    for moves in eval_moves
                ]

                eval_policies, eval_values = self.network.predict_batch(
                    board_tensors, pieces_vectors, moves_tuples, self.device
                )

                # Expand and backprop
                for idx, (path, leaf_game, leaf_moves, leaf_move_map) in enumerate(leaves_to_eval):
                    policy = eval_policies[idx]
                    value = eval_values[idx]

                    # Build children priors
                    children_priors = [
                        (encode_move(m), float(policy[j]))
                        for j, m in enumerate(leaf_moves)
                    ]

                    # Cache this node's state
                    path_tuple = tuple(path)
                    if path_tuple not in node_states:
                        node_states[path_tuple] = (leaf_game.clone(), leaf_moves, leaf_move_map)

                    # Expand and backprop in Rust
                    tree.expand_and_backprop(list(path), children_priors, float(value))
                    simulations_done += 1

        # Get final policy
        temperature = self.config.temperature if game.turn_number <= self.config.temperature_threshold else 0.1
        action_probs = tree.get_action_probs(temperature)

        # Build policy array
        policy_array = np.zeros(len(legal_moves))
        move_id_to_prob = dict(action_probs)

        for j, move in enumerate(legal_moves):
            move_id = encode_move(move)
            policy_array[j] = move_id_to_prob.get(move_id, 0.0)

        # Normalize
        if policy_array.sum() > 0:
            policy_array = policy_array / policy_array.sum()
        else:
            policy_array = np.ones(len(legal_moves)) / len(legal_moves)

        # Sample move
        idx = np.random.choice(len(legal_moves), p=policy_array)
        selected_move = legal_moves[idx]

        return selected_move, policy_array, tree.root_visits()

Native Python Blokus game implementation optimized for AlphaZero self-play.
Uses NumPy for fast board operations.

Set BLOKUS_USE_OPTIMIZED=1 to use the numba-optimized implementation (9x faster).

import os

    PIECE_CATALOG,

    PIECE_INDEX_BY_ID,

    PieceDefinition,
)

# Check if we should use the optimized implementation

# Board constants
BOARD_SIZE = 20
NUM_PLAYERS = 4

# Player colors (indices)
BLUE = 0
YELLOW = 1
RED = 2
GREEN = 3

PLAYER_NAMES = ['blue', 'yellow', 'red', 'green']

# Starting corners for each player (row, col)
STARTING_CORNERS = {
    BLUE: (0, 0),         # Top-left
    YELLOW: (0, 19),      # Top-right
    RED: (19, 19),        # Bottom-right
    GREEN: (19, 0),       # Bottom-left
}

# Direction offsets for orthogonal neighbors (edge adjacency)
EDGE_OFFSETS = [(-1, 0), (1, 0), (0, -1), (0, 1)]

# Direction offsets for diagonal neighbors (corner adjacency)
DIAGONAL_OFFSETS = [(-1, -1), (-1, 1), (1, -1), (1, 1)]

# Precomputed center distance heatmap for neural network input
# Cells closer to center (9.5, 9.5) have higher values (0-1 normalized)
# Max distance from center to corner is sqrt(9.5^2 + 9.5^2) ≈ 13.44
def _compute_center_distance_heatmap() -> np.ndarray:
    """Compute center distance heatmap once at module load."""
    heatmap = np.zeros((BOARD_SIZE, BOARD_SIZE), dtype=np.float32)
    center = (BOARD_SIZE - 1) / 2.0  # 9.5 for 20x20 board
    max_dist = np.sqrt(2) * center  # ~13.44
    for r in range(BOARD_SIZE):
        for c in range(BOARD_SIZE):
            dist = np.sqrt((r - center) ** 2 + (c - center) ** 2)
            heatmap[r, c] = 1.0 - (dist / max_dist)  # Closer to center = higher value
    return heatmap

_CENTER_DISTANCE_HEATMAP = _compute_center_distance_heatmap()


@dataclass
class Move:
    """A move in Blokus."""
    player: int            # Player index (0-3)
    piece_idx: int         # Piece index (0-20)
    orientation_idx: int   # Orientation index
    row: int               # Position row
    col: int               # Position col

    def __hash__(self):
        return hash((self.player, self.piece_idx, self.orientation_idx, self.row, self.col))

    def __eq__(self, other):
        if not isinstance(other, Move):

        return (self.player == other.player and
                self.piece_idx == other.piece_idx and
                self.orientation_idx == other.orientation_idx and
                self.row == other.row and
                self.col == other.col)


@dataclass
class PlayerState:

    is_eliminated: bool = False

class BlokusGame:
    """
    Native Python Blokus game optimized for self-play.

    Board representation:
    - board: numpy array of shape (20, 20), dtype int8
    - Values: 0 = empty, 1-4 = player color

    For fast legal move generation, we track:
    - valid_corners: set of (row, col) positions that are valid attachment points for each player
    """

    def __init__(self):
        # Board: 0 = empty, 1 = blue, 2 = yellow, 3 = red, 4 = green
        self.board = np.zeros((BOARD_SIZE, BOARD_SIZE), dtype=np.int8)

        # Player states
        self.players = [PlayerState() for _ in range(NUM_PLAYERS)]

        # Current player (0-3)
        self.current_player = BLUE

        # Turn tracking
        self.turn_number = 1
        self.consecutive_passes = 0
        self.is_game_over = False

        # For each player, track valid corner attachment points
        # Initially, each player can only attach at their starting corner
        self._valid_corners: List[Set[Tuple[int, int]]] = [
            {STARTING_CORNERS[i]} for i in range(NUM_PLAYERS)
        ]

        # Move history for potential replay
        self.move_history: List[Move] = []

    def clone(self) -> 'BlokusGame':
        """Create a deep copy of the game state."""
        game = BlokusGame.__new__(BlokusGame)
        game.board = self.board.copy()
        game.players = [p.clone() for p in self.players]
        game.current_player = self.current_player
        game.turn_number = self.turn_number
        game.consecutive_passes = self.consecutive_passes
        game.is_game_over = self.is_game_over
        game._valid_corners = [s.copy() for s in self._valid_corners]
        game.move_history = self.move_history.copy()
        return game

    def _is_within_bounds(self, row: int, col: int) -> bool:
        """Check if position is within board bounds."""
        return 0 <= row < BOARD_SIZE and 0 <= col < BOARD_SIZE

    def _get_piece_cells(self, piece: PieceDefinition, orientation_idx: int,
                         row: int, col: int) -> np.ndarray:
        """Get absolute board coordinates for piece at position."""
        return piece.get_cells(orientation_idx, row, col)

    def _cells_within_bounds(self, cells: np.ndarray) -> bool:
        """Check if all cells are within board bounds."""
        return (cells[:, 0] >= 0).all() and (cells[:, 0] < BOARD_SIZE).all() and \
               (cells[:, 1] >= 0).all() and (cells[:, 1] < BOARD_SIZE).all()

    def _cells_overlap(self, cells: np.ndarray) -> bool:
        """Check if any cell overlaps with existing pieces."""
        for r, c in cells:
            if self.board[r, c] != 0:
                return True
        return False

    def _touches_starting_corner(self, cells: np.ndarray, player: int) -> bool:
        """Check if any cell touches the player's starting corner."""
        corner = STARTING_CORNERS[player]
        for r, c in cells:
            if r == corner[0] and c == corner[1]:
                return True
        return False

    def _touches_own_edge(self, cells: np.ndarray, player: int) -> bool:
        """Check if any cell is edge-adjacent to own color (not allowed)."""
        player_color = player + 1  # Board uses 1-4 for colors
        cell_set = set(map(tuple, cells))

        for r, c in cells:
            for dr, dc in EDGE_OFFSETS:
                nr, nc = r + dr, c + dc
                if (nr, nc) in cell_set:
                    continue  # Part of the piece being placed
                if self._is_within_bounds(nr, nc) and self.board[nr, nc] == player_color:
                    return True
        return False

    def _touches_own_diagonal(self, cells: np.ndarray, player: int) -> bool:
        """Check if any cell is diagonally adjacent to own color (required after first move)."""
        player_color = player + 1
        cell_set = set(map(tuple, cells))

        for r, c in cells:
            for dr, dc in DIAGONAL_OFFSETS:
                nr, nc = r + dr, c + dc
                if (nr, nc) in cell_set:
                    continue
                if self._is_within_bounds(nr, nc) and self.board[nr, nc] == player_color:
                    return True
        return False

    def is_valid_move(self, move: Move) -> bool:
        """Check if a move is valid."""
        player = move.player
        piece_idx = move.piece_idx
        player_state = self.players[player]

        # Check if piece is available
        if piece_idx not in player_state.remaining_pieces:
            return False

        piece = PIECE_BY_INDEX[piece_idx]

        # Check orientation is valid
        if move.orientation_idx < 0 or move.orientation_idx >= piece.num_orientations:
            return False

        # Get absolute cells
        cells = self._get_piece_cells(piece, move.orientation_idx, move.row, move.col)

        # Check bounds
        if not self._cells_within_bounds(cells):
            return False

        # Check overlap
        if self._cells_overlap(cells):
            return False

        # Check edge adjacency to own color (not allowed)
        if self._touches_own_edge(cells, player):
            return False

        # First move must cover starting corner
        if not player_state.has_played:
            if not self._touches_starting_corner(cells, player):
                return False
        else:
            # Subsequent moves must touch own color diagonally
            if not self._touches_own_diagonal(cells, player):
                return False

        return True

    def get_legal_moves(self, player: Optional[int] = None) -> List[Move]:
        """
        Get all legal moves for the specified player (or current player).

        Optimized by checking only positions near valid corners.
        """
        if player is None:
            player = self.current_player

        if self.is_game_over:
            return []

        player_state = self.players[player]
        if player_state.is_eliminated:
            return []

        moves = []

        # First move: must cover starting corner
        if not player_state.has_played:
            corner = STARTING_CORNERS[player]
            for piece_idx in player_state.remaining_pieces:
                piece = PIECE_BY_INDEX[piece_idx]
                for orient_idx in range(piece.num_orientations):
                    # Try positions where the piece covers the corner
                    shape = piece.orientations[orient_idx]
                    for cell_r, cell_c in shape:
                        # Position such that this cell lands on corner
                        pos_r = corner[0] - cell_r
                        pos_c = corner[1] - cell_c
                        move = Move(player, piece_idx, orient_idx, pos_r, pos_c)
                        if self.is_valid_move(move):
                            moves.append(move)
        else:
            # Subsequent moves: must touch own color diagonally
            # Use valid corners for faster search
            valid_corners = self._valid_corners[player]

            for piece_idx in player_state.remaining_pieces:
                piece = PIECE_BY_INDEX[piece_idx]
                for orient_idx in range(piece.num_orientations):
                    shape = piece.orientations[orient_idx]
                    # For each valid corner, try positions where a piece cell lands on it
                    for corner_r, corner_c in valid_corners:
                        for cell_r, cell_c in shape:
                            pos_r = corner_r - cell_r
                            pos_c = corner_c - cell_c
                            move = Move(player, piece_idx, orient_idx, pos_r, pos_c)
                            if self.is_valid_move(move):
                                moves.append(move)

        return moves

    def _update_valid_corners(self, cells: np.ndarray, player: int):
        """Update valid corner attachment points after placing a piece."""
        player_color = player + 1

        # Remove corners that are now occupied
        for p in range(NUM_PLAYERS):
            self._valid_corners[p] -= set(map(tuple, cells))

        # Also remove corners that are now edge-adjacent to the placed piece
        for r, c in cells:
            for dr, dc in EDGE_OFFSETS:
                nr, nc = r + dr, c + dc
                if self._is_within_bounds(nr, nc):
                    self._valid_corners[player].discard((nr, nc))

        # Add new diagonal corners for the placing player
        for r, c in cells:
            for dr, dc in DIAGONAL_OFFSETS:
                nr, nc = r + dr, c + dc
                if not self._is_within_bounds(nr, nc):
                    continue
                if self.board[nr, nc] != 0:
                    continue  # Already occupied

                # Check it's not edge-adjacent to own color
                is_edge_adjacent = False
                for edr, edc in EDGE_OFFSETS:
                    er, ec = nr + edr, nc + edc
                    if self._is_within_bounds(er, ec) and self.board[er, ec] == player_color:
                        is_edge_adjacent = True
                        break

                if not is_edge_adjacent:
                    self._valid_corners[player].add((nr, nc))

    def make_move(self, move: Move) -> bool:
        """
        Apply a move to the game.
        Returns True if successful.
        """
        if self.is_game_over:
            return False

        if move.player != self.current_player:
            return False

        if not self.is_valid_move(move):
            return False

        player = move.player
        player_state = self.players[player]
        piece = PIECE_BY_INDEX[move.piece_idx]

        # Get cells and place on board
        cells = self._get_piece_cells(piece, move.orientation_idx, move.row, move.col)
        player_color = player + 1

        for r, c in cells:
            self.board[r, c] = player_color

        # Update player state
        player_state.remaining_pieces.remove(move.piece_idx)
        player_state.has_played = True
        player_state.score += piece.size

        # Update valid corners
        self._update_valid_corners(cells, player)

        # Record move
        self.move_history.append(move)

        # Reset consecutive passes
        self.consecutive_passes = 0

        # Advance turn
        self._advance_turn()

        return True

    def pass_turn(self) -> bool:
        """
        Pass the current player's turn (only allowed if no legal moves).
        Returns True if successful.
        """
        if self.is_game_over:
            return False

        # Check if player has any legal moves
        if self.get_legal_moves():
            return False  # Cannot pass if moves are available

        player_state = self.players[self.current_player]
        player_state.is_eliminated = True
        self.consecutive_passes += 1

        self._advance_turn()
        return True

    def _advance_turn(self):
        """Advance to the next player."""
        next_player = (self.current_player + 1) % NUM_PLAYERS
        checked = 0

        while checked < NUM_PLAYERS:
            if not self.players[next_player].is_eliminated:
                break
            next_player = (next_player + 1) % NUM_PLAYERS
            checked += 1

        # Check if game is over
        if checked >= NUM_PLAYERS or self.consecutive_passes >= NUM_PLAYERS:
            self.is_game_over = True
            return

        self.current_player = next_player
        self.turn_number += 1

    def auto_pass_if_needed(self) -> bool:
        """Auto-pass for players with no moves. Returns True if passed."""
        if self.is_game_over:
            return False

        if not self.get_legal_moves():
            return self.pass_turn()
        return False

    def get_scores(self) -> List[int]:
        """
        Get final scores for all players.
        Score = cells placed + bonuses - remaining piece cells
        """
        scores = []
        for i, player_state in enumerate(self.players):
            score = player_state.score

            # Check if all pieces used
            if len(player_state.remaining_pieces) == 0:
                score += 15  # Bonus for using all pieces

                # Additional bonus if last piece was monomino
                if self.move_history:
                    last_move = None
                    for m in reversed(self.move_history):
                        if m.player == i:
                            last_move = m
                            break
                    if last_move and last_move.piece_idx == 0:  # Monomino
                        score += 5
            else:
                # Subtract remaining piece sizes
                for piece_idx in player_state.remaining_pieces:
                    score -= PIECE_BY_INDEX[piece_idx].size

            scores.append(score)

        return scores

    def get_winner(self) -> Optional[int]:
        """Get the winning player index, or None if tie."""
        scores = self.get_scores()
        max_score = max(scores)
        winners = [i for i, s in enumerate(scores) if s == max_score]

        if len(winners) == 1:
            return winners[0]
        return None  # Tie

    def get_state_for_neural_net(self, perspective_player: Optional[int] = None) -> Tuple[np.ndarray, np.ndarray]:
        """
        Get state encoding for neural network input.

        Returns:
        - board_tensor: shape (10, 20, 20)
            Channel 0: Current player's pieces
            Channel 1-3: Other players' pieces (in turn order from current)
            Channel 4: Valid corner attachment points for current player
            Channel 5: Empty cells
            Channel 6: Turn number (normalized, broadcast)
            Channel 7: Current player indicator (one-hot style broadcast)
            Channel 8: Center distance (cells closer to center have higher values)
            Channel 9: Game phase (0.0=early, 0.5=mid, 1.0=late)

        - pieces_vector: shape (84,) - remaining pieces for all 4 players
            21 pieces * 4 players = 84 binary values
        """
        if perspective_player is None:
            perspective_player = self.current_player

        board_tensor = np.zeros((10, BOARD_SIZE, BOARD_SIZE), dtype=np.float32)

        # Player order from perspective of current player
        player_order = [(perspective_player + i) % NUM_PLAYERS for i in range(NUM_PLAYERS)]

        # Channel 0: Current player's pieces
        board_tensor[0] = (self.board == player_order[0] + 1).astype(np.float32)

        # Channels 1-3: Other players' pieces
        for i in range(1, NUM_PLAYERS):
            board_tensor[i] = (self.board == player_order[i] + 1).astype(np.float32)

        # Channel 4: Valid corners for current player
        for r, c in self._valid_corners[perspective_player]:
            board_tensor[4, r, c] = 1.0

        # Channel 5: Empty cells
        board_tensor[5] = (self.board == 0).astype(np.float32)

        # Channel 6: Turn number (normalized to 0-1, max ~80 turns)
        board_tensor[6] = min(self.turn_number / 80.0, 1.0)

        # Channel 7: Current player indicator (0.25 * (player + 1))
        board_tensor[7] = (perspective_player + 1) / 4.0

        # Channel 8: Center distance heatmap
        # Cells closer to center (9.5, 9.5) have higher values
        # Max distance from center to corner is sqrt(9.5^2 + 9.5^2) ≈ 13.44
        board_tensor[8] = _CENTER_DISTANCE_HEATMAP

        # Channel 9: Game phase (based on total pieces remaining)
        # 84 total pieces at start (21 per player * 4 players)
        # Early: 0-28 pieces played (0.0), Mid: 29-56 pieces played (0.5), Late: 57-84 (1.0)
        total_remaining = sum(len(p.remaining_pieces) for p in self.players)
        pieces_played = 84 - total_remaining
        if pieces_played <= 28:
            game_phase = 0.0  # Early game
        elif pieces_played <= 56:
            game_phase = 0.5  # Mid game
        else:
            game_phase = 1.0  # Late game
        board_tensor[9] = game_phase

        # Remaining pieces vector
        pieces_vector = np.zeros(NUM_PIECES * NUM_PLAYERS, dtype=np.float32)
        for i, player_idx in enumerate(player_order):
            for piece_idx in self.players[player_idx].remaining_pieces:
                pieces_vector[i * NUM_PIECES + piece_idx] = 1.0

        return board_tensor, pieces_vector

    def __repr__(self):
        return f"BlokusGame(turn={self.turn_number}, player={PLAYER_NAMES[self.current_player]}, over={self.is_game_over})"


def play_random_game() -> BlokusGame:
    """Play a random game and return the final state."""
    import random

    game = BlokusGame()

    while not game.is_game_over:
        moves = game.get_legal_moves()
        if moves:
            move = random.choice(moves)
            game.make_move(move)
        else:
            game.pass_turn()

    return game

"""
Optimized Blokus game implementation with fast legal move generation.

Key optimizations:
1. Precomputed move tables: For each (piece, orientation), precompute which cell
   offsets can land on a corner to create valid diagonal adjacency
2. Vectorized validation: Check multiple moves at once using numpy operations
3. Move deduplication: Use set to avoid checking same move twice
4. Early exit validation: Order checks from cheapest to most expensive
5. Cached piece data: All piece cells stored as numpy arrays
6. Numba JIT compilation for hot validation loop
"""

from typing import List, Tuple, Optional, Set, Dict, FrozenSet
from dataclasses import dataclass, field
from functools import lru_cache
import numpy as np
from numba import njit, types
from numba.typed import List as NumbaList

from .pieces import (
    PIECE_CATALOG,
    PIECE_BY_INDEX,
    NUM_PIECES,
    PieceDefinition,
)

# Board constants
BOARD_SIZE = 20
NUM_PLAYERS = 4

# Player colors (indices)
BLUE = 0
YELLOW = 1
RED = 2
GREEN = 3

PLAYER_NAMES = ['blue', 'yellow', 'red', 'green']

# Starting corners for each player (row, col)
STARTING_CORNERS = {
    BLUE: (0, 0),
    YELLOW: (0, 19),
    RED: (19, 19),
    GREEN: (19, 0),
}

# Direction offsets
EDGE_OFFSETS = np.array([[-1, 0], [1, 0], [0, -1], [0, 1]], dtype=np.int8)
DIAGONAL_OFFSETS = np.array([[-1, -1], [-1, 1], [1, -1], [1, 1]], dtype=np.int8)

# Precomputed center distance heatmap for neural network input
# Cells closer to center (9.5, 9.5) have higher values (0-1 normalized)
def _compute_center_distance_heatmap() -> np.ndarray:
    """Compute center distance heatmap once at module load."""
    heatmap = np.zeros((BOARD_SIZE, BOARD_SIZE), dtype=np.float32)
    center = (BOARD_SIZE - 1) / 2.0  # 9.5 for 20x20 board
    max_dist = np.sqrt(2) * center  # ~13.44
    for r in range(BOARD_SIZE):
        for c in range(BOARD_SIZE):
            dist = np.sqrt((r - center) ** 2 + (c - center) ** 2)
            heatmap[r, c] = 1.0 - (dist / max_dist)  # Closer to center = higher value
    return heatmap

_CENTER_DISTANCE_HEATMAP = _compute_center_distance_heatmap()


@njit(cache=True)
def _validate_move_numba(
    board: np.ndarray,
    cells: np.ndarray,
    row: int,
    col: int,
    player_color: int,
    has_played: bool,
    corner_r: int,
    corner_c: int,
) -> bool:
    """
    Numba-compiled move validation.
    Returns True if move is valid.
    """
    num_cells = cells.shape[0]

    # Check bounds and overlap, collect absolute positions
    abs_rows = np.empty(num_cells, dtype=np.int32)
    abs_cols = np.empty(num_cells, dtype=np.int32)

    for k in range(num_cells):
        r = cells[k, 0] + row
        c = cells[k, 1] + col

        if r < 0 or r >= 20 or c < 0 or c >= 20:
            return False

        if board[r, c] != 0:
            return False

        abs_rows[k] = r
        abs_cols[k] = c

    # Build set of relative cells for checking
    # (Using simple array search since numba doesn't support sets well)

    # Edge adjacency check
    edge_offsets = np.array([[-1, 0], [1, 0], [0, -1], [0, 1]], dtype=np.int32)

    for k in range(num_cells):
        r, c = abs_rows[k], abs_cols[k]

        for e in range(4):
            nr = r + edge_offsets[e, 0]
            nc = c + edge_offsets[e, 1]

            # Check if neighbor is part of piece
            is_piece = False
            for m in range(num_cells):
                if abs_rows[m] == nr and abs_cols[m] == nc:
                    is_piece = True
                    break

            if is_piece:
                continue

            if 0 <= nr < 20 and 0 <= nc < 20:
                if board[nr, nc] == player_color:
                    return False

    # Connection check
    if not has_played:
        # Must cover starting corner
        covers = False
        for k in range(num_cells):
            if abs_rows[k] == corner_r and abs_cols[k] == corner_c:
                covers = True
                break
        if not covers:
            return False
    else:
        # Must have diagonal adjacency
        diag_offsets = np.array([[-1, -1], [-1, 1], [1, -1], [1, 1]], dtype=np.int32)
        has_diag = False

        for k in range(num_cells):
            if has_diag:
                break
            r, c = abs_rows[k], abs_cols[k]

            for d in range(4):
                nr = r + diag_offsets[d, 0]
                nc = c + diag_offsets[d, 1]

                # Check if neighbor is part of piece
                is_piece = False
                for m in range(num_cells):
                    if abs_rows[m] == nr and abs_cols[m] == nc:
                        is_piece = True
                        break

                if is_piece:
                    continue

                if 0 <= nr < 20 and 0 <= nc < 20:
                    if board[nr, nc] == player_color:
                        has_diag = True
                        break

        if not has_diag:
            return False

    return True


@njit(cache=True)
def _batch_validate_numba(
    board: np.ndarray,
    candidates: np.ndarray,  # (N, 4) - piece_idx, orient_idx, row, col
    remaining_mask: np.ndarray,  # (21,) bool - which pieces available
    num_orients: np.ndarray,  # (21,) - num orientations per piece
    all_cells: np.ndarray,  # (21, 8, 5, 2) - cell offsets
    cell_counts: np.ndarray,  # (21,) - num cells per piece
    player_color: int,
    has_played: bool,
    corner_r: int,
    corner_c: int,
) -> np.ndarray:
    """
    Batch validate moves with numba.
    """
    n = candidates.shape[0]
    results = np.ones(n, dtype=np.bool_)

    for i in range(n):
        piece_idx = candidates[i, 0]
        orient_idx = candidates[i, 1]
        row = candidates[i, 2]
        col = candidates[i, 3]

        # Check piece available
        if not remaining_mask[piece_idx]:
            results[i] = False
            continue

        # Check valid orientation
        if orient_idx >= num_orients[piece_idx]:
            results[i] = False
            continue

        # Get cells for this piece/orientation
        num_cells = cell_counts[piece_idx]
        cells = all_cells[piece_idx, orient_idx, :num_cells, :]

        # Validate the move
        if not _validate_move_numba(
            board, cells, row, col, player_color, has_played, corner_r, corner_c
        ):
            results[i] = False

    return results


@dataclass
class Move:
    """A move in Blokus."""
    player: int
    piece_idx: int
    orientation_idx: int
    row: int
    col: int

    def __hash__(self):
        return hash((self.player, self.piece_idx, self.orientation_idx, self.row, self.col))

    def __eq__(self, other):
        if not isinstance(other, Move):
            return False
        return (self.player == other.player and
                self.piece_idx == other.piece_idx and
                self.orientation_idx == other.orientation_idx and
                self.row == other.row and
                self.col == other.col)

class PrecomputedMoveData:
    """
    Precomputed data for fast move generation.

    For each (piece, orientation), stores:
    - cells: numpy array of cell offsets
    - cell_set: frozenset for fast membership testing
    - For each cell index, the position offset to place piece such that
      this cell lands on a given corner
    """

    _instance = None

    @classmethod
    def get(cls) -> 'PrecomputedMoveData':
        if cls._instance is None:
            cls._instance = cls()
        return cls._instance

    def __init__(self):
        self.piece_cells: List[List[np.ndarray]] = []  # [piece_idx][orient_idx] -> (N, 2)
        self.piece_cell_sets: List[List[FrozenSet[Tuple[int, int]]]] = []
        self.piece_sizes: List[int] = []
        self.piece_num_orients: List[int] = []

        # For starting corner moves: (piece_idx, orient_idx, pos_row, pos_col) tuples
        # that place a piece covering each starting corner
        self.starting_corner_moves: Dict[Tuple[int, int], List[Tuple[int, int, int, int]]] = {}

        # Numba-compatible arrays
        self.all_cells: np.ndarray = None  # (21, 8, 5, 2)
        self.cell_counts: np.ndarray = None  # (21,)
        self.num_orients_arr: np.ndarray = None  # (21,)

        self._build()

    def _build(self):
        """Build all precomputed data."""
        # Initialize numba arrays
        self.all_cells = np.zeros((NUM_PIECES, 8, 5, 2), dtype=np.int32)
        self.cell_counts = np.zeros(NUM_PIECES, dtype=np.int32)
        self.num_orients_arr = np.zeros(NUM_PIECES, dtype=np.int32)

        for piece_idx, piece in enumerate(PIECE_CATALOG):
            self.piece_sizes.append(piece.size)
            self.piece_num_orients.append(piece.num_orientations)
            self.cell_counts[piece_idx] = piece.size
            self.num_orients_arr[piece_idx] = piece.num_orientations

            piece_cells = []
            piece_cell_sets = []

            for orient_idx in range(piece.num_orientations):
                cells = np.array(piece.orientations[orient_idx], dtype=np.int8)
                piece_cells.append(cells)
                piece_cell_sets.append(frozenset(map(tuple, cells)))

                # Store in numba array
                for k, (r, c) in enumerate(piece.orientations[orient_idx]):
                    self.all_cells[piece_idx, orient_idx, k, 0] = r
                    self.all_cells[piece_idx, orient_idx, k, 1] = c

            # Pad with empty arrays for unused orientations
            while len(piece_cells) < 8:
                piece_cells.append(np.zeros((0, 2), dtype=np.int8))
                piece_cell_sets.append(frozenset())

            self.piece_cells.append(piece_cells)
            self.piece_cell_sets.append(piece_cell_sets)

        # Precompute starting corner moves
        for player in range(NUM_PLAYERS):
            corner = STARTING_CORNERS[player]
            moves = []

            for piece_idx in range(NUM_PIECES):
                for orient_idx in range(self.piece_num_orients[piece_idx]):
                    cells = self.piece_cells[piece_idx][orient_idx]
                    for cell in cells:
                        # Position such that this cell lands on corner
                        pos_r = corner[0] - cell[0]
                        pos_c = corner[1] - cell[1]
                        moves.append((piece_idx, orient_idx, pos_r, pos_c))

            self.starting_corner_moves[corner] = moves


class BlokusGameOptimized:
    """
    Optimized Blokus game with fast legal move generation.
    """

    def __init__(self):
        self.board = np.zeros((BOARD_SIZE, BOARD_SIZE), dtype=np.int8)
        self.players = [PlayerState() for _ in range(NUM_PLAYERS)]
        self.current_player = BLUE
        self.turn_number = 1
        self.consecutive_passes = 0
        self.is_game_over = False

        self._valid_corners: List[Set[Tuple[int, int]]] = [
            {STARTING_CORNERS[i]} for i in range(NUM_PLAYERS)
        ]

        self.move_history: List[Move] = []
        self._precomputed = PrecomputedMoveData.get()

        # Cache for legal moves (invalidated on state change)
        self._legal_moves_cache: Optional[List[Move]] = None
        self._legal_moves_cache_player: Optional[int] = None

    def clone(self) -> 'BlokusGameOptimized':
        """Create a deep copy of the game state."""
        game = BlokusGameOptimized.__new__(BlokusGameOptimized)
        game.board = self.board.copy()
        game.players = [p.clone() for p in self.players]
        game.current_player = self.current_player
        game.turn_number = self.turn_number
        game.consecutive_passes = self.consecutive_passes
        game.is_game_over = self.is_game_over
        game._valid_corners = [s.copy() for s in self._valid_corners]
        game.move_history = []  # Don't copy history - not needed for MCTS
        game._precomputed = self._precomputed
        game._legal_moves_cache = None
        game._legal_moves_cache_player = None
        return game

    def _invalidate_cache(self):
        """Invalidate the legal moves cache."""
        self._legal_moves_cache = None
        self._legal_moves_cache_player = None

    def is_valid_move_fast(self, piece_idx: int, orient_idx: int, row: int, col: int, player: int) -> bool:
        """
        Fast move validation with early exits.
        Checks are ordered from cheapest to most expensive.
        """
        player_state = self.players[player]

        # Check 1: Piece available (O(1) set lookup)
        if piece_idx not in player_state.remaining_pieces:
            return False

        # Check 2: Valid orientation
        if orient_idx >= self._precomputed.piece_num_orients[piece_idx]:
            return False

        # Get cells
        cells = self._precomputed.piece_cells[piece_idx][orient_idx]
        num_cells = len(cells)

        # Check 3: Bounds check (vectorized)
        abs_rows = cells[:, 0] + row
        abs_cols = cells[:, 1] + col

        if (abs_rows < 0).any() or (abs_rows >= BOARD_SIZE).any():
            return False
        if (abs_cols < 0).any() or (abs_cols >= BOARD_SIZE).any():
            return False

        # Check 4: Overlap check (vectorized board lookup)
        board_values = self.board[abs_rows, abs_cols]
        if (board_values != 0).any():
            return False

        player_color = player + 1

        # Check 5: Edge adjacency to own color (not allowed)
        # Build set of piece cells for fast lookup
        cell_set = self._precomputed.piece_cell_sets[piece_idx][orient_idx]

        for i in range(num_cells):
            r, c = abs_rows[i], abs_cols[i]
            for dr, dc in EDGE_OFFSETS:
                nr, nc = r + dr, c + dc
                # Skip if neighbor is part of the piece
                rel_nr, rel_nc = nr - row, nc - col
                if (rel_nr, rel_nc) in cell_set:
                    continue
                if 0 <= nr < BOARD_SIZE and 0 <= nc < BOARD_SIZE:
                    if self.board[nr, nc] == player_color:
                        return False

        # Check 6: Connection rule
        if not player_state.has_played:
            # First move: must cover starting corner
            corner = STARTING_CORNERS[player]
            covers_corner = False
            for i in range(num_cells):
                if abs_rows[i] == corner[0] and abs_cols[i] == corner[1]:
                    covers_corner = True
                    break
            if not covers_corner:
                return False
        else:
            # Subsequent moves: must have diagonal adjacency to own color
            has_diagonal = False
            for i in range(num_cells):
                r, c = abs_rows[i], abs_cols[i]
                for dr, dc in DIAGONAL_OFFSETS:
                    nr, nc = r + dr, c + dc
                    rel_nr, rel_nc = nr - row, nc - col
                    if (rel_nr, rel_nc) in cell_set:
                        continue
                    if 0 <= nr < BOARD_SIZE and 0 <= nc < BOARD_SIZE:
                        if self.board[nr, nc] == player_color:
                            has_diagonal = True
                            break
                if has_diagonal:
                    break
            if not has_diagonal:
                return False

        return True

    def get_legal_moves(self, player: Optional[int] = None) -> List[Move]:
        """
        Get all legal moves with caching and batch validation.
        """
        if player is None:
            player = self.current_player

        # Check cache
        if self._legal_moves_cache is not None and self._legal_moves_cache_player == player:
            return self._legal_moves_cache

        if self.is_game_over:
            return []

        player_state = self.players[player]
        if player_state.is_eliminated:
            return []

        remaining = player_state.remaining_pieces

        # Collect all candidate moves first, then batch validate
        candidates: List[Tuple[int, int, int, int]] = []
        seen: Set[Tuple[int, int, int, int]] = set()

        if not player_state.has_played:
            corner = STARTING_CORNERS[player]
            for piece_idx, orient_idx, pos_r, pos_c in self._precomputed.starting_corner_moves[corner]:
                if piece_idx not in remaining:
                    continue
                key = (piece_idx, orient_idx, pos_r, pos_c)
                if key not in seen:
                    seen.add(key)
                    candidates.append(key)
        else:
            valid_corners = list(self._valid_corners[player])

            for piece_idx in remaining:
                num_orients = self._precomputed.piece_num_orients[piece_idx]

                for orient_idx in range(num_orients):
                    cells = self._precomputed.piece_cells[piece_idx][orient_idx]

                    for corner_r, corner_c in valid_corners:
                        for cell in cells:
                            pos_r = corner_r - cell[0]
                            pos_c = corner_c - cell[1]

                            key = (piece_idx, orient_idx, pos_r, pos_c)
                            if key not in seen:
                                seen.add(key)
                                candidates.append(key)

        # Batch validate all candidates
        if not candidates:
            self._legal_moves_cache = []
            self._legal_moves_cache_player = player
            return []

        valid_mask = self._batch_validate_moves(candidates, player)
        moves = [
            Move(player, c[0], c[1], c[2], c[3])
            for c, valid in zip(candidates, valid_mask)
            if valid
        ]

        self._legal_moves_cache = moves
        self._legal_moves_cache_player = player
        return moves

    def _batch_validate_moves(
        self,
        candidates: List[Tuple[int, int, int, int]],
        player: int
    ) -> List[bool]:
        """
        Validate multiple moves using numba-compiled function.
        """
        n = len(candidates)
        if n == 0:
            return []

        player_state = self.players[player]
        player_color = player + 1
        has_played = player_state.has_played
        remaining = player_state.remaining_pieces
        corner = STARTING_CORNERS[player]
        precomputed = self._precomputed

        # Build remaining mask
        remaining_mask = np.zeros(NUM_PIECES, dtype=np.bool_)
        for p in remaining:
            remaining_mask[p] = True

        # Convert candidates to numpy array
        candidates_arr = np.array(candidates, dtype=np.int32)

        # Call numba function
        results = _batch_validate_numba(
            self.board.astype(np.int32),
            candidates_arr,
            remaining_mask,
            precomputed.num_orients_arr,
            precomputed.all_cells,
            precomputed.cell_counts,
            player_color,
            has_played,
            corner[0],
            corner[1],
        )

        return results.tolist()

    def is_valid_move(self, move: Move) -> bool:
        """Check if a move is valid."""
        return self.is_valid_move_fast(
            move.piece_idx, move.orientation_idx, move.row, move.col, move.player
        )

    def _update_valid_corners(self, cells: np.ndarray, player: int):
        """Update valid corner attachment points after placing a piece."""
        player_color = player + 1
        cell_tuples = set(map(tuple, cells))

        # Remove occupied cells from all players' corners
        for p in range(NUM_PLAYERS):
            self._valid_corners[p] -= cell_tuples

        # Remove edge-adjacent cells from placing player's corners
        for r, c in cells:
            for dr, dc in EDGE_OFFSETS:
                nr, nc = int(r + dr), int(c + dc)
                if 0 <= nr < BOARD_SIZE and 0 <= nc < BOARD_SIZE:
                    self._valid_corners[player].discard((nr, nc))

        # Add diagonal cells as new valid corners
        for r, c in cells:
            for dr, dc in DIAGONAL_OFFSETS:
                nr, nc = int(r + dr), int(c + dc)
                if not (0 <= nr < BOARD_SIZE and 0 <= nc < BOARD_SIZE):
                    continue
                if self.board[nr, nc] != 0:
                    continue

                # Check not edge-adjacent to own color
                is_edge_adjacent = False
                for edr, edc in EDGE_OFFSETS:
                    er, ec = nr + edr, nc + edc
                    if 0 <= er < BOARD_SIZE and 0 <= ec < BOARD_SIZE:
                        if self.board[er, ec] == player_color:
                            is_edge_adjacent = True
                            break

                if not is_edge_adjacent:
                    self._valid_corners[player].add((nr, nc))

    def make_move(self, move: Move) -> bool:
        """Apply a move to the game."""
        if self.is_game_over:
            return False

        if move.player != self.current_player:
            return False

        if not self.is_valid_move(move):
            return False

        player = move.player
        player_state = self.players[player]

        # Get cells and place
        cells = self._precomputed.piece_cells[move.piece_idx][move.orientation_idx]
        abs_cells = cells + np.array([[move.row, move.col]], dtype=np.int8)

        player_color = player + 1
        for r, c in abs_cells:
            self.board[r, c] = player_color

        # Update player state
        player_state.remaining_pieces.remove(move.piece_idx)
        player_state.has_played = True
        player_state.score += len(cells)

        # Update valid corners
        self._update_valid_corners(abs_cells, player)

        # Record and reset
        self.move_history.append(move)
        self.consecutive_passes = 0
        self._invalidate_cache()

        # Advance turn
        self._advance_turn()

        return True

    def pass_turn(self) -> bool:
        """Pass the current player's turn."""
        if self.is_game_over:
            return False

        if self.get_legal_moves():
            return False

        self.players[self.current_player].is_eliminated = True
        self.consecutive_passes += 1
        self._invalidate_cache()

        self._advance_turn()
        return True

    def _advance_turn(self):
        """Advance to the next player."""
        next_player = (self.current_player + 1) % NUM_PLAYERS
        checked = 0

        while checked < NUM_PLAYERS:
            if not self.players[next_player].is_eliminated:
                break
            next_player = (next_player + 1) % NUM_PLAYERS
            checked += 1

        if checked >= NUM_PLAYERS or self.consecutive_passes >= NUM_PLAYERS:
            self.is_game_over = True
            return

        self.current_player = next_player
        self.turn_number += 1
        self._invalidate_cache()

    def get_scores(self) -> List[int]:
        """Get final scores for all players."""
        scores = []
        for i, player_state in enumerate(self.players):
            score = player_state.score

            if len(player_state.remaining_pieces) == 0:
                score += 15
                if self.move_history:
                    for m in reversed(self.move_history):
                        if m.player == i:
                            if m.piece_idx == 0:  # Monomino
                                score += 5
                            break
            else:
                for piece_idx in player_state.remaining_pieces:
                    score -= self._precomputed.piece_sizes[piece_idx]

            scores.append(score)

        return scores

    def get_winner(self) -> Optional[int]:
        """Get the winning player index, or None if tie."""
        scores = self.get_scores()
        max_score = max(scores)
        winners = [i for i, s in enumerate(scores) if s == max_score]

        if len(winners) == 1:
            return winners[0]
        return None

    def get_state_for_neural_net(self, perspective_player: Optional[int] = None) -> Tuple[np.ndarray, np.ndarray]:
        """
        Get state encoding for neural network input.

        Returns:
        - board_tensor: shape (10, 20, 20)
            Channel 0: Current player's pieces
            Channel 1-3: Other players' pieces (in turn order from current)
            Channel 4: Valid corner attachment points for current player
            Channel 5: Empty cells
            Channel 6: Turn number (normalized, broadcast)
            Channel 7: Current player indicator (one-hot style broadcast)
            Channel 8: Center distance (cells closer to center have higher values)
            Channel 9: Game phase (0.0=early, 0.5=mid, 1.0=late)

        - pieces_vector: shape (84,) - remaining pieces for all 4 players
        """
        if perspective_player is None:
            perspective_player = self.current_player

        board_tensor = np.zeros((10, BOARD_SIZE, BOARD_SIZE), dtype=np.float32)

        player_order = [(perspective_player + i) % NUM_PLAYERS for i in range(NUM_PLAYERS)]

        # Channels 0-3: Player pieces
        for i in range(NUM_PLAYERS):
            board_tensor[i] = (self.board == player_order[i] + 1).astype(np.float32)

        # Channel 4: Valid corners
        for r, c in self._valid_corners[perspective_player]:
            board_tensor[4, r, c] = 1.0

        # Channel 5: Empty cells
        board_tensor[5] = (self.board == 0).astype(np.float32)

        # Channel 6: Turn number
        board_tensor[6] = min(self.turn_number / 80.0, 1.0)

        # Channel 7: Current player
        board_tensor[7] = (perspective_player + 1) / 4.0

        # Channel 8: Center distance heatmap
        board_tensor[8] = _CENTER_DISTANCE_HEATMAP

        # Channel 9: Game phase (based on total pieces remaining)
        # 84 total pieces at start (21 per player * 4 players)
        total_remaining = sum(len(p.remaining_pieces) for p in self.players)
        pieces_played = 84 - total_remaining
        if pieces_played <= 28:
            game_phase = 0.0  # Early game
        elif pieces_played <= 56:
            game_phase = 0.5  # Mid game
        else:
            game_phase = 1.0  # Late game
        board_tensor[9] = game_phase

        # Pieces vector
        pieces_vector = np.zeros(NUM_PIECES * NUM_PLAYERS, dtype=np.float32)
        for i, player_idx in enumerate(player_order):
            for piece_idx in self.players[player_idx].remaining_pieces:
                pieces_vector[i * NUM_PIECES + piece_idx] = 1.0

        return board_tensor, pieces_vector

"""

All 21 standard Blokus pieces with pre-computed orientations.

from typing import List, Tuple, Set
import numpy as np

# Coordinate type: (row, col) relative to piece origin
Coord = Tuple[int, int]
PieceShape = List[Coord]


# Base shapes for all 21 Blokus pieces
# Coordinates are [row, col] relative to top-left of bounding box
BASE_PIECES = [
    # 1 cell (monomino)
    ('1', [(0, 0)]),

    # 2 cells (domino)
    ('2', [(0, 0), (0, 1)]),

    # 3 cells (triominoes)
    ('I3', [(0, 0), (0, 1), (0, 2)]),
    ('V3', [(0, 0), (0, 1), (1, 0)]),

    # 4 cells (tetrominoes)
    ('I4', [(0, 0), (0, 1), (0, 2), (0, 3)]),
    ('O4', [(0, 0), (0, 1), (1, 0), (1, 1)]),
    ('T4', [(0, 0), (0, 1), (0, 2), (1, 1)]),
    ('L4', [(0, 0), (1, 0), (2, 0), (2, 1)]),
    ('S4', [(0, 1), (0, 2), (1, 0), (1, 1)]),

    # 5 cells (pentominoes)
    ('F', [(0, 1), (0, 2), (1, 0), (1, 1), (2, 1)]),
    ('I5', [(0, 0), (0, 1), (0, 2), (0, 3), (0, 4)]),
    ('L5', [(0, 0), (1, 0), (2, 0), (3, 0), (3, 1)]),
    ('N', [(0, 0), (0, 1), (1, 1), (1, 2), (1, 3)]),
    ('P', [(0, 0), (0, 1), (1, 0), (1, 1), (2, 0)]),
    ('T5', [(0, 0), (0, 1), (0, 2), (1, 1), (2, 1)]),
    ('U', [(0, 0), (0, 2), (1, 0), (1, 1), (1, 2)]),
    ('V', [(0, 0), (1, 0), (2, 0), (2, 1), (2, 2)]),
    ('W', [(0, 0), (1, 0), (1, 1), (2, 1), (2, 2)]),
    ('X', [(0, 1), (1, 0), (1, 1), (1, 2), (2, 1)]),
    ('Y', [(0, 1), (1, 0), (1, 1), (1, 2), (1, 3)]),
    ('Z', [(0, 0), (0, 1), (1, 1), (2, 1), (2, 2)]),
]


def normalize_shape(shape: PieceShape) -> PieceShape:
    """
    Normalize a shape by translating it so the minimum row and col are 0,
    then sorting coordinates for consistent comparison.
    """
    if not shape:
        return []

    min_row = min(r for r, c in shape)
    min_col = min(c for r, c in shape)

    normalized = [(r - min_row, c - min_col) for r, c in shape]
    normalized.sort()
    return normalized


def rotate_shape(shape: PieceShape) -> PieceShape:
    """
    Rotate shape 90 degrees clockwise.
    [row, col] -> [col, -row] (then normalize)
    """
    rotated = [(c, -r) for r, c in shape]
    return normalize_shape(rotated)


def flip_shape(shape: PieceShape) -> PieceShape:
    """
    Flip shape horizontally (mirror).
    [row, col] -> [row, -col] (then normalize)
    """
    flipped = [(r, -c) for r, c in shape]
    return normalize_shape(flipped)


def shape_to_tuple(shape: PieceShape) -> Tuple[Coord, ...]:
    """Convert shape to tuple for hashing."""
    return tuple(shape)


def generate_orientations(base_shape: PieceShape) -> List[PieceShape]:
    """
    Generate all unique orientations (up to 8) for a shape.
    """
    seen: Set[Tuple[Coord, ...]] = set()
    orientations: List[PieceShape] = []

    current = normalize_shape(base_shape)

    # Try all 4 rotations
    for _ in range(4):
        key = shape_to_tuple(current)
        if key not in seen:
            seen.add(key)
            orientations.append(current)
        current = rotate_shape(current)

    # Flip and try all 4 rotations again
    current = flip_shape(normalize_shape(base_shape))
    for _ in range(4):
        key = shape_to_tuple(current)
        if key not in seen:
            seen.add(key)
            orientations.append(current)
        current = rotate_shape(current)

    return orientations


class PieceDefinition:
    """A Blokus piece with all orientations pre-computed."""

    def __init__(self, piece_id: str, base_shape: PieceShape):
        self.id = piece_id
        self.size = len(base_shape)
        self.orientations = generate_orientations(base_shape)
        self.num_orientations = len(self.orientations)

        # Pre-compute numpy arrays for each orientation for fast operations
        self._numpy_orientations = [
            np.array(orient, dtype=np.int8) for orient in self.orientations
        ]

    def get_cells(self, orientation_idx: int, row: int, col: int) -> np.ndarray:
        """
        Get absolute board coordinates for piece at given position.
        Returns array of shape (size, 2) with [row, col] coordinates.
        """
        base = self._numpy_orientations[orientation_idx]
        offset = np.array([[row, col]], dtype=np.int8)
        return base + offset

    def __repr__(self):
        return f"PieceDefinition({self.id}, size={self.size}, orientations={self.num_orientations})"


# Build the piece catalog
PIECE_CATALOG: List[PieceDefinition] = [
    PieceDefinition(piece_id, shape) for piece_id, shape in BASE_PIECES
]

# Maps for quick lookup
PIECE_BY_ID = {p.id: p for p in PIECE_CATALOG}
PIECE_BY_INDEX = {i: p for i, p in enumerate(PIECE_CATALOG)}
PIECE_INDEX_BY_ID = {p.id: i for i, p in enumerate(PIECE_CATALOG)}

# All piece IDs in order
ALL_PIECE_IDS = [p.id for p in PIECE_CATALOG]
NUM_PIECES = len(PIECE_CATALOG)

# Total unique orientations across all pieces
TOTAL_ORIENTATIONS = sum(p.num_orientations for p in PIECE_CATALOG)


def get_piece(piece_id: str) -> PieceDefinition:
    """Get piece by ID."""
    return PIECE_BY_ID[piece_id]


def get_piece_by_index(idx: int) -> PieceDefinition:
    """Get piece by index (0-20)."""
    return PIECE_BY_INDEX[idx]

"""
Handcrafted heuristics for bootstrapping AlphaZero policy learning.

These heuristics provide an initial policy bias to help MCTS generate
focused targets during early training, addressing the "cold start" problem
where a random network produces uniform MCTS visits.

The heuristics are blended with the neural network output using a decay
schedule: high heuristic weight early in training, fading to pure NN later.
"""

import numpy as np
from typing import List, Tuple, Optional
from dataclasses import dataclass

from .game import BlokusGame, Move, BOARD_SIZE
from .pieces import PIECE_CATALOG, PIECE_BY_INDEX


# Board center coordinates
CENTER_ROW = (BOARD_SIZE - 1) / 2.0  # 9.5 for 20x20
CENTER_COL = (BOARD_SIZE - 1) / 2.0


@dataclass
class HeuristicConfig:
    """Configuration for heuristic policy bootstrapping."""
    # Blend factor: 0.0 = pure NN, 1.0 = pure heuristic
    # This should decay over training generations
    blend_factor: float = 0.5

    # Feature weights (higher = more important)
    piece_size_weight: float = 1.0      # Prefer larger pieces early
    center_weight: float = 0.5          # Prefer center positions
    corner_creation_weight: float = 0.3  # Prefer moves that create corners
    mobility_weight: float = 0.2        # Prefer moves that maintain flexibility

    # Game phase thresholds
    early_game_turns: int = 20          # Before this, piece size matters most
    mid_game_turns: int = 50            # After this, flexibility matters more


def compute_move_heuristic_score(
    move: Move,
    game: BlokusGame,
    config: HeuristicConfig,
) -> float:
    """
    Compute a heuristic score for a single move.

    Higher score = better move according to handcrafted rules.

    Returns:
        Score >= 0 (not normalized)
    """
    score = 1.0  # Base score for any legal move

    piece = PIECE_BY_INDEX[move.piece_idx]
    turn = game.turn_number

    # 1. PIECE SIZE BONUS (STRONG EFFECT)
    # Early game: STRONGLY prefer larger pieces (5-cells)
    # Rationale: Large pieces are harder to place later when board fills up
    # This is THE most important heuristic for Blokus
    if turn <= config.early_game_turns:
        # Scale: size 1 -> 0.0 bonus, size 5 -> 4.0 bonus
        # Make this exponential to really prefer 5s over 4s over 3s
        size_bonus = (piece.size ** 2) / 5.0  # 1->0.2, 2->0.8, 3->1.8, 4->3.2, 5->5.0
        score += config.piece_size_weight * size_bonus
    elif turn <= config.mid_game_turns:
        # Mid game: moderate piece size preference
        size_bonus = (piece.size ** 1.5) / 5.0
        score += config.piece_size_weight * size_bonus * 0.5

    # 2. CENTER POSITION BONUS
    # Prefer moves that place pieces closer to center
    # Rationale: Central control gives more options for future placements
    piece_cells = piece.get_cells(move.orientation_idx, move.row, move.col)
    avg_row = piece_cells[:, 0].mean()
    avg_col = piece_cells[:, 1].mean()

    # Distance from center (0 = center, ~14 = corner)
    center_dist = np.sqrt((avg_row - CENTER_ROW) ** 2 + (avg_col - CENTER_COL) ** 2)
    max_dist = np.sqrt(2) * CENTER_ROW  # ~13.4

    # Closer to center = higher bonus (0 to 1.5)
    # Use squared distance for stronger center preference
    center_bonus = (1.0 - (center_dist / max_dist)) ** 2 * 1.5
    score += config.center_weight * center_bonus

    # 3. CORNER CREATION BONUS
    # Prefer moves that create more attachment points for future pieces
    # Rationale: More corners = more options = more flexibility
    corners_created = 0
    player_color = move.player + 1
    cell_set = set(map(tuple, piece_cells))

    for r, c in piece_cells:
        for dr, dc in [(-1, -1), (-1, 1), (1, -1), (1, 1)]:  # Diagonals
            nr, nc = r + dr, c + dc
            if 0 <= nr < BOARD_SIZE and 0 <= nc < BOARD_SIZE:
                # Check if this could be a valid corner
                if game.board[nr, nc] == 0:  # Empty
                    # Check not edge-adjacent to own color
                    is_valid = True
                    for edr, edc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
                        er, ec = nr + edr, nc + edc
                        if 0 <= er < BOARD_SIZE and 0 <= ec < BOARD_SIZE:
                            if game.board[er, ec] == player_color:
                                is_valid = False
                                break
                        # Also check if edge-adjacent to piece being placed
                        if (er, ec) in cell_set:
                            is_valid = False
                            break
                    if is_valid:
                        corners_created += 1

    # Normalize corners (typically 0-8 possible)
    corner_bonus = min(corners_created / 4.0, 2.0)  # Cap at 2.0
    score += config.corner_creation_weight * corner_bonus

    # 4. EDGE AVOIDANCE (more penalty)
    # Penalize moves at board edges - getting trapped is bad
    edge_cells = 0
    for r, c in piece_cells:
        if r <= 1 or r >= BOARD_SIZE - 2 or c <= 1 or c >= BOARD_SIZE - 2:
            edge_cells += 1
    edge_penalty = edge_cells / piece.size * 0.5
    score -= edge_penalty

    # 5. BLOCKING POTENTIAL (late game)
    # In mid/late game, prefer moves that interfere with opponent corners
    if turn > config.early_game_turns:
        blocking_bonus = 0
        for r, c in piece_cells:
            # Check if we're adjacent to opponent pieces (blocking their expansion)
            for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
                nr, nc = r + dr, c + dc
                if 0 <= nr < BOARD_SIZE and 0 <= nc < BOARD_SIZE:
                    cell_val = game.board[nr, nc]
                    if cell_val != 0 and cell_val != player_color:
                        blocking_bonus += 0.1
        score += blocking_bonus

    return max(score, 0.01)  # Ensure positive score


def compute_heuristic_policy(
    game: BlokusGame,
    legal_moves: List[Move],
    config: Optional[HeuristicConfig] = None,
) -> np.ndarray:
    """
    Compute a heuristic policy distribution over legal moves.

    Args:
        game: Current game state
        legal_moves: List of legal moves
        config: Heuristic configuration

    Returns:
        Probability distribution over moves (sums to 1)
    """
    if not legal_moves:
        return np.array([])

    if config is None:
        config = HeuristicConfig()

    # Compute raw scores
    scores = np.array([
        compute_move_heuristic_score(move, game, config)
        for move in legal_moves
    ])

    # Apply softmax with temperature to convert to probabilities
    # Lower temperature = more peaked distribution
    temperature = 0.15  # Very low temp makes good moves MUCH more prominent
    scores = scores / temperature
    scores = scores - scores.max()  # Numerical stability
    exp_scores = np.exp(scores)
    policy = exp_scores / exp_scores.sum()

    return policy


def blend_policies(
    nn_policy: np.ndarray,
    heuristic_policy: np.ndarray,
    blend_factor: float,
) -> np.ndarray:
    """
    Blend neural network policy with heuristic policy.

    Args:
        nn_policy: Policy from neural network (normalized)
        heuristic_policy: Policy from heuristics (normalized)
        blend_factor: 0.0 = pure NN, 1.0 = pure heuristic

    Returns:
        Blended policy (normalized)
    """
    if len(nn_policy) != len(heuristic_policy):
        raise ValueError(
            f"Policy lengths don't match: NN={len(nn_policy)}, heur={len(heuristic_policy)}"
        )

    if len(nn_policy) == 0:
        return nn_policy

    # Linear blend
    blended = (1 - blend_factor) * nn_policy + blend_factor * heuristic_policy

    # Renormalize (should already sum to 1, but ensure it)
    total = blended.sum()
    if total > 0:
        blended = blended / total
    else:
        # Fallback to uniform
        blended = np.ones_like(blended) / len(blended)

    return blended


def get_blend_factor_for_generation(
    generation: int,
    initial_blend: float = 0.5,
    decay_generations: int = 30,
    min_blend: float = 0.0,
) -> float:
    """
    Compute blend factor that decays over training generations.

    Starts with initial_blend, decays linearly to min_blend over
    decay_generations, then stays at min_blend.

    Args:
        generation: Current training generation (1-indexed)
        initial_blend: Starting blend factor (high = more heuristic)
        decay_generations: Number of generations to decay over
        min_blend: Minimum blend factor (0 = pure NN after decay)

    Returns:
        Blend factor for current generation
    """
    if generation >= decay_generations:
        return min_blend

    # Linear decay
    progress = generation / decay_generations
    blend = initial_blend * (1 - progress) + min_blend * progress

    return blend

"""
Heuristic pretraining for AlphaZero cold start problem.

Pretrains the policy head on heuristic-scored moves before self-play begins.
This helps MCTS generate focused targets during early training instead of
uniform visit counts from a random network.

Phase 0 (before self-play):
1. Generate diverse random game positions
2. Score all legal moves with heuristics for each position
3. Train policy head (freeze backbone) for N steps
4. Unfreeze all, continue to normal self-play
"""

import random
from typing import List, Dict, Optional, Callable

import numpy as np
import torch
import torch.nn.functional as F
import torch.optim as optim

from app.logging_config import log
from .game import BlokusGame
from .network import BlokusNet
from .heuristics import HeuristicConfig, compute_heuristic_policy


def generate_pretraining_positions(
    num_positions: int = 10000,
    heuristic_config: Optional[HeuristicConfig] = None,
) -> List[Dict]:
    """
    Generate diverse positions by playing random games and sampling states.

    Args:
        num_positions: Target number of positions to generate
        heuristic_config: Configuration for heuristic scoring

    Returns:
        List of position dictionaries with:
        - board_tensor: numpy array for neural network input
        - pieces_vector: numpy array for neural network input
        - legal_moves: List[Move] for this position
        - target_policy: numpy array of heuristic policy (normalized)
    """
    if heuristic_config is None:
        heuristic_config = HeuristicConfig()

    positions = []

    # Estimate games needed (~50 positions per game on average)
    games_needed = (num_positions // 50) + 10  # Add buffer

    log.info(f"Generating {num_positions} pretraining positions from ~{games_needed} random games")

    for game_idx in range(games_needed):
        if len(positions) >= num_positions:
            break

        game = BlokusGame()
        depth = 0

        while not game.is_game_over and depth < 80:
            legal_moves = game.get_legal_moves()
            if not legal_moves:
                game.pass_turn()
                depth += 1
                continue

            # Sample position every 4-5 moves (always in early game for more diversity)
            should_sample = (depth % 5 == 0) or (depth < 15 and depth % 3 == 0)

            if should_sample and len(positions) < num_positions:
                # Get state tensors
                board_tensor, pieces_vector = game.get_state_for_neural_net()

                # Compute heuristic policy for this position
                heuristic_policy = compute_heuristic_policy(
                    game, legal_moves, heuristic_config
                )

                positions.append({
                    'board_tensor': board_tensor,
                    'pieces_vector': pieces_vector,
                    'legal_moves': legal_moves,
                    'target_policy': heuristic_policy,
                })

            # Random move to continue the game
            move = random.choice(legal_moves)
            game.make_move(move)
            depth += 1

        if (game_idx + 1) % 100 == 0:
            log.info(f"Generated {len(positions)}/{num_positions} positions from {game_idx + 1} games")

    log.info(f"Generated {len(positions)} pretraining positions")
    return positions[:num_positions]


def compute_policy_cross_entropy(
    policy_logits: torch.Tensor,
    batch: List[Dict],
    device: str,
) -> torch.Tensor:
    """
    Compute cross-entropy loss between predicted logits and heuristic targets.

    Args:
        policy_logits: Tensor of shape (batch_size, max_moves) from network
        batch: List of position dictionaries with target_policy
        device: Device for tensors

    Returns:
        Scalar loss tensor
    """
    max_len = max(len(b['target_policy']) for b in batch)

    # Prepare target and mask arrays
    target_np = np.zeros((len(batch), max_len), dtype=np.float32)
    mask_np = np.zeros((len(batch), max_len), dtype=np.float32)

    for i, b in enumerate(batch):
        plen = len(b['target_policy'])
        target_np[i, :plen] = b['target_policy']
        mask_np[i, :plen] = 1.0

    targets = torch.from_numpy(target_np).to(device)
    mask = torch.from_numpy(mask_np).to(device)

    # Mask invalid positions with -inf before softmax
    masked_logits = policy_logits[:, :max_len].clone()
    masked_logits[mask == 0] = float('-inf')

    # Compute log probabilities
    log_probs = F.log_softmax(masked_logits, dim=1)
    # Replace NaN with 0 (can happen if all logits are -inf)
    log_probs = torch.nan_to_num(log_probs, nan=0.0)

    # Cross-entropy: -sum(target * log_probs) per sample, then mean
    loss = -(targets * log_probs).sum(dim=1).mean()

    return loss


def pretrain_policy_head(
    network: BlokusNet,
    device: str,
    num_positions: int = 10000,
    num_steps: int = 1000,
    batch_size: int = 256,
    learning_rate: float = 0.001,
    progress_callback: Optional[Callable[[int, int, float], None]] = None,
) -> Dict[str, float]:
    """
    Pretrain policy head on heuristic-scored positions.

    Freezes backbone (stem, res_tower, value_head), trains only:
    - move_encoder
    - policy_query

    Args:
        network: The BlokusNet model
        device: Device string ('cuda' or 'cpu')
        num_positions: Number of positions to generate for training
        num_steps: Number of training steps
        batch_size: Batch size for training
        learning_rate: Learning rate for policy head
        progress_callback: Optional callback(step, total_steps, loss)

    Returns:
        Dictionary with training statistics
    """
    log.info("=" * 60)
    log.info("PHASE 0: Heuristic Pretraining")
    log.info("=" * 60)
    log.info(f"  Positions: {num_positions}")
    log.info(f"  Steps: {num_steps}")
    log.info(f"  Batch size: {batch_size}")
    log.info(f"  Learning rate: {learning_rate}")

    # Generate training data
    log.info("Generating training positions...")
    training_data = generate_pretraining_positions(num_positions)

    if len(training_data) < batch_size:
        log.warning(f"Not enough training data ({len(training_data)} < {batch_size}), skipping pretraining")
        return {'avg_loss': 0.0, 'skipped': True}

    # Freeze backbone layers, train only policy head
    frozen_count = 0
    trainable_count = 0

    for name, param in network.named_parameters():
        # Keep move_encoder and policy_query trainable
        if 'move_encoder' in name or 'policy_query' in name:
            param.requires_grad_(True)
            trainable_count += 1
        else:
            param.requires_grad_(False)
            frozen_count += 1

    log.info(f"Frozen parameters: {frozen_count}, Trainable: {trainable_count}")

    # Optimizer for policy head only
    policy_params = [p for p in network.parameters() if p.requires_grad]
    optimizer = optim.Adam(policy_params, lr=learning_rate)

    network.train()
    total_loss = 0.0
    loss_history = []

    for step in range(num_steps):
        # Sample batch
        batch = random.sample(training_data, min(batch_size, len(training_data)))

        # Prepare batch tensors
        board_np = np.stack([b['board_tensor'] for b in batch])
        pieces_np = np.stack([b['pieces_vector'] for b in batch])
        legal_moves_list = [b['legal_moves'] for b in batch]

        board_tensors = torch.from_numpy(board_np).to(device)
        pieces_vectors = torch.from_numpy(pieces_np).to(device)

        # Forward pass
        policy_logits, _ = network(board_tensors, pieces_vectors, legal_moves_list)

        # Compute loss
        loss = compute_policy_cross_entropy(policy_logits, batch, device)

        # Backward pass
        optimizer.zero_grad()
        loss.backward()
        torch.nn.utils.clip_grad_norm_(policy_params, 1.0)
        optimizer.step()

        loss_value = loss.item()
        total_loss += loss_value
        loss_history.append(loss_value)

        # Progress callback
        if progress_callback and (step + 1) % 10 == 0:
            progress_callback(step + 1, num_steps, loss_value)

        # Log progress
        if (step + 1) % 100 == 0:
            avg_recent = sum(loss_history[-100:]) / len(loss_history[-100:])
            log.info(f"Pretrain step {step + 1}/{num_steps} - loss: {loss_value:.4f} (avg: {avg_recent:.4f})")

    # Unfreeze all parameters for main training
    for param in network.parameters():
        param.requires_grad_(True)

    avg_loss = total_loss / num_steps if num_steps > 0 else 0.0

    log.info("=" * 60)
    log.info(f"Pretraining complete! Avg loss: {avg_loss:.4f}")
    log.info("=" * 60)

    return {
        'avg_loss': avg_loss,
        'final_loss': loss_history[-1] if loss_history else 0.0,
        'num_positions': len(training_data),
        'num_steps': num_steps,
    }

async def get_move(request: AlphaZeroMoveRequest):
    """Get AI move for a game state using a trained model."""
    from .game import BlokusGame, Move, PLAYER_NAMES
    from .network import BlokusNet, create_network
    from .mcts import MCTS, MCTSConfig

    # Load model
    model_path = Path(request.model_path)
    if not model_path.exists():
        raise HTTPException(status_code=404, detail=f"Model not found: {model_path}")

    device = 'cuda' if torch.cuda.is_available() else 'cpu'

    # Load checkpoint
    checkpoint = torch.load(model_path, map_location=device)

    # Get config from checkpoint or use defaults
    config = checkpoint.get('config', {})
    num_res_blocks = config.get('num_res_blocks', 15)
    channels = config.get('channels', 256)

    # Detect input channels from checkpoint
    input_channels = checkpoint['model_state_dict']['stem.0.weight'].shape[1]

    # Create and load network
    network = create_network(num_res_blocks=num_res_blocks, channels=channels, input_channels=input_channels, device=device)
    network.load_state_dict(checkpoint['model_state_dict'])
    network.eval()

    # Reconstruct game state
    game = _dict_to_game(request.game_state)

    legal_moves = game.get_legal_moves()

    mcts_config = MCTSConfig(
        num_simulations=request.mcts_simulations,
        temperature_threshold=0,  # Always use low temperature for inference

    mcts = MCTS(network, mcts_config, device)

    move, policy, root = mcts.select_move(game, add_noise=False, temperature=0.1)

    # Get value from root
    value = root.q_value if root.visit_count > 0 else 0.0

    return MoveResponse(
        piece_idx=move.piece_idx,
        orientation_idx=move.orientation_idx,
        row=move.row,
        col=move.col,
        policy=policy.tolist(),
        value=value,
    )


def _dict_to_game(state: Dict[str, Any]) -> 'BlokusGame':
    """Convert a game state dictionary to a BlokusGame object."""
    from .game import BlokusGame, PlayerState, PLAYER_NAMES
    from .pieces import PIECE_INDEX_BY_ID
    import numpy as np

    game = BlokusGame()

    # Set board
    board = state.get('board', [])
    for r, row in enumerate(board):
        for c, cell in enumerate(row):
            if cell is not None:
                # Map color name to player index
                player_idx = PLAYER_NAMES.index(cell) if cell in PLAYER_NAMES else 0
                game.board[r, c] = player_idx + 1

    # Set player states
    players = state.get('players', {})
    for color, pstate in players.items():
        if color in PLAYER_NAMES:
            player_idx = PLAYER_NAMES.index(color)

            # Convert remaining pieces from IDs to indices
            remaining = pstate.get('remainingPieces', [])
            game.players[player_idx].remaining_pieces = {
                PIECE_INDEX_BY_ID.get(pid, 0) for pid in remaining
            }
            game.players[player_idx].has_played = pstate.get('hasPlayed', False)
            game.players[player_idx].is_eliminated = pstate.get('isEliminated', False)
            game.players[player_idx].score = pstate.get('score', 0)

    # Set current player
    current = state.get('currentPlayer', 'blue')
    if current in PLAYER_NAMES:
        game.current_player = PLAYER_NAMES.index(current)

    # Set other state
    game.turn_number = state.get('turnNumber', 1)
    game.is_game_over = state.get('isGameOver', False)
    game.consecutive_passes = state.get('consecutivePasses', 0)

    # Rebuild valid corners (simplified - just compute from scratch)
    game._valid_corners = [set() for _ in range(4)]
    for player_idx in range(4):
        if not game.players[player_idx].has_played:
            from .game import STARTING_CORNERS
            game._valid_corners[player_idx].add(STARTING_CORNERS[player_idx])
        else:
            # Scan board for diagonal attachment points
            player_color = player_idx + 1
            for r in range(20):
                for c in range(20):
                    if game.board[r, c] != 0:
                        continue
                    # Check if diagonally adjacent to own color
                    for dr, dc in [(-1, -1), (-1, 1), (1, -1), (1, 1)]:
                        nr, nc = r + dr, c + dc
                        if 0 <= nr < 20 and 0 <= nc < 20 and game.board[nr, nc] == player_color:
                            # Check not edge adjacent
                            is_edge = False
                            for edr, edc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
                                er, ec = r + edr, c + edc
                                if 0 <= er < 20 and 0 <= ec < 20 and game.board[er, ec] == player_color:
                                    is_edge = True
                                    break
                            if not is_edge:
                                game._valid_corners[player_idx].add((r, c))
                                break

    return game

async def export_to_onnx(run_id: str, request: OnnxExportRequest = None):
    """
    Export AlphaZero model to ONNX format.

    Args:
        run_id: Training run ID
        request: Optional request body with checkpoint_id

    Returns:
        Export metadata including path, size, and tensor shapes
    """
    from .network import BlokusNet, create_network, export_to_onnx as onnx_export

    # Determine checkpoint path
    model_dir = MODELS_DIR / "alphazero" / run_id
    if not model_dir.exists():
        raise HTTPException(status_code=404, detail=f"Model directory not found: {model_dir}")

    checkpoint_id = request.checkpoint_id if request else None
    if checkpoint_id:
        # Use specific checkpoint
        model_path = model_dir / f"{checkpoint_id}.pt"
    else:
        # Use best.pt by default, fall back to latest checkpoint
        model_path = model_dir / "best.pt"
        if not model_path.exists():
            # Find the latest checkpoint file
            checkpoint_files = sorted(model_dir.glob("*.pt"), key=lambda p: p.stat().st_mtime, reverse=True)
            if checkpoint_files:
                model_path = checkpoint_files[0]
                log.info(f"No best.pt found, using latest checkpoint: {model_path.name}")

    if not model_path.exists():
        raise HTTPException(status_code=404, detail=f"No checkpoint found in {model_dir}")

    log.info(f"Exporting model to ONNX: {model_path}")

    device = 'cuda' if torch.cuda.is_available() else 'cpu'

    # Load checkpoint
    checkpoint = torch.load(model_path, map_location=device)

    # Get config from checkpoint
    config = checkpoint.get('config', {})
    num_res_blocks = config.get('num_res_blocks', 15)
    channels = config.get('channels', 256)

    # Detect input channels from the checkpoint's stem layer
    state_dict = checkpoint['model_state_dict']
    stem_weight_shape = state_dict['stem.0.weight'].shape
    input_channels = stem_weight_shape[1]  # Shape is [out_channels, in_channels, H, W]
    actual_channels = stem_weight_shape[0]
    log.info(f"Detected model: {input_channels} input channels, {actual_channels} channels, {num_res_blocks} res blocks")

    # Create network with correct architecture
    network = BlokusNet(
        num_res_blocks=num_res_blocks,
        channels=actual_channels,
    )
    # Override INPUT_CHANNELS if different from default
    if input_channels != BlokusNet.INPUT_CHANNELS:
        # Recreate stem with correct input channels
        import torch.nn as nn
        network.stem = nn.Sequential(
            nn.Conv2d(input_channels, actual_channels, 3, padding=1, bias=False),
            nn.BatchNorm2d(actual_channels),
            nn.ReLU(),
        )
    network = network.to(device)
    network.load_state_dict(state_dict)
    network.eval()

    # Export to ONNX
    onnx_path = model_dir / "model.onnx"
    export_metadata = onnx_export(network, str(onnx_path), input_channels=input_channels)

    log.info(f"ONNX export complete: {onnx_path} ({export_metadata['file_size_mb']} MB)")

    return OnnxExportResponse(
        run_id=run_id,
        onnx_path=str(onnx_path),
        file_size_mb=export_metadata['file_size_mb'],
        input_shapes=export_metadata['input_shapes'],
        output_shapes=export_metadata['output_shapes'],
        output_names=export_metadata['output_names'],
    )

use pyo3::prelude::*;

mod node;
mod tree;
mod game_state;

pub use node::MCTSNode;
pub use tree::MCTSTree;
pub use game_state::GameState;

/// Rust-accelerated MCTS for AlphaZero
#[pymodule]
fn rust_mcts(m: &Bound<'_, PyModule>) -> PyResult<()> {
    m.add_class::<MCTSTree>()?;
    m.add_class::<GameState>()?;
    m.add_function(wrap_pyfunction!(batch_select_leaves, m)?)?;
    m.add_function(wrap_pyfunction!(batch_expand_and_backprop, m)?)?;
    m.add_function(wrap_pyfunction!(benchmark_tree_ops, m)?)?;
    Ok(())
}

/// Select multiple leaves from multiple trees in parallel
#[pyfunction]
fn batch_select_leaves(
    trees: Vec<PyRef<MCTSTree>>,
    c_puct: f32,
    batch_size: usize,
) -> PyResult<Vec<(usize, Vec<u32>, Vec<f32>)>> {
    // Returns: [(tree_idx, path_indices, move_priors), ...]
    let mut results = Vec::with_capacity(trees.len() * batch_size);

    for (tree_idx, tree) in trees.iter().enumerate() {
        for _ in 0..batch_size {
            if let Some((path, priors)) = tree.select_leaf(c_puct) {
                results.push((tree_idx, path, priors));
            }
        }
    }

    Ok(results)
}

/// Expand nodes and backpropagate values in batch
#[pyfunction]
fn batch_expand_and_backprop(
    trees: &Bound<'_, pyo3::types::PyList>,
    expansions: Vec<(usize, Vec<u32>, Vec<(u32, f32)>, f32)>,
) -> PyResult<()> {
    // expansions: [(tree_idx, path, [(move_id, prior), ...], value), ...]
    for (tree_idx, path, children_priors, value) in expansions {
        let tree: PyRef<MCTSTree> = trees.get_item(tree_idx)?.extract()?;
        tree.expand_and_backprop_internal(&path, &children_priors, value);
    }
    Ok(())
}

/// Benchmark pure tree operations (no Python overhead in the loop)
#[pyfunction]
fn benchmark_tree_ops(num_simulations: usize, num_children: usize, c_puct: f32) -> PyResult<(f64, f64, f64)> {
    use std::time::Instant;

    // Create a tree
    let tree = MCTSTree::new(0);

    // Benchmark select
    let start = Instant::now();
    for _ in 0..num_simulations {
        let _ = tree.select_leaf(c_puct);
    }
    let select_time = start.elapsed().as_secs_f64();

    // Benchmark expand
    let children: Vec<(u32, f32)> = (0..num_children as u32)
        .map(|i| (i, 1.0 / num_children as f32))
        .collect();

    let start = Instant::now();
    for _i in 0..num_simulations {
        let path: Vec<u32> = vec![0];
        tree.expand_and_backprop_internal(&path, &children, 0.5);
    }
    let expand_time = start.elapsed().as_secs_f64();

    // Benchmark full cycle
    let tree2 = MCTSTree::new(0);
    let start = Instant::now();
    for i in 0..num_simulations {
        let _leaf = tree2.select_leaf(c_puct);
        let path: Vec<u32> = vec![0];
        tree2.expand_and_backprop_internal(&path, &children, if i % 2 == 0 { 1.0 } else { -1.0 });
    }
    let full_time = start.elapsed().as_secs_f64();

    Ok((select_time, expand_time, full_time))
}

use crate::node::MCTSNode;
use parking_lot::RwLock;
use pyo3::prelude::*;
use smallvec::SmallVec;
use std::sync::atomic::{AtomicU32, Ordering};

/// MCTS Tree with fast Rust operations
#[pyclass]
pub struct MCTSTree {
    /// All nodes in the tree (arena allocation)
    nodes: RwLock<Vec<MCTSNode>>,

    /// Next node ID
    next_id: AtomicU32,

    /// Root node index (always 0)
    root: u32,

    /// Current player at root
    root_player: u32,
}

// Internal methods (not exposed to Python)
impl MCTSTree {
    /// Backpropagate value up the tree
    fn backpropagate_internal(&self, path: &[u32], value: f32) {
        let nodes = self.nodes.read();
        let mut current_value = value;

        // Backprop from leaf to root, alternating sign for opponent
        for (i, &node_idx) in path.iter().rev().enumerate() {
            let node = &nodes[node_idx as usize];

            // Remove virtual loss if applied during selection
            if i > 0 {
                node.remove_virtual_loss();
            }

            // Add visit and value
            node.backprop(current_value);

            // Flip value for opponent's perspective
            current_value = -current_value;
        }
    }

    /// Internal expand and backprop (takes references)
    pub fn expand_and_backprop_internal(&self, path: &[u32], children_priors: &[(u32, f32)], value: f32) {
        if path.is_empty() {
            return;
        }

        let leaf_idx = *path.last().unwrap();

        // Create child nodes if priors provided
        if !children_priors.is_empty() {
            let nodes = self.nodes.write();
            let leaf = &nodes[leaf_idx as usize];

            // Store priors for later expansion
            leaf.set_child_priors(children_priors);

            // Mark as expanded with priors stored
            if !leaf.is_expanded() {
                leaf.expand(&[]);
            }
        }

        // Backpropagate
        self.backpropagate_internal(path, value);
    }
}

#[pymethods]
impl MCTSTree {
    #[new]
    pub fn new(root_player: u32) -> Self {
        let root_node = MCTSNode::new(0, 0, 1.0, u32::MAX);
        Self {
            nodes: RwLock::new(vec![root_node]),
            next_id: AtomicU32::new(1),
            root: 0,
            root_player,
        }
    }

    /// Get number of nodes in tree
    pub fn size(&self) -> usize {
        self.nodes.read().len()
    }

    /// Get root visit count
    pub fn root_visits(&self) -> u32 {
        self.nodes.read()[0].visit_count()
    }

    /// Set root policy (child priors)
    pub fn set_root_policy(&self, moves_and_priors: Vec<(u32, f32)>) {
        let nodes = self.nodes.read();
        nodes[0].set_child_priors(&moves_and_priors);
    }

    /// Select a leaf node, returning the path and unexpanded move priors if any
    /// Returns: (path_indices, child_priors_at_leaf)
    pub fn select_leaf(&self, c_puct: f32) -> Option<(Vec<u32>, Vec<f32>)> {
        let nodes = self.nodes.read();
        let mut path = Vec::with_capacity(32);
        let mut current_idx = self.root;

        loop {
            path.push(current_idx);
            let node = &nodes[current_idx as usize];

            // If not expanded, return with priors for expansion
            if !node.is_expanded() {
                let priors: Vec<f32> = node.get_child_priors()
                    .iter()
                    .map(|(_, p)| *p)
                    .collect();
                return Some((path, priors));
            }

            let children = node.get_children();
            if children.is_empty() {
                // Terminal node
                return Some((path, vec![]));
            }

            // Get parent visits for UCB calculation
            let parent_visits = node.visit_count();

            // Find best child considering both expanded and unexpanded
            let mut best_score = f32::NEG_INFINITY;
            let mut best_child_idx: Option<u32> = None;
            let mut best_unexpanded: Option<(u32, f32)> = None;

            // Check expanded children
            for &(_move_id, child_idx) in children.iter() {
                let child = &nodes[child_idx as usize];
                let score = child.ucb_score(parent_visits, c_puct);
                if score > best_score {
                    best_score = score;
                    best_child_idx = Some(child_idx);
                    best_unexpanded = None;
                }
            }

            // Check unexpanded children (if any priors remain)
            let child_priors = node.get_child_priors();
            let expanded_moves: SmallVec<[u32; 32]> = children.iter().map(|(m, _)| *m).collect();

            for &(move_id, prior) in child_priors.iter() {
                if !expanded_moves.contains(&move_id) {
                    let score = MCTSNode::unexpanded_ucb(prior, parent_visits, c_puct);
                    if score > best_score {
                        best_score = score;
                        best_child_idx = None;
                        best_unexpanded = Some((move_id, prior));
                    }
                }
            }

            match (best_child_idx, best_unexpanded) {
                (Some(idx), _) => {
                    // Apply virtual loss and continue
                    nodes[idx as usize].apply_virtual_loss();
                    current_idx = idx;
                }
                (None, Some((_move_id, _prior))) => {
                    // Need to expand this child - return path for expansion
                    let priors: Vec<f32> = child_priors
                        .iter()
                        .filter(|(m, _)| !expanded_moves.contains(m))
                        .map(|(_, p)| *p)
                        .collect();
                    return Some((path, priors));
                }
                (None, None) => {
                    // No children at all - terminal
                    return Some((path, vec![]));
                }
            }
        }
    }

    /// Expand a leaf node and backpropagate value (Python interface)
    pub fn expand_and_backprop(&self, path: Vec<u32>, children_priors: Vec<(u32, f32)>, value: f32) {
        self.expand_and_backprop_internal(&path, &children_priors, value);
    }

    /// Add a child node to the tree
    #[pyo3(signature = (parent_idx, move_id, prior))]
    pub fn add_child(&self, parent_idx: u32, move_id: u32, prior: f32) -> u32 {
        let child_id = self.next_id.fetch_add(1, Ordering::Relaxed);
        let child_node = MCTSNode::new(child_id, move_id, prior, parent_idx);

        let mut nodes = self.nodes.write();
        let child_idx = nodes.len() as u32;
        nodes.push(child_node);

        // Add to parent's children
        let parent = &nodes[parent_idx as usize];
        let mut children = parent.get_children();
        children.push((move_id, child_idx));
        parent.expand(&children);

        child_idx
    }

    /// Get action probabilities based on visit counts
    #[pyo3(signature = (temperature = 1.0))]
    pub fn get_action_probs(&self, temperature: f32) -> Vec<(u32, f32)> {
        let nodes = self.nodes.read();
        let root = &nodes[0];

        if !root.is_expanded() {
            return vec![];
        }

        let children = root.get_children();
        if children.is_empty() {
            // Return priors if no children expanded yet
            return root.get_child_priors().into_iter().collect();
        }

        let mut moves_visits: Vec<(u32, f32)> = children
            .iter()
            .map(|&(move_id, child_idx)| {
                let visits = nodes[child_idx as usize].visit_count() as f32;
                (move_id, visits)
            })
            .collect();

        if moves_visits.is_empty() {
            return vec![];
        }

        // Apply temperature
        if temperature < 0.01 {
            // Greedy - pick best
            let max_visits = moves_visits.iter().map(|(_, v)| *v).fold(0.0f32, f32::max);
            for (_, v) in moves_visits.iter_mut() {
                *v = if (*v - max_visits).abs() < 0.001 { 1.0 } else { 0.0 };
            }
        } else {
            // Apply temperature
            for (_, v) in moves_visits.iter_mut() {
                *v = v.powf(1.0 / temperature);
            }
        }

        // Normalize
        let sum: f32 = moves_visits.iter().map(|(_, v)| *v).sum();
        if sum > 0.0 {
            for (_, v) in moves_visits.iter_mut() {
                *v /= sum;
            }
        }

        moves_visits
    }

    /// Get the best move (most visited)
    pub fn best_move(&self) -> Option<u32> {
        let probs = self.get_action_probs(0.0);
        probs.into_iter().max_by(|a, b| a.1.partial_cmp(&b.1).unwrap()).map(|(m, _)| m)
    }

    /// Get Q-value for root children
    pub fn get_root_q_values(&self) -> Vec<(u32, f32)> {
        let nodes = self.nodes.read();
        let root = &nodes[0];
        let children = root.get_children();

        children
            .iter()
            .map(|&(move_id, child_idx)| {
                let q = nodes[child_idx as usize].q_value();
                (move_id, q)
            })
            .collect()
    }
}

use parking_lot::RwLock;
use smallvec::SmallVec;
use std::sync::atomic::{AtomicU32, AtomicI32, Ordering};

/// A node in the MCTS tree, optimized for concurrent access
pub struct MCTSNode {
    /// Unique identifier for this node
    pub id: u32,

    /// Move that led to this node (piece_idx << 24 | orientation << 16 | row << 8 | col)
    pub move_id: u32,

    /// Prior probability from neural network
    pub prior: f32,

    /// Visit count (atomic for thread safety)
    visit_count: AtomicU32,

    /// Value sum (stored as fixed-point for atomic ops: actual = value_sum / 1000000)
    value_sum: AtomicI32,

    /// Virtual loss count (for parallel MCTS)
    virtual_loss: AtomicU32,

    /// Parent node index (u32::MAX if root)
    pub parent: u32,

    /// Children: stored as (move_id, node_index) pairs
    /// Using RwLock for expansion, but reads are lock-free via atomic visit_count
    children: RwLock<SmallVec<[(u32, u32); 32]>>,

    /// Child priors (stored until expanded)
    child_priors: RwLock<SmallVec<[(u32, f32); 64]>>,

    /// Whether this node has been expanded
    is_expanded: std::sync::atomic::AtomicBool,
}

impl MCTSNode {
    pub fn new(id: u32, move_id: u32, prior: f32, parent: u32) -> Self {
        Self {
            id,
            move_id,
            prior,
            visit_count: AtomicU32::new(0),
            value_sum: AtomicI32::new(0),
            virtual_loss: AtomicU32::new(0),
            parent,
            children: RwLock::new(SmallVec::new()),
            child_priors: RwLock::new(SmallVec::new()),
            is_expanded: std::sync::atomic::AtomicBool::new(false),
        }
    }

    #[inline]
    pub fn visit_count(&self) -> u32 {
        self.visit_count.load(Ordering::Relaxed)
    }

    #[inline]
    pub fn value_sum(&self) -> f32 {
        self.value_sum.load(Ordering::Relaxed) as f32 / 1_000_000.0
    }

    #[inline]
    pub fn q_value(&self) -> f32 {
        let visits = self.visit_count();
        if visits == 0 {
            0.0
        } else {
            self.value_sum() / visits as f32
        }
    }

    #[inline]
    pub fn virtual_loss_count(&self) -> u32 {
        self.virtual_loss.load(Ordering::Relaxed)
    }

    /// Calculate UCB score for child selection
    #[inline]
    pub fn ucb_score(&self, parent_visits: u32, c_puct: f32) -> f32 {
        let visits = self.visit_count() + self.virtual_loss_count();
        let q = if visits == 0 {
            0.0
        } else {
            // Adjust Q for virtual loss
            let vl = self.virtual_loss_count() as f32;
            (self.value_sum() - vl * 3.0) / visits as f32
        };

        let exploration = c_puct * self.prior * (parent_visits as f32).sqrt() / (1.0 + visits as f32);
        q + exploration
    }

    /// Calculate UCB score for an unexpanded child (no node yet)
    #[inline]
    pub fn unexpanded_ucb(prior: f32, parent_visits: u32, c_puct: f32) -> f32 {
        c_puct * prior * (parent_visits as f32).sqrt()
    }

    /// Apply virtual loss (for parallel MCTS)
    #[inline]
    pub fn apply_virtual_loss(&self) {
        self.visit_count.fetch_add(1, Ordering::Relaxed);
        self.virtual_loss.fetch_add(1, Ordering::Relaxed);
    }

    /// Remove virtual loss
    #[inline]
    pub fn remove_virtual_loss(&self) {
        self.visit_count.fetch_sub(1, Ordering::Relaxed);
        self.virtual_loss.fetch_sub(1, Ordering::Relaxed);
    }

    /// Backpropagate a value
    #[inline]
    pub fn backprop(&self, value: f32) {
        self.visit_count.fetch_add(1, Ordering::Relaxed);
        // Convert to fixed-point and add atomically
        let value_fixed = (value * 1_000_000.0) as i32;
        self.value_sum.fetch_add(value_fixed, Ordering::Relaxed);
    }

    /// Check if expanded
    #[inline]
    pub fn is_expanded(&self) -> bool {
        self.is_expanded.load(Ordering::Acquire)
    }

    /// Set child priors (before expansion)
    pub fn set_child_priors(&self, priors: &[(u32, f32)]) {
        let mut cp = self.child_priors.write();
        cp.clear();
        cp.extend(priors.iter().copied());
    }

    /// Get child priors
    pub fn get_child_priors(&self) -> SmallVec<[(u32, f32); 64]> {
        self.child_priors.read().clone()
    }

    /// Mark as expanded and add children
    pub fn expand(&self, children_ids: &[(u32, u32)]) {
        let mut children = self.children.write();
        children.clear();
        children.extend(children_ids.iter().copied());
        self.is_expanded.store(true, Ordering::Release);
    }

    /// Get children (returns copy to avoid holding lock)
    pub fn get_children(&self) -> SmallVec<[(u32, u32); 32]> {
        self.children.read().clone()
    }

    /// Check if has children
    pub fn has_children(&self) -> bool {
        !self.children.read().is_empty()
    }
}

use pyo3::prelude::*;

/// Minimal game state for move encoding/decoding
#[pyclass]
#[derive(Clone)]
pub struct GameState {
    /// Current player (0-3)
    #[pyo3(get, set)]
    pub current_player: u8,

    /// Board state as flat array (20x20 = 400 cells)
    /// Values: 0=empty, 1-4=player pieces
    board: Vec<u8>,
}

#[pymethods]
impl GameState {
    #[new]
    pub fn new() -> Self {
        Self {
            current_player: 0,
            board: vec![0; 400],
        }
    }

    /// Create from Python board array
    #[staticmethod]
    pub fn from_board(board: Vec<u8>, current_player: u8) -> Self {
        Self {
            current_player,
            board,
        }
    }

    /// Encode a move as u32: piece_idx(8) | orientation(8) | row(8) | col(8)
    #[staticmethod]
    pub fn encode_move(piece_idx: u8, orientation: u8, row: u8, col: u8) -> u32 {
        ((piece_idx as u32) << 24)
            | ((orientation as u32) << 16)
            | ((row as u32) << 8)
            | (col as u32)
    }

    /// Decode a move from u32
    #[staticmethod]
    pub fn decode_move(move_id: u32) -> (u8, u8, u8, u8) {
        let piece_idx = ((move_id >> 24) & 0xFF) as u8;
        let orientation = ((move_id >> 16) & 0xFF) as u8;
        let row = ((move_id >> 8) & 0xFF) as u8;
        let col = (move_id & 0xFF) as u8;
        (piece_idx, orientation, row, col)
    }

    /// Get board as Python list
    pub fn get_board(&self) -> Vec<u8> {
        self.board.clone()
    }

    /// Set board from Python list
    pub fn set_board(&mut self, board: Vec<u8>) {
        self.board = board;
    }
}
